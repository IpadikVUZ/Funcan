\subsection{Изометричность вложения \(E\) в \(E^{**}\). Критерий слабой сходимости последовательности}

\begin{definition}[Слабая сходимость]
	Пусть \(E\) --- нормированное пространство. Последовательность
	\(\{x_n\}\subset E\) называется \textbf{слабо сходящейся} к \(x\in E\), если
	\[
	\forall f\in E^*
	\qquad
	f(x_n)\to f(x).
	\]
	Обозначение:
	\[
	x_n\weakconv x.
	\]
\end{definition}

\begin{remark}[Полезные свойства слабой сходимости]
	\begin{itemize}
		\item Если \(\|x_n-x\|\to0\), то \(x_n\weakconv x\), так как для любого \(f\in E^*\)
		\[
		|f(x_n)-f(x)|
		=
		|f(x_n-x)|
		\leqslant
		\|f\|\cdot \|x_n-x\|
		\to0.
		\]
		
		\item Слабый предел единственен. Действительно, если
		\[
		x_n\weakconv x',
		\qquad
		x_n\weakconv x'',
		\]
		то
		\[
		f(x')=f(x'')
		\qquad
		\forall f\in E^*.
		\]
		По следствию из теоремы Хана--Банаха функционалы из \(E^*\) разделяют точки, поэтому
		\[
		x'=x''.
		\]
		
		\item В конечномерном пространстве слабая сходимость эквивалентна сходимости по норме:
		\[
		x_n\weakconv x
		\quad\Longleftrightarrow\quad
		\|x_n-x\|\to0.
		\]
		Идея: слабая сходимость даёт сходимость координатных функционалов, а в конечномерном
		пространстве покоординатная сходимость эквивалентна сходимости по норме.
		
		\item В бесконечномерном пространстве слабая сходимость вообще говоря слабее сходимости по норме.
		Например, в \(\ell_2\)
		\[
		e_n=(0,\ldots,0,1,0,\ldots)
		\]
		слабо сходится к нулю, поскольку
		\[
		(e_n,y)=y_n\to0
		\qquad
		\forall y\in\ell_2,
		\]
		но
		\[
		\|e_n\|=1.
		\]
		
		\item В гильбертовом пространстве \(H\) по теореме Рисса--Фреше слабая сходимость проверяется через скалярное произведение:
		\[
		x_n\weakconv x
		\quad\Longleftrightarrow\quad
		\forall y\in H
		\qquad
		(x_n,y)\to(x,y).
		\]
	\end{itemize}
\end{remark}

\begin{definition}[Каноническое вложение \(E\) в \(E^{**}\)]
	Пусть \(E\) --- нормированное пространство. Для каждого \(x\in E\) определим функционал
	\[
	F_x:E^*\to\K,
	\qquad
	F_x(f)=f(x).
	\]
	Отображение
	\[
	J:E\to E^{**},
	\qquad
	Jx=F_x,
	\]
	называется \textbf{каноническим вложением} пространства \(E\) в \(E^{**}\).
\end{definition}

\begin{theorem}[Изометричность канонического вложения]
	Каноническое вложение
	\[
	J:E\to E^{**},
	\qquad
	(Jx)(f)=f(x),
	\]
	является линейной изометрией:
	\[
	\|Jx\|_{E^{**}}=\|x\|_E
	\qquad
	\forall x\in E.
	\]
\end{theorem}

\begin{proof}
	\textbf{1. Корректность определения \(Jx\).}
	
	Зафиксируем \(x\in E\). Нужно проверить, что
	\[
	F_x(f)=f(x)
	\]
	действительно задаёт элемент \(E^{**}\), то есть линейный ограниченный функционал на \(E^*\).
	
	Линейность по \(f\) следует из линейности функционалов:
	\[
	F_x(\alpha f+\beta g)
	=
	(\alpha f+\beta g)(x)
	=
	\alpha f(x)+\beta g(x)
	=
	\alpha F_x(f)+\beta F_x(g).
	\]
	Ограниченность:
	\[
	|F_x(f)|
	=
	|f(x)|
	\leqslant
	\|f\|\cdot\|x\|.
	\]
	Следовательно,
	\[
	F_x\in E^{**},
	\qquad
	\|F_x\|\leqslant \|x\|.
	\]
	
	\textbf{2. Линейность отображения \(J\).}
	
	Для любых \(x,y\in E\), \(\alpha,\beta\in\K\) и \(f\in E^*\) имеем
	\[
	J(\alpha x+\beta y)(f)
	=
	f(\alpha x+\beta y)
	=
	\alpha f(x)+\beta f(y)
	=
	(\alpha Jx+\beta Jy)(f).
	\]
	Значит,
	\[
	J(\alpha x+\beta y)=\alpha Jx+\beta Jy.
	\]
	
	\textbf{3. Верхняя оценка нормы.}
	
	Из оценки выше:
	\[
	\|Jx\|
	=
	\|F_x\|
	\leqslant
	\|x\|.
	\]
	
	\textbf{4. Нижняя оценка нормы.}
	
	Если \(x=0\), то равенство очевидно. Пусть \(x\ne0\). По следствию из теоремы Хана--Банаха
	существует функционал \(f_0\in E^*\), такой что
	\[
	\|f_0\|=1,
	\qquad
	|f_0(x)|=\|x\|.
	\]
	Тогда
	\[
	\|Jx\|
	=
	\sup_{\|f\|\leqslant1}|(Jx)(f)|
	\geqslant
	|(Jx)(f_0)|
	=
	|f_0(x)|
	=
	\|x\|.
	\]
	
	Итак,
	\[
	\|Jx\|\leqslant \|x\|
	\qquad
	\text{и}
	\qquad
	\|Jx\|\geqslant \|x\|.
	\]
	Следовательно,
	\[
	\|Jx\|=\|x\|.
	\]
	Отображение \(J\) линейно и сохраняет норму, значит, \(J\) --- линейная изометрия.
\end{proof}

\begin{remark}[Слабая сходимость как поточечная сходимость в \(E^{**}\)]
	По определению канонического вложения
	\[
	(Jx_n)(f)=f(x_n),
	\qquad
	(Jx)(f)=f(x).
	\]
	Поэтому
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\forall f\in E^*
	\qquad
	(Jx_n)(f)\to(Jx)(f).
	\]
	То есть слабая сходимость \(x_n\weakconv x\) означает поточечную сходимость функционалов
	\(Jx_n\) к \(Jx\) на пространстве \(E^*\).
\end{remark}

\begin{theorem}[Критерий поточечной сходимости операторов]
	Пусть \(X\) --- банахово пространство, \(Y\) --- нормированное пространство,
	\[
	A_n,A\in\mathcal{L}(X,Y),
	\]
	и пусть \(S\subset X\) таково, что
	\[
	\overline{\operatorname{span}S}=X.
	\]
	Тогда
	\[
	A_nx\to Ax
	\qquad
	\forall x\in X
	\]
	тогда и только тогда, когда выполнены условия:
	\[
	\{\|A_n\|\}\ \text{ограничена}
	\]
	и
	\[
	A_ns\to As
	\qquad
	\forall s\in S.
	\]
\end{theorem}

\begin{remark}
	Критерий поточечной сходимости является следствием теоремы Банаха--Штейнгауза.
	В этом билете он используется как готовый инструмент.
\end{remark}

\begin{corollary}[Слабо сходящаяся последовательность ограничена]
	Если
	\[
	x_n\weakconv x,
	\]
	то последовательность
	\[
	\{\|x_n\|\}
	\]
	ограничена.
\end{corollary}

\begin{proof}
	Из слабой сходимости следует, что
	\[
	Jx_n\to Jx
	\]
	поточечно на \(E^*\). По теореме Банаха--Штейнгауза получаем
	\[
	\sup_n\|Jx_n\|<\infty.
	\]
	Так как \(J\) --- изометрия,
	\[
	\|Jx_n\|=\|x_n\|.
	\]
	Значит,
	\[
	\sup_n\|x_n\|<\infty.
	\]
\end{proof}

\begin{theorem}[Критерий слабой сходимости последовательности]
	Пусть \(E\) --- нормированное пространство,
	\[
	S\subset E^*,
	\qquad
	\overline{\operatorname{span}S}=E^*.
	\]
	Тогда
	\[
	x_n\weakconv x
	\]
	тогда и только тогда, когда выполнены условия:
	\[
	\{\|x_n\|\}\ \text{ограничена}
	\]
	и
	\[
	s(x_n)\to s(x)
	\qquad
	\forall s\in S.
	\]
\end{theorem}

\begin{proof}
	\textbf{1. Сведение к поточечной сходимости операторов.}
	
	Рассмотрим канонические образы
	\[
	Jx_n,Jx\in E^{**}.
	\]
	По замечанию выше
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	Jx_n\to Jx
	\quad
	\text{поточечно на } E^*.
	\]
	
	\textbf{2. Применение критерия поточечной сходимости.}
	
	Применим критерий поточечной сходимости к пространствам
	\[
	X=E^*,
	\qquad
	Y=\K,
	\]
	и операторам
	\[
	A_n=Jx_n,
	\qquad
	A=Jx.
	\]
	Так как
	\[
	\overline{\operatorname{span}S}=E^*,
	\]
	получаем, что поточечная сходимость \(Jx_n\to Jx\) эквивалентна двум условиям:
	\[
	\{\|Jx_n\|\}\ \text{ограничена}
	\]
	и
	\[
	(Jx_n)(s)\to(Jx)(s)
	\qquad
	\forall s\in S.
	\]
	
	\textbf{3. Возврат к исходной последовательности.}
	
	Так как \(J\) --- изометрия,
	\[
	\|Jx_n\|=\|x_n\|.
	\]
	Кроме того, по определению \(J\)
	\[
	(Jx_n)(s)=s(x_n),
	\qquad
	(Jx)(s)=s(x).
	\]
	Значит, условия из критерия поточечной сходимости переписываются как
	\[
	\{\|x_n\|\}\ \text{ограничена}
	\]
	и
	\[
	s(x_n)\to s(x)
	\qquad
	\forall s\in S.
	\]
	Следовательно,
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\begin{cases}
		\{\|x_n\|\}\ \text{ограничена},\\
		s(x_n)\to s(x)\quad \forall s\in S.
	\end{cases}
	\]
\end{proof}

\begin{idea}[Скелет билета]
	Главная идея билета: слабую сходимость можно рассматривать как поточечную сходимость
	второсопряжённых функционалов.
	
	\begin{itemize}
		\item Слабая сходимость:
		\[
		x_n\weakconv x
		\Longleftrightarrow
		f(x_n)\to f(x)
		\quad
		\forall f\in E^*.
		\]
		
		\item Каноническое вложение:
		\[
		Jx=F_x,
		\qquad
		F_x(f)=f(x).
		\]
		
		\item Изометричность:
		\[
		\|Jx\|\leqslant\|x\|
		\]
		из оценки \(|f(x)|\leqslant\|f\|\|x\|\), а
		\[
		\|Jx\|\geqslant\|x\|
		\]
		по следствию из теоремы Хана--Банаха.
		
		\item Слабая сходимость \(x_n\weakconv x\) равносильна поточечной сходимости
		\[
		Jx_n\to Jx
		\]
		на \(E^*\).
		
		\item Критерий слабой сходимости получается из критерия поточечной сходимости операторов:
		\[
		\|Jx_n\|=\|x_n\|,
		\qquad
		(Jx_n)(s)=s(x_n).
		\]
	\end{itemize}
\end{idea}
