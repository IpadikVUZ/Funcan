\subsection{Норма сопряженного оператора (в линейном нормированном пространстве)}

\begin{definition}[Сопряженное пространство]
	Пусть \(E\) --- линейное нормированное пространство над полем \(\mathbb{K}\).
	Сопряженным пространством к \(E\) называется пространство всех линейных ограниченных
	функционалов на \(E\):
	\[
	E^*=\mathcal{L}(E,\mathbb{K}).
	\]
\end{definition}

\begin{definition}[Сопряженный оператор]
	Пусть \(E_1,E_2\) --- линейные нормированные пространства и
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Сопряженным к \(A\) называется оператор
	\[
	A^*:E_2^*\to E_1^*,
	\]
	который каждому функционалу \(g\in E_2^*\) сопоставляет функционал
	\[
	A^*g\in E_1^*
	\]
	по правилу
	\[
	(A^*g)(x)=g(Ax),
	\qquad
	x\in E_1.
	\]
	То есть
	\[
	A^*g=g\circ A.
	\]
\end{definition}


\begin{theorem}[Норма сопряженного оператора]
	Пусть \(E_1,E_2\) --- линейные нормированные пространства и
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Тогда сопряженный оператор
	\[
	A^*:E_2^*\to E_1^*
	\]
	существует, является линейным ограниченным оператором и
	\[
	\|A^*\|=\|A\|.
	\]
\end{theorem}

\subsection{Сопряженные операторы в гильбертовом пространстве. Равенство \texorpdfstring{\(H=\overline{\operatorname{Im} A}\oplus\ker A^*\)}{H = cl Im A plus ker A*}}

\begin{definition}[Сопряженный оператор в гильбертовом пространстве]
	Пусть \(H_1,H_2\) --- гильбертовы пространства и
	\[
	A\in\mathcal{L}(H_1,H_2).
	\]
	Оператор
	\[
	A^*:H_2\to H_1
	\]
	называется \textbf{сопряженным} к \(A\), если
	\[
	(Ax,y)_{H_2}=(x,A^*y)_{H_1}
	\qquad
	\forall x\in H_1,\ \forall y\in H_2.
	\]
\end{definition}

\begin{remark}[Свойства сопряженного оператора]
	Сопряженный оператор в гильбертовом пространстве существует и единственен. Также
	выполнены свойства:
	\[
	I^*=I,
	\]
	\[
	(\alpha A+\beta B)^*
	=
	\overline{\alpha}A^*+\overline{\beta}B^*,
	\]
	\[
	(AB)^*=B^*A^*.
	\]
	Они доказываются прямой проверкой из определения.
\end{remark}

\begin{theorem}[Разложение гильбертова пространства]
	Пусть \(H\) --- гильбертово пространство над \(\mathbb{K}\), и
	\[
	A\in\mathcal{L}(H).
	\]
	Тогда
	\[
	H=\overline{\operatorname{Im}A}\oplus\ker A^*.
	\]
\end{theorem}

\begin{proof}
	Докажем разложение через ортогональное дополнение к образу.

	\textbf{1. Простая часть: докажем равенство \((\operatorname{Im}A)^\perp=\ker A^*\).}

	Пусть
	\[
	y\in(\operatorname{Im}A)^\perp.
	\]
	Это означает, что
	\[
	(Az,y)=0
	\qquad
	\forall z\in H.
	\]
	По определению сопряженного оператора
	\[
	(Az,y)=(z,A^*y).
	\]
	Значит,
	\[
	(z,A^*y)=0
	\qquad
	\forall z\in H.
	\]
	Отсюда
	\[
	A^*y=0.
	\]
	Следовательно,
	\[
	y\in\ker A^*.
	\]

	Обратно, если
	\[
	y\in\ker A^*,
	\]
	то
	\[
	A^*y=0.
	\]
	Тогда для любого \(z\in H\)
	\[
	(Az,y)=(z,A^*y)=(z,0)=0.
	\]
	Значит,
	\[
	y\in(\operatorname{Im}A)^\perp.
	\]

	Итак,
	\[
	(\operatorname{Im}A)^\perp=\ker A^*.
	\]

	\textbf{2. Сложная часть: применим теорему о проекции.}

	Знаем, что для любого множества \(S\subset H\)
	\[
	S^\perp=\left(\overline{S}\right)^\perp.
	\]
	Поэтому
	\[
	(\operatorname{Im}A)^\perp
	=
	\left(\overline{\operatorname{Im}A}\right)^\perp.
	\]
	Так как
	\[
	\overline{\operatorname{Im}A}
	\]
	--- замкнутое подпространство в \(H\), то по теореме Рисса о проекции
	\[
	H
	=
	\overline{\operatorname{Im}A}
	\oplus
	\left(\overline{\operatorname{Im}A}\right)^\perp.
	\]
	Но
	\[
	\left(\overline{\operatorname{Im}A}\right)^\perp
	=
	(\operatorname{Im}A)^\perp
	=
	\ker A^*.
	\]
	Следовательно,
	\[
	H=\overline{\operatorname{Im}A}\oplus\ker A^*.
	\]
\end{proof}

\begin{remark}
	Аналогично, применяя доказанное равенство к оператору \(A^*\), получаем
	\[
	H=\overline{\operatorname{Im}A^*}\oplus\ker A.
	\]
\end{remark}

\subsection{Свойства квадратичной формы \texorpdfstring{\((Ax,x)\)}{(Ax,x)} и собственных значений самосопряженного оператора \texorpdfstring{\(A\)}{A}}

\begin{definition}[Самосопряженный оператор]
	Пусть \(H\) --- гильбертово пространство над \(\mathbb{C}\), и
	\[
	A\in\mathcal{L}(H).
	\]
	Оператор \(A\) называется \textbf{самосопряженным}, если
	\[
	A^*=A.
	\]
	Эквивалентно,
	\[
	(Ax,y)=(x,Ay)
	\qquad
	\forall x,y\in H.
	\]
\end{definition}

\begin{definition}[Квадратичная форма самосопряженного оператора]
	Пусть \(A\in\mathcal{L}(H)\). С оператором \(A\) связывают квадратичную форму
	\[
	K:H\to\mathbb{C},
	\]
	заданную правилом
	\[
	K(x)=(Ax,x).
	\]
	Идея состоит в том, что вместо оператора \(A\) часто удобно изучать значения формы
	\[
	(Ax,x).
	\]
\end{definition}

\begin{theorem}[Свойства квадратичной формы и собственных значений]
	Пусть \(H\) --- гильбертово пространство над \(\mathbb{C}\), и
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда:

	\begin{itemize}
		\item для любого \(x\in H\)
		\[
		(Ax,x)\in\mathbb{R};
		\]

		\item если \(\lambda\) --- собственное значение оператора \(A\), то
		\[
		\lambda\in\mathbb{R};
		\]

		\item если \(\lambda_1\neq\lambda_2\) --- собственные значения оператора \(A\), а
		\(e_1,e_2\) --- соответствующие им собственные векторы, то
		\[
		(e_1,e_2)=0.
		\]
	\end{itemize}
\end{theorem}

\begin{remark}[Обратное утверждение]
	Верно и обратное: если для любого \(x\in H\)
	\[
	(Ax,x)\in\mathbb{R},
	\]
	то оператор \(A\) самосопряжен.

	Это доказывается поляризацией: рассматривают значения формы на \(x+y\) и \(x+iy\), после чего
	из вещественности соответствующих выражений получают
	\[
	(Ax,y)=(x,Ay).
	\]
	Для основного билета это утверждение обычно можно оставить как дополнительное.
\end{remark}

\subsection{Разложение гильбертова пространства \texorpdfstring{\(H=\overline{\operatorname{Im}(A-\lambda I)}\oplus\ker(A-\lambda I)\)}{H = cl Im(A-lambda I) plus Ker(A-lambda I)}, где \texorpdfstring{\(A\)}{A} --- самосопряженный оператор}

\begin{definition}[Оператор \(A_\lambda\)]
	Пусть \(H\) --- гильбертово пространство над \(\mathbb{C}\),
	\[
	A\in\mathcal{L}(H),
	\qquad
	\lambda\in\mathbb{C}.
	\]
	Обозначим
	\[
	A_\lambda=A-\lambda I.
	\]
\end{definition}

\begin{theorem}
	Пусть \(H\) --- гильбертово пространство над \(\mathbb{C}\), и
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда для любого \(\lambda\in\mathbb{C}\)
	\[
	H=\overline{\operatorname{Im}A_\lambda}\oplus_{\perp}\ker A_\lambda.
	\]
	То есть
	\[
	H
	=
	\overline{\operatorname{Im}(A-\lambda I)}
	\oplus_{\perp}
	\ker(A-\lambda I).
	\]
\end{theorem}

\begin{proof}
	Будем использовать разложение из билета 10.2:
	\[
	H=\overline{\operatorname{Im}B}\oplus_{\perp}\ker B^*
	\]
	для любого
	\[
	B\in\mathcal{L}(H).
	\]
	Применим его к оператору
	\[
	B=A_\lambda=A-\lambda I.
	\]
	Тогда
	\[
	H=\overline{\operatorname{Im}A_\lambda}\oplus_{\perp}\ker A_\lambda^*.
	\]
	Остаётся понять, почему в данном случае
	\[
	\ker A_\lambda^*=\ker A_\lambda.
	\]

	\textbf{1. Случай \(\lambda\in\mathbb{R}\).}

	Так как \(A\) самосопряжен, то
	\[
	A^*=A.
	\]
	Кроме того,
	\[
	(\lambda I)^*=\overline{\lambda}I.
	\]
	Поэтому
	\[
	A_\lambda^*
	=
	(A-\lambda I)^*
	=
	A^*-\overline{\lambda}I
	=
	A-\overline{\lambda}I.
	\]
	Если \(\lambda\in\mathbb{R}\), то
	\[
	\overline{\lambda}=\lambda.
	\]
	Значит,
	\[
	A_\lambda^*=A-\lambda I=A_\lambda.
	\]
	Следовательно,
	\[
	\ker A_\lambda^*=\ker A_\lambda.
	\]
	Подставляя это в разложение из 10.2, получаем
	\[
	H=\overline{\operatorname{Im}A_\lambda}\oplus_{\perp}\ker A_\lambda.
	\]

	\textbf{2. Случай \(\lambda\notin\mathbb{R}\).}

	По формуле выше
	\[
	A_\lambda^*=A-\overline{\lambda}I.
	\]
	Так как \(A\) самосопряжен, по билету 11.1 все собственные значения \(A\) вещественны.

	Но
	\[
	\lambda\notin\mathbb{R},
	\qquad
	\overline{\lambda}\notin\mathbb{R}.
	\]
	Значит, ни \(\lambda\), ни \(\overline{\lambda}\) не являются собственными значениями оператора \(A\).
	Следовательно,
	\[
	\ker(A-\lambda I)=\{0\},
	\]
	и
	\[
	\ker(A-\overline{\lambda}I)=\{0\}.
	\]
	То есть
	\[
	\ker A_\lambda=\{0\},
	\qquad
	\ker A_\lambda^*=\{0\}.
	\]

	Из разложения из 10.2 имеем
	\[
	H=\overline{\operatorname{Im}A_\lambda}\oplus_{\perp}\ker A_\lambda^*.
	\]
	Но \(\ker A_\lambda^*=\{0\}\), значит
	\[
	H=\overline{\operatorname{Im}A_\lambda}.
	\]
	Так как одновременно \(\ker A_\lambda=\{0\}\), можно записать
	\[
	H=\overline{\operatorname{Im}A_\lambda}\oplus_{\perp}\ker A_\lambda.
	\]
\end{proof}

\subsection{Критерий принадлежности числа спектру самосопряженного оператора.
	Вещественность спектра самосопряженного оператора}

\begin{definition}[Оператор \(A_\lambda\)]
	Пусть
	\[
	A\in\mathcal{L}(H),
	\qquad
	\lambda\in\mathbb{C}.
	\]
	Обозначим
	\[
	A_\lambda=A-\lambda I.
	\]
\end{definition}

\begin{theorem}[Критерий принадлежности числа спектру]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда
	\[
	\lambda\in\rho(A)
	\Longleftrightarrow
	A_\lambda \text{ ограничен снизу},
	\]
	то есть
	\[
	\exists m>0
	\quad
	\forall x\in H
	\qquad
	\|A_\lambda x\|\geqslant m\|x\|.
	\]

	Кроме того,
	\[
	\lambda\in\sigma(A)
	\Longleftrightarrow
	\exists \{x_n\}\subset H:
	\quad
	\|x_n\|=1,
	\qquad
	A_\lambda x_n\to0.
	\]
\end{theorem}

\begin{idea}
	Иными словами, $x_n$ становятся всё более похожи на собственные векторы с собственным значением $\lambda$, хотя настоящего собственного вектора может не существовать.
\end{idea}

\begin{proof}
	\textbf{1. Докажем критерий для резольвенты.}

	\textbf{1.1. Пусть \(\lambda\in\rho(A)\). Покажем, что \(A_\lambda\) ограничен снизу.}

	По определению резольвентного множества
	\[
	A_\lambda^{-1}\in\mathcal{L}(H).
	\]
	По билету 8.1 отсюда следует, что \(A_\lambda\) ограничен снизу:
	\[
	\exists m>0
	\quad
	\forall x\in H
	\qquad
	\|A_\lambda x\|\geqslant m\|x\|.
	\]

	\textbf{1.2. Пусть \(A_\lambda\) ограничен снизу. Покажем, что
		\(\lambda\in\rho(A)\).}

	Для этого нужно показать, что \(A_\lambda\) непрерывно обратим.

	Из ограниченности снизу по билету 8.1 получаем
	\[
	\ker A_\lambda=\{0\}
	\]
	и
	\[
	\operatorname{Im}A_\lambda
	\quad\text{замкнут}.
	\]

	Теперь осталось показать сюръективность. По билету 11.2
\[
H=
\overline{\operatorname{Im}A_\lambda}
\oplus_{\perp}
\ker A_\lambda.
\]
Так как
\[
\ker A_\lambda=\{0\},
\qquad
\operatorname{Im}A_\lambda
\text{ замкнут},
\]
то
\[
\operatorname{Im}A_\lambda=H.
\]

	Значит, \(A_\lambda\) --- биекция. Так как он ограничен снизу,
	по билету 8.1
	\[
	A_\lambda^{-1}\in\mathcal{L}(H).
	\]
	Следовательно,
	\[
	\lambda\in\rho(A).
	\]

	Итак,
	\[
	\boxed{
		\lambda\in\rho(A)
		\Longleftrightarrow
		A_\lambda\text{ ограничен снизу}.
	}
	\]

	\textbf{2. Докажем критерий для спектра.}

	Из первого пункта
	\[
	\lambda\in\sigma(A)
	\Longleftrightarrow
	A_\lambda\text{ не ограничен снизу}.
	\]

	\textbf{2.1. Пусть \(\lambda\in\sigma(A)\). Построим последовательность \(x_n\).}

	Так как \(A_\lambda\) не ограничен снизу,
	\[
	\forall m>0
	\quad
	\exists u\neq0:
	\qquad
	\|A_\lambda u\|<m\|u\|.
	\]
	Для
	\[
	m=\frac1n
	\]
	выберем \(u_n\neq0\), такое что
	\[
	\|A_\lambda u_n\|
	<
	\frac1n\|u_n\|.
	\]
	Положим
	\[
	x_n=\frac{u_n}{\|u_n\|}.
	\]
	Тогда
	\[
	\|x_n\|=1
	\]
	и
	\[
	\|A_\lambda x_n\|
	<
	\frac1n.
	\]
	Следовательно,
	\[
	A_\lambda x_n\to0.
	\]

	\textbf{2.2. Пусть существует \(x_n\), такое что
		\(\|x_n\|=1\) и \(A_\lambda x_n\to0\).
		Покажем, что \(\lambda\in\sigma(A)\).}

	Если бы \(A_\lambda\) был ограничен снизу, то для некоторого \(m>0\)
	\[
	\|A_\lambda x_n\|
	\geqslant
	m\|x_n\|
	=
	m
	\]
	для всех \(n\), что противоречит
	\[
	A_\lambda x_n\to0.
	\]
	Значит, \(A_\lambda\) не ограничен снизу, поэтому
	\[
	\lambda\in\sigma(A).
	\]

	Итак,
	\[
	\boxed{
		\lambda\in\sigma(A)
		\Longleftrightarrow
		\exists x_n:
		\quad
		\|x_n\|=1,
		\qquad
		A_\lambda x_n\to0.
	}
	\]
\end{proof}

\begin{theorem}[Вещественность спектра]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда
	\[
	\sigma(A)\subset\mathbb{R}.
	\]
\end{theorem}

\begin{proof}
	Докажем равносильное утверждение:
	\[
	\lambda\notin\mathbb{R}
	\Longrightarrow
	\lambda\in\rho(A).
	\]

	Пусть
	\[
	\lambda=\mu+i\nu,
	\qquad
	\mu,\nu\in\mathbb{R},
	\qquad
	\nu\neq0.
	\]
	По критерию принадлежности числа резольвенте достаточно показать,
	что \(A_\lambda\) ограничен снизу.

	Положим
	\[
	A_\mu=A-\mu I.
	\]
	Так как \(A\) самосопряжен и \(\mu\in\mathbb{R}\),
	оператор \(A_\mu\) самосопряжен.

	Для любого \(x\in H\)
	\[
	A_\lambda x=A_\mu x-i\nu x,
	\]
	поэтому
	\[
	\begin{aligned}
		\|A_\lambda x\|^2
		&=
		\|A_\mu x-i\nu x\|^2
		\\
		&=
		\|A_\mu x\|^2
		-i\nu(x,A_\mu x)
		+i\nu(A_\mu x,x)
		+\nu^2\|x\|^2.
	\end{aligned}
	\]

	Так как \(A_\mu\) самосопряжен,
	\[
	(x,A_\mu x)=(A_\mu x,x).
	\]
	Поэтому смешанные слагаемые сокращаются:
	\[
	\|A_\lambda x\|^2
	=
	\|A_\mu x\|^2+\nu^2\|x\|^2
	\geqslant
	\nu^2\|x\|^2.
	\]
	Значит,
	\[
	\|A_\lambda x\|
	\geqslant
	|\nu|\|x\|.
	\]

	То есть \(A_\lambda\) ограничен снизу, поэтому
	\[
	\lambda\in\rho(A).
	\]
	Следовательно,
	\[
	\boxed{\sigma(A)\subset\mathbb{R}}.
	\]
\end{proof}

\subsection{Теорема о спектре самосопряженного оператора:
	\texorpdfstring{\(\sigma(A)\subset[m_-,m_+]\), \(r(A)=\|A\|\)}
	{sigma(A) subset [m-,m+], r(A)=||A||}}

\begin{definition}
	Пусть \(A\in\mathcal L(H)\) --- самосопряженный оператор. Положим
	\[
	m_-=\inf_{\|x\|=1}(Ax,x),
	\qquad
	m_+=\sup_{\|x\|=1}(Ax,x).
	\]
	Так как
	\[
	(Ax,x)\in\mathbb R,
	\qquad
	|(Ax,x)|\leqslant\|A\|
	\quad (\|x\|=1),
	\]
	числа \(m_-,m_+\) корректно определены.
\end{definition}

\begin{lemma}[Норма квадрата самосопряженного оператора]
	Если \(A=A^*\), то
	\[
	\|A^2\|=\|A\|^2.
	\]
\end{lemma}

\begin{proof}
	Оценка
	\[
	\|A^2\|\leqslant\|A\|^2
	\]
	очевидна.

	Для \(\|x\|=1\)
	\[
	\|Ax\|^2
	=
	(A^2x,x)
	\leqslant
	\|A^2\|.
	\]
	Беря супремум по \(\|x\|=1\), получаем
	\[
	\|A\|^2\leqslant\|A^2\|.
	\]
	Следовательно,
	\[
	\|A^2\|=\|A\|^2.
	\]
\end{proof}

\begin{remark}[Полускалярное произведение]
	Если \(B=B^*\) и
	\[
	(Bx,x)\geqslant0
	\qquad
	\forall x\in H,
	\]
	то
	\[
	\langle x,y\rangle_B=(Bx,y)
	\]
	задаёт полускалярное произведение, для которого
	\[
	|\langle x,y\rangle_B|^2
	\leqslant
	\langle x,x\rangle_B
	\langle y,y\rangle_B.
	\]
\end{remark}

\begin{theorem}[Спектр самосопряженного оператора]
	Пусть \(A\in\mathcal L(H)\) --- самосопряженный оператор. Тогда
	\[
	\sigma(A)\subset[m_-,m_+],
	\qquad
	m_-,m_+\in\sigma(A),
	\]
	и
	\[
	r(A)=\max\{|m_-|,|m_+|\}=\|A\|.
	\]
\end{theorem}

\begin{proof}
	\textbf{1. Докажем, что \(\sigma(A)\subset[m_-,m_+]\).}

	По предыдущему билету
	\[
	\sigma(A)\subset\mathbb R.
	\]
	Поэтому достаточно показать, что при
	\[
	\lambda>m_+
	\quad\text{или}\quad
	\lambda<m_-
	\]
	оператор \(A_\lambda=A-\lambda I\) ограничен снизу.

	Пусть \(\lambda>m_+\). Для \(x\neq0\)
	\[
	\begin{aligned}
		\|A_\lambda x\|\|x\|
		&\geqslant
		|(A_\lambda x,x)|
		\\
		&=
		\lambda\|x\|^2-(Ax,x)
		\\
		&\geqslant
		(\lambda-m_+)\|x\|^2.
	\end{aligned}
	\]
	Следовательно,
	\[
	\|A_\lambda x\|
	\geqslant
	(\lambda-m_+)\|x\|.
	\]
	По критерию предыдущего билета
	\[
	\lambda\in\rho(A).
	\]

	При \(\lambda<m_-\) аналогично
	\[
	\|A_\lambda x\|
	\geqslant
	(m_--\lambda)\|x\|,
	\]
	поэтому также
	\[
	\lambda\in\rho(A).
	\]

	Значит,
	\[
	\boxed{\sigma(A)\subset[m_-,m_+]}.
	\]

	\textbf{2. Докажем, что \(m_+,m_-\in\sigma(A)\).}

	Сначала покажем
	\[
	m_+\in\sigma(A).
	\]
	По определению супремума существует последовательность
	\[
	\|x_n\|=1,
	\qquad
	(Ax_n,x_n)\to m_+.
	\]

	По критерию принадлежности спектру хотим получить
	\[
	(A-m_+I)x_n\to0.
	\]
	Для перехода от сходимости скалярных произведений к сходимости по норме положим
	\[
	B=m_+I-A.
	\]
	Тогда \(B=B^*\) и
	\[
	(Bx,x)\geqslant0
	\qquad
	\forall x\in H.
	\]

	По неравенству Коши--Буняковского для полускалярного произведения
	\[
	\begin{aligned}
		\|Bx_n\|^4
		&=
		|\langle x_n,Bx_n\rangle_B|^2
		\\
		&\leqslant
		\langle x_n,x_n\rangle_B
		\langle Bx_n,Bx_n\rangle_B
		\\
		&\leqslant
		\bigl(m_+-(Ax_n,x_n)\bigr)\|B\|^3
		\to0.
	\end{aligned}
	\]
	Следовательно,
	\[
	Bx_n\to0,
	\]
	то есть
	\[
	(A-m_+I)x_n\to0.
	\]
	Так как \(\|x_n\|=1\), по критерию принадлежности спектру
	\[
	m_+\in\sigma(A).
	\]

	Аналогично, взяв
	\[
	(Ay_n,y_n)\to m_-,
	\qquad
	B=A-m_-I,
	\]
	получаем
	\[
	m_-\in\sigma(A).
	\]

	Таким образом,
	\[
	\sigma(A)\subset[m_-,m_+],
	\qquad
	m_-,m_+\in\sigma(A),
	\]
	поэтому
	\[
	r(A)=\max\{|m_-|,|m_+|\}.
	\]

	\textbf{3. Докажем, что \(r(A)=\|A\|\).}

	По лемме
	\[
	\|A^2\|=\|A\|^2.
	\]
	Так как \(A^{2^k}\) самосопряжен, последовательным применением леммы
	\[
	\|A^{2^k}\|=\|A\|^{2^k}.
	\]

	По формуле спектрального радиуса
	\[
	r(A)
	=
	\lim_{n\to\infty}\sqrt[n]{\|A^n\|}.
	\]
	По подпоследовательности \(n=2^k\)
	\[
	r(A)
	=
	\lim_{k\to\infty}
	\sqrt[2^k]{\|A^{2^k}\|}
	=
	\|A\|.
	\]

	Итак,
	\[
	\boxed{
		r(A)=\max\{|m_-|,|m_+|\}=\|A\|.
	}
	\]

\end{proof}

\begin{idea}
	\textbf{Что вообще происходит в этом билете.}

	Пусть
	\[
	A=A^*\in\mathcal L(H).
	\]

	Для единичного вектора \(x\) рассмотрим число
	\[
	(Ax,x).
	\]

	Так как \(A\) самосопряжен,
	\[
	(Ax,x)\in\mathbb R.
	\]

	Поэтому можно посмотреть, насколько маленьким и насколько большим
	может быть это число:
	\[
	m_-=\inf_{\|x\|=1}(Ax,x),
	\qquad
	m_+=\sup_{\|x\|=1}(Ax,x).
	\]

	Иными словами,
	\[
	\boxed{
		m_-\leq (Ax,x)\leq m_+
		\qquad
		\forall\,\|x\|=1.
	}
	\]

	Для произвольного \(x\neq0\), применяя это к
	\[
	\frac{x}{\|x\|},
	\]
	получаем более удобную форму:
	\[
	\boxed{
		m_-\|x\|^2
		\leq
		(Ax,x)
		\leq
		m_+\|x\|^2.
	}
	\]

	Главная цель билета:
	\[
	\boxed{
		\sigma(A)\subset[m_-,m_+],
		\qquad
		m_-,m_+\in\sigma(A).
	}
	\]

	После этого спектральный радиус сразу будет
	\[
	r(A)=\max\{|m_-|,|m_+|\},
	\]
	а отдельно докажем, что для самосопряженного оператора
	\[
	r(A)=\|A\|.
	\]

	\medskip

	\textbf{1. Сначала хотим понять, почему спектр не может выйти за
		\([m_-,m_+]\).}

	Возьмем, например,
	\[
	\lambda>m_+.
	\]

	Хотим доказать:
	\[
	\lambda\notin\sigma(A),
	\]
	то есть
	\[
	\lambda\in\rho(A).
	\]

	Для этого рассматриваем
	\[
	A_\lambda=A-\lambda I.
	\]

	Из определения \(m_+\) знаем:
	\[
	(Ax,x)\leq m_+\|x\|^2.
	\]

	Поэтому
	\[
	\begin{aligned}
		(A_\lambda x,x)
		&=
		(Ax,x)-\lambda\|x\|^2
		\\
		&\leq
		-(\lambda-m_+)\|x\|^2.
	\end{aligned}
	\]

	Следовательно,
	\[
	|(A_\lambda x,x)|
	\geq
	(\lambda-m_+)\|x\|^2.
	\]

	С другой стороны, по Коши--Буняковскому
	\[
	|(A_\lambda x,x)|
	\leq
	\|A_\lambda x\|\|x\|.
	\]

	Объединяем:
	\[
	\|A_\lambda x\|\|x\|
	\geq
	(\lambda-m_+)\|x\|^2.
	\]

	Для \(x\neq0\):
	\[
	\boxed{
		\|A_\lambda x\|
		\geq
		(\lambda-m_+)\|x\|.
	}
	\]

	То есть \(A_\lambda\) ограничен снизу.

	По критерию из предыдущего билета для самосопряженного оператора
	это означает
	\[
	\lambda\in\rho(A).
	\]

	То же самое при
	\[
	\lambda<m_-:
	\]
	получаем
	\[
	\|A_\lambda x\|
	\geq
	(m_--\lambda)\|x\|,
	\]
	а значит,
	\[
	\lambda\in\rho(A).
	\]

	Таким образом, все точки вне отрезка являются точками резольвенты:
	\[
	(-\infty,m_-)\cup(m_+,\infty)
	\subset\rho(A).
	\]

	Следовательно,
	\[
	\boxed{
		\sigma(A)\subset[m_-,m_+].
	}
	\]

	Смысл этого шага очень простой:

	если \(\lambda\) находится правее \(m_+\), то оператор
	\[
	A-\lambda I
	\]
	уже ``слишком далеко от нуля'':
	\[
	\|(A-\lambda I)x\|
	\geq c\|x\|.
	\]

	Поэтому такая \(\lambda\) не может принадлежать спектру.

	\medskip

	\textbf{2. Теперь главный вопрос: почему сами границы \(m_+\) и \(m_-\)
		обязательно принадлежат спектру?}

	Разберем \(m_+\).

	По определению
	\[
	m_+
	=
	\sup_{\|x\|=1}(Ax,x).
	\]

	Супремум не обязан достигаться, поэтому мы не можем сразу взять
	вектор \(x\), для которого
	\[
	(Ax,x)=m_+.
	\]

	Но можем взять последовательность, приближающую супремум:
	\[
	\|x_n\|=1,
	\qquad
	(Ax_n,x_n)\to m_+.
	\]

	Хотим доказать
	\[
	m_+\in\sigma(A).
	\]

	По критерию принадлежности спектру достаточно получить
	приближенные собственные векторы:
	\[
	\boxed{
		\|(A-m_+I)x_n\|\to0.
	}
	\]

	И вот здесь возникает главная проблема билета.

	Мы знаем только
	\[
	(Ax_n,x_n)\to m_+.
	\]

	То есть знаем про одно число --- скалярное произведение.

	А нам нужна гораздо более сильная вещь:
	\[
	(A-m_+I)x_n\to0
	\]
	по норме.

	Напрямую из
	\[
	((A-m_+I)x_n,x_n)\to0
	\]
	это вообще не следует.

	Поэтому нужен специальный прием.

	\medskip

	\textbf{3. Хотим превратить приближение \(m_+\) в нормовую сходимость.}

	Для этого вводим
	\[
	B=m_+I-A.
	\]

	Почему именно такой оператор?

	Потому что из определения \(m_+\)
	\[
	(Ax,x)\leq m_+\|x\|^2.
	\]

	Следовательно,
	\[
	(Bx,x)
	=
	m_+\|x\|^2-(Ax,x)
	\geq0.
	\]

	То есть
	\[
	\boxed{B\geq0.}
	\]

	Кроме того, для нашей последовательности
	\[
	(Bx_n,x_n)
	=
	m_+-(Ax_n,x_n)
	\to0.
	\]

	Таким образом, мы получили положительный оператор \(B\) и знаем, что
	\[
	(Bx_n,x_n)\to0.
	\]

	Хотим отсюда вывести
	\[
	Bx_n\to0.
	\]

	Но это уже ровно
	\[
	(m_+I-A)x_n\to0,
	\]
	то есть то, что нам нужно.

	\medskip

	\textbf{4. Зачем здесь появляется полускалярное произведение?}

	Для положительного оператора \(B\) выражение
	\[
	\langle x,y\rangle_B=(Bx,y)
	\]
	ведет себя как скалярное произведение, возможно вырожденное.

	Поэтому для него справедливо неравенство Коши--Буняковского:
	\[
	|\langle x,y\rangle_B|^2
	\leq
	\langle x,x\rangle_B
	\langle y,y\rangle_B.
	\]

	Наша цель --- получить норму
	\[
	\|Bx_n\|.
	\]

	Поэтому в это неравенство подставляем
	\[
	x=x_n,
	\qquad
	y=Bx_n.
	\]

	Тогда левая часть:
	\[
	\langle x_n,Bx_n\rangle_B
	=
	(Bx_n,Bx_n)
	=
	\|Bx_n\|^2.
	\]

	Значит,
	\[
	|\langle x_n,Bx_n\rangle_B|^2
	=
	\|Bx_n\|^4.
	\]

	А первый множитель справа:
	\[
	\langle x_n,x_n\rangle_B
	=
	(Bx_n,x_n)
	=
	m_+-(Ax_n,x_n)
	\to0.
	\]

	Второй множитель
	\[
	\langle Bx_n,Bx_n\rangle_B
	=
	(B^2x_n,Bx_n)
	\]
	просто равномерно ограничен.

	Действительно, так как
	\[
	\|x_n\|=1,
	\]
	имеем
	\[
	\begin{aligned}
		|(B^2x_n,Bx_n)|
		&\leq
		\|B^2x_n\|\,\|Bx_n\|
		\\
		&\leq
		\|B\|^2\|x_n\|\,
		\|B\|\|x_n\|
		\\
		&=
		\|B\|^3.
	\end{aligned}
	\]

	Поэтому
	\[
	\|Bx_n\|^4
	\leq
	\bigl(m_+-(Ax_n,x_n)\bigr)\|B\|^3
	\to0.
	\]

	Следовательно,
	\[
	\boxed{Bx_n\to0.}
	\]

	Но
	\[
	B=m_+I-A,
	\]
	поэтому
	\[
	(A-m_+I)x_n\to0.
	\]

	При этом
	\[
	\|x_n\|=1.
	\]

	По критерию принадлежности спектру
	\[
	\boxed{m_+\in\sigma(A).}
	\]

	\medskip

	\textbf{То есть смысл всего приема с \(B\) следующий:}

	мы знаем лишь
	\[
	(Ax_n,x_n)\to m_+,
	\]
	или, что то же самое,
	\[
	(Bx_n,x_n)\to0.
	\]

	Положительность \(B\) позволяет усилить это до
	\[
	Bx_n\to0.
	\]

	Именно для этого здесь нужны
	\[
	B=m_+I-A
	\]
	и полускалярное произведение.

	\medskip

	\textbf{5. Для \(m_-\) происходит абсолютно то же самое.}

	По определению инфимума выбираем
	\[
	\|y_n\|=1,
	\qquad
	(Ay_n,y_n)\to m_-.
	\]

	Теперь берем
	\[
	B=A-m_-I.
	\]

	Почему знак поменялся?

	Потому что
	\[
	(Ax,x)\geq m_-\|x\|^2.
	\]

	Следовательно,
	\[
	(Bx,x)
	=
	(Ax,x)-m_-\|x\|^2
	\geq0.
	\]

	И для \(y_n\)
	\[
	(By_n,y_n)
	=
	(Ay_n,y_n)-m_-
	\to0.
	\]

	Тем же рассуждением получаем
	\[
	By_n\to0,
	\]
	то есть
	\[
	(A-m_-I)y_n\to0.
	\]

	По критерию спектра
	\[
	\boxed{m_-\in\sigma(A).}
	\]

	\medskip

	\textbf{6. Что мы уже получили про спектральный радиус?}

	Мы доказали одновременно
	\[
	\sigma(A)\subset[m_-,m_+]
	\]
	и
	\[
	m_-,m_+\in\sigma(A).
	\]

	Спектральный радиус определяется как
	\[
	r(A)
	=
	\sup_{\lambda\in\sigma(A)}|\lambda|.
	\]

	Весь спектр лежит на отрезке
	\[
	[m_-,m_+],
	\]
	а обе его крайние точки принадлежат спектру.

	Поэтому самое большое значение модуля достигается на одном из концов:
	\[
	\boxed{
		r(A)=\max\{|m_-|,|m_+|\}.
	}
	\]

	\medskip

	\textbf{7. Осталось понять, почему для самосопряженного оператора
		\(r(A)=\|A\|\).}

	Для произвольного ограниченного оператора всегда
	\[
	r(A)\leq\|A\|.
	\]

	Но равенство вообще говоря не обязано выполняться.

	У самосопряженного оператора есть специальное свойство:
	\[
	\boxed{\|A^2\|=\|A\|^2.}
	\]

	Почему?

	Одна сторона следует из свойства операторной нормы:
	\[
	\|A^2\|\leq\|A\|^2.
	\]

	Для обратной оценки используем самосопряженность:
	\[
	\|Ax\|^2
	=
	(Ax,Ax)
	=
	(A^2x,x).
	\]

	При \(\|x\|=1\)
	\[
	\|Ax\|^2
	\leq
	\|A^2\|.
	\]

	Берем супремум:
	\[
	\|A\|^2\leq\|A^2\|.
	\]

	Значит,
	\[
	\boxed{\|A^2\|=\|A\|^2.}
	\]

	Теперь замечаем, что
	\[
	A^2,A^4,A^8,\ldots
	\]
	тоже самосопряжены.

	Поэтому повторяем равенство:
	\[
	\|A^{2^k}\|
	=
	\|A\|^{2^k}.
	\]

	\medskip

	По формуле спектрального радиуса
	\[
	r(A)
	=
	\lim_{n\to\infty}
	\|A^n\|^{1/n}.
	\]

	Значит, тот же предел можно взять по подпоследовательности
	\[
	n=2^k:
	\]
	\[
	\begin{aligned}
		r(A)
		&=
		\lim_{k\to\infty}
		\|A^{2^k}\|^{1/2^k}
		\\
		&=
		\lim_{k\to\infty}
		\left(
		\|A\|^{2^k}
		\right)^{1/2^k}
		\\
		&=
		\|A\|.
	\end{aligned}
	\]

	Следовательно,
	\[
	\boxed{r(A)=\|A\|.}
	\]

	\medskip

	\textbf{8. Вся логика билета целиком.}

	Сначала числа
	\[
	m_-,m_+
	\]
	описывают крайние значения квадратичной формы
	\[
	(Ax,x)
	\]
	на единичной сфере.

	Если
	\[
	\lambda>m_+
	\quad\text{или}\quad
	\lambda<m_-,
	\]
	то
	\[
	A-\lambda I
	\]
	ограничен снизу, поэтому
	\[
	\lambda\notin\sigma(A).
	\]

	Отсюда
	\[
	\sigma(A)\subset[m_-,m_+].
	\]

	Далее берем последовательности, почти достигающие супремума и инфимума:
	\[
	(Ax_n,x_n)\to m_+,
	\qquad
	(Ay_n,y_n)\to m_-.
	\]

	Положительность операторов
	\[
	m_+I-A\geq0,
	\qquad
	A-m_-I\geq0
	\]
	позволяет усилить эти скалярные сходимости до
	\[
	(A-m_+I)x_n\to0,
	\qquad
	(A-m_-I)y_n\to0.
	\]

	Следовательно,
	\[
	m_+,m_-\in\sigma(A).
	\]

	Поэтому
	\[
	r(A)=\max\{|m_-|,|m_+|\}.
	\]

	Наконец,
	\[
	\|A^2\|=\|A\|^2
	\]
	и формула спектрального радиуса дают
	\[
	r(A)=\|A\|.
	\]

	Итого:
	\[
	\boxed{
		\sigma(A)\subset[m_-,m_+],
		\qquad
		m_-,m_+\in\sigma(A),
		\qquad
		r(A)=\max\{|m_-|,|m_+|\}=\|A\|.
	}
	\]

	\medskip

	\textbf{Главная картинка для запоминания:}

	\[
	\boxed{
		m_-
		\ \leq\
		\frac{(Ax,x)}{\|x\|^2}
		\ \leq\
		m_+
	}
	\]

	задает ``разрешенный диапазон'' для спектра.

	Точки снаружи:
	\[
	\lambda\notin[m_-,m_+]
	\]
	слишком далеко от квадратичной формы, поэтому
	\[
	A-\lambda I
	\]
	ограничен снизу и обратим.

	А к самим краям \(m_-\) и \(m_+\) можно подобрать векторы,
	для которых
	\[
	Ax\approx m_\pm x.
	\]

	То есть края являются приближенными собственными значениями,
	а значит принадлежат спектру.
\end{idea}

\subsection{Свойства компактных операторов}

\begin{definition}[Предкомпактное множество]
	Множество
	\[
	M\subset E
	\]
	называется предкомпактным, если его замыкание
	\[
	\overline M
	\]
	компактно.
\end{definition}

\begin{definition}[Компактный оператор]
	Пусть
	\[
	A\in\mathcal L(E_1,E_2).
	\]
	Оператор \(A\) называется компактным, если образ любого ограниченного
	множества является предкомпактным:
	\[
	M\subset E_1 \text{ ограничено}
	\quad\Longrightarrow\quad
	A(M)\subset E_2 \text{ предкомпактно}.
	\]

	Множество компактных операторов обозначается
	\[
	\mathcal K(E_1,E_2).
	\]
\end{definition}

\begin{remark}[Критерии компактности]
	Для линейного ограниченного оператора достаточно проверять единичный шар:
	\[
	A\in\mathcal K(E_1,E_2)
	\iff
	A(\overline B_1(0))
	\text{ предкомпактно}.
	\]

	Эквивалентно:
	для любой ограниченной последовательности
	\[
	\{x_n\}\subset E_1
	\]
	последовательность
	\[
	\{Ax_n\}
	\]
	имеет сходящуюся подпоследовательность.
\end{remark}

\begin{theorem}[Конечномерный случай]
	Пусть
	\[
	A\in\mathcal L(E_1,E_2).
	\]
	Если
	\[
	\dim E_1<\infty
	\qquad\text{или}\qquad
	\dim E_2<\infty,
	\]
	то
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать, что образ ограниченного множества предкомпактен.}

	Если
	\[
	\dim E_1<\infty,
	\]
	то
	\[
	\dim\operatorname{Im}A\leq\dim E_1<\infty.
	\]
	Образ ограниченного множества ограничен в конечномерном пространстве
	\(\operatorname{Im}A\), следовательно, предкомпактен.

	Если
	\[
	\dim E_2<\infty,
	\]
	то \(A(M)\) ограничено в конечномерном \(E_2\), следовательно,
	также предкомпактно.
\end{proof}

\begin{theorem}
	Если \(E\) бесконечномерно, то тождественный оператор не компактен:
	\[
	I\notin\mathcal K(E).
	\]
\end{theorem}

\begin{proof}
	\textbf{Предположим противное.}

	Если \(I\) компактен, то
	\[
	I(\overline B_1(0))
	=
	\overline B_1(0)
	\]
	предкомпактен.

	Но в бесконечномерном нормированном пространстве единичный шар
	не предкомпактен.

	Противоречие.
\end{proof}

\begin{theorem}[Идеальное свойство]
	Пусть
	\[
	A\in\mathcal K(E),
	\qquad
	B\in\mathcal L(E).
	\]
	Тогда
	\[
	AB,BA\in\mathcal K(E).
	\]
\end{theorem}

\begin{proof}
	Пусть \(M\subset E\) ограничено.

	\begin{enumerate}
		\item
		\textbf{Хотим доказать компактность \(AB\).}

		Так как \(B\) ограничен,
		\[
		B(M)
		\]
		ограничено.

		Так как \(A\) компактен,
		\[
		A(B(M))
		\]
		предкомпактно.

		Следовательно,
		\[
		AB\in\mathcal K(E).
		\]

		\item
		\textbf{Хотим доказать компактность \(BA\).}

		Так как \(A\) компактен,
		\[
		\overline{A(M)}
		\]
		компактно.

		Оператор \(B\) непрерывен, поэтому
		\[
		B(\overline{A(M)})
		\]
		компактно.

		Так как
		\[
		B(A(M))
		\subset
		B(\overline{A(M)}),
		\]
		множество \(B(A(M))\) предкомпактно.

		Следовательно,
		\[
		BA\in\mathcal K(E).
		\]
	\end{enumerate}
\end{proof}

\begin{corollary}
	Если \(E\) бесконечномерно и
	\[
	A\in\mathcal K(E),
	\]
	то \(A^{-1}\), если существует, не может быть ограниченным.
\end{corollary}

\begin{proof}
	Если бы
	\[
	A^{-1}\in\mathcal L(E),
	\]
	то по идеальному свойству
	\[
	I=A^{-1}A
	\]
	был бы компактным.

	Но в бесконечномерном пространстве
	\[
	I\notin\mathcal K(E).
	\]
\end{proof}

\begin{theorem}[Предел компактных операторов]
	Пусть \(E_2\) --- банахово пространство,
	\[
	A_n\in\mathcal K(E_1,E_2)
	\]
	и
	\[
	\|A_n-A\|\to0.
	\]
	Тогда
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать, что \(A(\overline B_1(0))\) предкомпактно.}

	Так как \(E_2\) банахово, достаточно построить для него конечную
	\(\varepsilon\)-сеть.

	Пусть
	\[
	\varepsilon>0.
	\]

	\textbf{1. Хотим заменить \(A\) близким компактным оператором.}

	Выбираем \(N\), такое что
	\[
	\|A_N-A\|<\frac{\varepsilon}{2}.
	\]

	Тогда для
	\[
	\|x\|\leq1
	\]
	имеем
	\[
	\|Ax-A_Nx\|
	\leq
	\|A-A_N\|\|x\|
	<
	\frac{\varepsilon}{2}.
	\]

	\textbf{2. Используем компактность \(A_N\).}

	Так как \(A_N\) компактен, для множества
	\[
	A_N(\overline B_1(0))
	\]
	существует конечная \(\varepsilon/2\)-сеть
	\[
	y_1,\ldots,y_m.
	\]

	Поэтому для любого \(\|x\|\leq1\) найдется \(y_j\), такое что
	\[
	\|A_Nx-y_j\|<\frac{\varepsilon}{2}.
	\]

	Тогда
	\[
	\|Ax-y_j\|
	\leq
	\|Ax-A_Nx\|+\|A_Nx-y_j\|
	<
	\varepsilon.
	\]

	Значит,
	\[
	y_1,\ldots,y_m
	\]
	--- конечная \(\varepsilon\)-сеть для \(A(\overline B_1(0))\).

	Следовательно,
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{proof}

\begin{theorem}[Компактный оператор улучшает слабую сходимость]
	Пусть
	\[
	A\in\mathcal K(E_1,E_2)
	\]
	и
	\[
	x_n\rightharpoonup x.
	\]
	Тогда
	\[
	\|Ax_n-Ax\|\to0.
	\]
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим использовать компактность \(A\).}

		Слабо сходящаяся последовательность ограничена, поэтому
		\[
		\{x_n\}\cup\{x\}
		\]
		ограничена.

		Так как \(A\) компактен,
		\[
		\{Ax_n\}\cup\{Ax\}
		\]
		предкомпактна.

		\item
		\textbf{Хотим определить возможный предел образов.}

		Линейный ограниченный оператор сохраняет слабую сходимость:
		\[
		x_n\rightharpoonup x
		\quad\Longrightarrow\quad
		Ax_n\rightharpoonup Ax.
		\]

		\item
		\textbf{Хотим получить сходимость по норме.}

		Предположим, что
		\[
		Ax_n\not\to Ax
		\]
		по норме.

		Тогда существуют \(\varepsilon>0\) и подпоследовательность
		\[
		Ax_{n_k},
		\]
		для которой
		\[
		\|Ax_{n_k}-Ax\|\geq\varepsilon.
		\]

		Из предкомпактности выделяем еще одну подпоследовательность
		\[
		Ax_{n_{k_j}}\to y
		\]
		по норме.

		Тогда
		\[
		Ax_{n_{k_j}}\rightharpoonup y.
		\]

		Но одновременно
		\[
		Ax_{n_{k_j}}\rightharpoonup Ax.
		\]

		Слабый предел единственен, поэтому
		\[
		y=Ax.
		\]

		Получили
		\[
		Ax_{n_{k_j}}\to Ax
		\]
		по норме, что противоречит
		\[
		\|Ax_{n_k}-Ax\|\geq\varepsilon.
		\]

		Следовательно,
		\[
		\boxed{\|Ax_n-Ax\|\to0.}
		\]
	\end{enumerate}
\end{proof}

\subsection{Свойства собственных значений компактного оператора}

\begin{definition}[Собственное значение]
	Пусть \(E\) --- комплексное банахово пространство и
	\[
	A\in\mathcal{L}(E).
	\]
	Число \(\lambda\in\mathbb{C}\) называется \textbf{собственным значением} оператора \(A\), если
	существует ненулевой вектор \(x\in E\), такой что
	\[
	Ax=\lambda x.
	\]
	Вектор \(x\) называется \textbf{собственным вектором}.
\end{definition}

\begin{remark}[Почему ноль особый]
	Для компактного оператора ненулевые собственные значения ведут себя почти как в конечномерной
	линейной алгебре: вне любой окрестности нуля их может быть только конечное число.

	Но \(0\) может быть собственным значением бесконечной кратности.
\end{remark}

\begin{example}
	Пусть оператор \(A:l_2\to l_2\) задаётся диагональной матрицей
	\[
	A=
	\begin{pmatrix}
		1 & 0 & 0 & \cdots\\
		0 & \frac12 & 0 & \cdots\\
		0 & 0 & \frac13 & \cdots\\
		\vdots & \vdots & \vdots & \ddots
	\end{pmatrix}.
	\]
	То есть
	\[
	A(x_1,x_2,x_3,\ldots)
	=
	\left(x_1,\frac{x_2}{2},\frac{x_3}{3},\ldots\right).
	\]
	Этот оператор компактен.

	Его ненулевые собственные значения:
	\[
	1,\frac12,\frac13,\ldots
	\]
	и они стремятся к нулю. При этом \(0\) является собственным значением бесконечной кратности
	для подходящих компактных операторов, поэтому \(0\) нельзя рассматривать так же, как
	ненулевые собственные значения.
\end{example}

\begin{theorem}[Собственные значения компактного самосопряженного оператора вне нуля]
	Пусть \(H\) --- гильбертово пространство и
	\[
	A\in\mathcal K(H)
	\]
	--- самосопряженный оператор. Тогда для любого
	\[
	\delta>0
	\]
	существует лишь конечное число собственных значений \(\lambda\), для которых
	\[
	|\lambda|\geq\delta.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим получить противоречие с компактностью \(A\).}

	Предположим, что существуют различные собственные значения
	\[
	\lambda_1,\lambda_2,\ldots,
	\qquad
	|\lambda_n|\geq\delta.
	\]
	Выберем соответствующие нормированные собственные векторы
	\[
	Ae_n=\lambda_n e_n,
	\qquad
	\|e_n\|=1.
	\]

	\textbf{Хотим показать, что образы \(Ae_n\) не могут иметь сходящейся подпоследовательности.}

	Так как \(A\) самосопряжен, собственные векторы различных собственных значений
	ортогональны:
	\[
	e_n\perp e_m,
	\qquad n\neq m.
	\]

	Поэтому
	\[
	\begin{aligned}
		\|Ae_n-Ae_m\|^2
		&=
		\|\lambda_n e_n-\lambda_m e_m\|^2
		\\
		&=
		|\lambda_n|^2+|\lambda_m|^2
		\\
		&\geq
		2\delta^2.
	\end{aligned}
	\]
	Следовательно,
	\[
	\|Ae_n-Ae_m\|
	\geq
	\sqrt2\,\delta,
	\qquad n\neq m.
	\]

	Значит, последовательность
	\[
	\{Ae_n\}
	\]
	не содержит фундаментальной, а значит и сходящейся подпоследовательности.

	Но
	\[
	\|e_n\|=1,
	\]
	то есть \(\{e_n\}\) ограничена. Так как \(A\) компактен,
	последовательность \(\{Ae_n\}\) должна иметь сходящуюся подпоследовательность.

	Получили противоречие.

	Следовательно, для любого \(\delta>0\) существует лишь конечное число
	собственных значений, удовлетворяющих
	\[
	|\lambda|\geq\delta.
	\]
\end{proof}

\begin{remark}[Общий случай]
	В общем банаховом случае это утверждение доказывается похожей идеей, но вместо
	ортогональности используют теорему Рисса о почти перпендикуляре.

	На экзамене обычно достаточно понимать смысл: если бы ненулевых собственных значений
	вдали от нуля было бесконечно много, компактность дала бы сходящуюся подпоследовательность
	образов, но собственные векторы можно выбрать так, что образы остаются отделёнными друг от друга.
\end{remark}

\subsection{Теорема Фредгольма для компактных самосопряженных операторов}

\begin{definition}[Оператор \(A_\lambda\)]
	Пусть
	\[
	A\in\mathcal L(H),
	\qquad
	\lambda\in\mathbb C.
	\]
	Обозначим
	\[
	A_\lambda=A-\lambda I.
	\]
\end{definition}

\begin{lemma}
	Пусть
	\[
	A\in\mathcal K(H)
	\]
	--- самосопряженный оператор и
	\[
	\lambda\neq0.
	\]
	Тогда на подпространстве
	\[
	M=(\ker A_\lambda)^\perp
	\]
	оператор \(A_\lambda\) ограничен снизу:
	\[
	\exists c>0
	\qquad
	\|A_\lambda x\|\geq c\|x\|,
	\qquad x\in M.
	\]
\end{lemma}

\begin{proof}
	\textbf{Хотим получить противоречие с компактностью \(A\).}

	Предположим, что \(A_\lambda\) не ограничен снизу на \(M\).
	Тогда существуют
	\[
	x_n\in M,
	\qquad
	\|x_n\|=1,
	\]
	такие что
	\[
	A_\lambda x_n\to0.
	\]

	Так как \(A\) компактен, из ограниченной последовательности \(\{x_n\}\)
	можно выбрать подпоследовательность, для которой
	\[
	Ax_{n_k}\to y.
	\]

	Но
	\[
	\lambda x_{n_k}
	=
	Ax_{n_k}-A_\lambda x_{n_k}
	\to y.
	\]
	Так как \(\lambda\neq0\),
	\[
	x_{n_k}\to x:=\frac{y}{\lambda}.
	\]

	Подпространство \(M\) замкнуто, поэтому
	\[
	x\in M,
	\qquad
	\|x\|=1.
	\]

	Переходя к пределу в
	\[
	A_\lambda x_{n_k}\to0,
	\]
	получаем
	\[
	A_\lambda x=0.
	\]
	Следовательно,
	\[
	x\in\ker A_\lambda.
	\]

	Таким образом,
	\[
	x\in M\cap\ker A_\lambda=\{0\},
	\]
	что противоречит
	\[
	\|x\|=1.
	\]
\end{proof}

\begin{lemma}
	При тех же условиях
	\[
	\operatorname{Im}A_\lambda
	\]
	замкнут.
\end{lemma}

\begin{proof}
	По предыдущей лемме оператор
	\[
	A_\lambda|_{(\ker A_\lambda)^\perp}
	\]
	ограничен снизу. Так как \((\ker A_\lambda)^\perp\) --- банахово пространство,
	его образ замкнут. Следовательно,
	\[
	\operatorname{Im}A_\lambda
	\]
	замкнут.
\end{proof}

\begin{theorem}[Фредгольма]
	Пусть
	\[
	A\in\mathcal K(H)
	\]
	--- компактный самосопряженный оператор и
	\[
	\lambda\neq0.
	\]
	Тогда
	\[
	H
	=
	\operatorname{Im}A_\lambda
	\oplus_{\perp}
	\ker A_\lambda.
	\]
\end{theorem}

\begin{proof}
	Так как \(A_\lambda\) самосопряжен, имеем стандартное разложение
	\[
	H
	=
	\overline{\operatorname{Im}A_\lambda}
	\oplus_{\perp}
	\ker A_\lambda.
	\]

	По предыдущей лемме
	\[
	\operatorname{Im}A_\lambda
	\]
	замкнут, поэтому
	\[
	\overline{\operatorname{Im}A_\lambda}
	=
	\operatorname{Im}A_\lambda.
	\]

	Следовательно,
	\[
	\boxed{
		H
		=
		\operatorname{Im}A_\lambda
		\oplus_{\perp}
		\ker A_\lambda.
	}
	\]
\end{proof}

\subsection{Теорема Гильберта--Шмидта}

\begin{lemma}[Наличие ненулевого собственного значения]
	Пусть
	\[
	A\in\mathcal K(H)
	\]
	--- компактный самосопряженный оператор. Если
	\[
	A\neq0,
	\]
	то у \(A\) существует ненулевое собственное значение.
\end{lemma}

\begin{proof}
	\textbf{Хотим найти ненулевую точку спектра.}

	Так как \(A\) самосопряжен,
	\[
	r(A)=\|A\|.
	\]
	Поскольку \(A\neq0\),
	\[
	r(A)=\|A\|>0.
	\]
	Следовательно, существует
	\[
	\lambda\in\sigma(A),
	\qquad
	|\lambda|=r(A)>0.
	\]

	Для компактного оператора всякая ненулевая точка спектра является
	собственным значением. Значит,
	\[
	\lambda\neq0
	\]
	--- собственное значение \(A\).
\end{proof}

\begin{theorem}[Гильберта--Шмидта]
	Пусть \(H\) --- сепарабельное комплексное гильбертово пространство и
	\[
	A\in\mathcal K(H)
	\]
	--- самосопряженный оператор. Тогда в \(H\) существует ортонормированный
	базис из собственных векторов \(A\).
\end{theorem}

\begin{proof}
	\textbf{Хотим собрать все собственные векторы и показать, что они порождают \(H\).}

	Ненулевых собственных значений компактного оператора не более чем счетно.
	Обозначим их через
	\[
	\lambda_1,\lambda_2,\ldots
	\]
	с учетом возможной конечности множества.

	Для каждого \(\lambda_j\neq0\) собственное подпространство
	\[
	H_{\lambda_j}=\ker(A-\lambda_jI)
	\]
	конечномерно. Выберем в нем ОНБ
	\[
	\{e_k^{(j)}\}_{k=1}^{m_j}.
	\]

	Так как \(A\) самосопряжен, собственные подпространства,
	соответствующие различным собственным значениям, ортогональны.
	Поэтому система всех \(e_k^{(j)}\) ортонормирована.

	Обозначим
	\[
	M=
	\overline{\operatorname{span}}
	\{e_k^{(j)}:\lambda_j\neq0\}.
	\]

	\textbf{Хотим понять, что остается в \(M^\perp\).}

	Подпространство \(M\) инвариантно относительно \(A\), поэтому
	из самосопряженности \(A\)
	\[
	A(M^\perp)\subset M^\perp.
	\]

	Рассмотрим сужение
	\[
	\widetilde A=A|_{M^\perp}.
	\]
	Оно компактно и самосопряженно.

	Если бы
	\[
	\widetilde A\neq0,
	\]
	то по предыдущей лемме у \(\widetilde A\) существовал бы ненулевой
	собственный вектор \(u\in M^\perp\):
	\[
	Au=\mu u,
	\qquad
	\mu\neq0.
	\]
	Но тогда \(u\) принадлежит собственному подпространству \(H_\mu\),
	а значит, по построению,
	\[
	u\in M.
	\]
	Получаем
	\[
	u\in M\cap M^\perp=\{0\},
	\]
	противоречие.

	Следовательно,
	\[
	\widetilde A=0,
	\]
	то есть
	\[
	M^\perp\subset\ker A.
	\]

	С другой стороны, собственные векторы для собственного значения \(0\)
	ортогональны собственным векторам для всех \(\lambda_j\neq0\), поэтому
	\[
	\ker A\subset M^\perp.
	\]

	Таким образом,
	\[
	\boxed{M^\perp=\ker A}.
	\]

	Следовательно,
	\[
	H=M\oplus_{\perp}\ker A.
	\]

	Выберем ОНБ в \(\ker A\) и объединим его с системой
	\[
	\{e_k^{(j)}\}.
	\]
	Полученная система является ортонормированным базисом \(H\),
	состоящим из собственных векторов \(A\).
\end{proof}

\begin{idea}
	\textbf{Что вообще утверждает теорема Гильберта--Шмидта и как устроено доказательство.}

	Пусть
	\[
	A\in\mathcal K(H)
	\]
	--- компактный самосопряженный оператор.

	Хотим доказать, что в \(H\) существует ОНБ из собственных векторов \(A\).

	То есть хотим разложить всё пространство на направления, на каждом из которых
	оператор действует просто умножением на число:
	\[
	Ae_k=\lambda_k e_k.
	\]

	Иначе говоря, хотим показать, что собственных векторов оператора достаточно,
	чтобы породить всё \(H\).

	\medskip

	\textbf{Какие известные факты нам понадобятся.}

	Из предыдущих билетов используем:

	\[
	\boxed{
		A\neq0
		\quad\Longrightarrow\quad
		A\text{ имеет ненулевое собственное значение}
	}
	\]
	для компактного самосопряженного \(A\);

	\[
	\boxed{
		\lambda\neq\mu
		\quad\Longrightarrow\quad
		H_\lambda\perp H_\mu
	}
	\]
	для самосопряженного оператора;

	\[
	\boxed{
		\lambda\neq0
		\quad\Longrightarrow\quad
		\dim H_\lambda<\infty
	}
	\]
	для компактного оператора;

	и ненулевых собственных значений компактного оператора
	не более чем счетно.

	Последний факт получается из того, что для каждого
	\[
	\delta>0
	\]
	существует лишь конечное число собственных значений, для которых
	\[
	|\lambda|\geq\delta.
	\]
	Поэтому все ненулевые собственные значения можно собрать как
	\[
	\bigcup_{n=1}^{\infty}
	\left\{
	\lambda:
	|\lambda|\geq\frac1n
	\right\},
	\]
	то есть в счетное объединение конечных множеств.

	\medskip

	\textbf{1. Почему сначала нужна лемма о существовании ненулевого собственного значения?}

	В основном доказательстве мы в какой-то момент получим некоторое
	подпространство, на котором оператор всё ещё действует ненулевым образом.

	Нам нужно будет сказать:

	\[
	A|_N\neq0
	\quad\Longrightarrow\quad
	\text{в }N\text{ есть ненулевой собственный вектор}.
	\]

	Именно для этого нужна предварительная лемма.

	Почему сама лемма верна?

	Если
	\[
	A\neq0,
	\]
	то
	\[
	\|A\|>0.
	\]

	Для самосопряженного оператора
	\[
	r(A)=\|A\|.
	\]

	Следовательно,
	\[
	r(A)>0,
	\]
	то есть в спектре есть точка
	\[
	\lambda\in\sigma(A),
	\qquad
	|\lambda|=r(A)>0.
	\]

	Но для компактного оператора любая ненулевая точка спектра является
	собственным значением.

	Поэтому
	\[
	\boxed{
		A\neq0
		\Longrightarrow
		\exists\,\lambda\neq0,\quad
		Ax=\lambda x
	}
	\]
	для некоторого \(x\neq0\).

	Это главный инструмент, которым потом мы будем искать
	``пропущенные'' собственные векторы.

	\medskip

	\textbf{2. Собираем все уже существующие ненулевые собственные направления.}

	Пусть
	\[
	\lambda_1,\lambda_2,\ldots
	\]
	--- все ненулевые собственные значения \(A\).

	Для каждого \(\lambda_j\) имеется собственное подпространство
	\[
	H_{\lambda_j}
	=
	\ker(A-\lambda_jI).
	\]

	Так как
	\[
	\lambda_j\neq0
	\]
	и \(A\) компактен, каждое \(H_{\lambda_j}\) конечномерно.

	Поэтому в каждом из них можем выбрать конечный ОНБ:
	\[
	e_1^{(j)},\ldots,e_{m_j}^{(j)}.
	\]

	Причем если
	\[
	i\neq j,
	\]
	то
	\[
	H_{\lambda_i}\perp H_{\lambda_j},
	\]
	поскольку \(A\) самосопряжен.

	Значит, если объединить все выбранные базисы, получится
	ортонормированная система.

	\medskip

	Теперь берем замкнутую линейную оболочку всех собственных векторов,
	отвечающих ненулевым собственным значениям:
	\[
	M=
	\overline{\operatorname{span}}
	\{e_k^{(j)}\}.
	\]

	Смысл \(M\):

	\[
	\boxed{
		M=
		\text{вся часть пространства, которую мы уже объяснили
			ненулевыми собственными векторами}.
	}
	\]

	Теперь главный вопрос всего доказательства:

	\[
	\boxed{
		\text{что находится в }M^\perp?
	}
	\]

	Если докажем, что
	\[
	M^\perp=\ker A,
	\]
	то всё закончено.

	Действительно, тогда
	\[
	H=M\oplus_\perp\ker A.
	\]

	В \(M\) уже есть ОНБ из собственных векторов с
	\[
	\lambda\neq0,
	\]
	а в \(\ker A\) любой ненулевой вектор является собственным:
	\[
	Ax=0.
	\]

	Останется выбрать ОНБ в \(\ker A\) и объединить два базиса.

	\medskip

	\textbf{3. Почему \(M^\perp\) вообще можно рассматривать отдельно?}

	Чтобы изучать оператор на \(M^\perp\), нужно знать, что \(A\)
	не выбрасывает векторы из этого пространства.

	Сначала \(M\) инвариантно относительно \(A\):
	\[
	A(M)\subset M,
	\]
	поскольку каждый образующий собственный вектор удовлетворяет
	\[
	Ae_k^{(j)}
	=
	\lambda_j e_k^{(j)}\in M,
	\]
	а затем используем линейность, непрерывность \(A\) и замкнутость \(M\).

	Теперь используем самосопряженность.

	Если
	\[
	y\in M^\perp,
	\qquad
	x\in M,
	\]
	то
	\[
	(Ay,x)
	=
	(y,Ax).
	\]

	Но
	\[
	Ax\in M,
	\]
	а \(y\perp M\), поэтому
	\[
	(y,Ax)=0.
	\]

	Следовательно,
	\[
	(Ay,x)=0
	\qquad
	\forall x\in M.
	\]

	Значит,
	\[
	Ay\in M^\perp.
	\]

	Итак,
	\[
	\boxed{
		A(M^\perp)\subset M^\perp.
	}
	\]

	Поэтому корректно рассматривать сужение
	\[
	\widetilde A
	=
	A|_{M^\perp}
	:
	M^\perp\to M^\perp.
	\]

	\medskip

	\textbf{4. Что мы хотим доказать про это сужение?}

	Хотим показать
	\[
	\boxed{\widetilde A=0.}
	\]

	Почему это сразу решит задачу?

	Если
	\[
	\widetilde A=0,
	\]
	то для любого
	\[
	x\in M^\perp
	\]
	имеем
	\[
	Ax=0.
	\]

	Следовательно,
	\[
	M^\perp\subset\ker A.
	\]

	То есть всё, что осталось после удаления ненулевых собственных направлений,
	может относиться только к собственному значению \(0\).

	\medskip

	\textbf{5. Почему \(\widetilde A\) обязан быть нулевым?}

	Предположим противное:
	\[
	\widetilde A\neq0.
	\]

	Так как \(M^\perp\) --- замкнутое подпространство гильбертова пространства,
	оно само является гильбертовым пространством.

	Кроме того,
	\[
	\widetilde A
	\]
	остается компактным и самосопряженным.

	По предварительной лемме ненулевой компактный самосопряженный оператор
	имеет ненулевое собственное значение.

	Поэтому существуют
	\[
	u\in M^\perp,
	\qquad
	u\neq0,
	\qquad
	\mu\neq0,
	\]
	такие что
	\[
	\widetilde Au=\mu u.
	\]

	Но \(\widetilde A\) --- просто сужение \(A\), поэтому
	\[
	Au=\mu u.
	\]

	Значит, \(u\) является собственным вектором исходного оператора \(A\)
	для ненулевого собственного значения \(\mu\).

	Но \(M\) специально был построен как замкнутая оболочка
	\textbf{всех} собственных подпространств для ненулевых собственных значений.

	Поэтому
	\[
	u\in M.
	\]

	С другой стороны, мы нашли его внутри
	\[
	M^\perp.
	\]

	Получаем
	\[
	u\in M\cap M^\perp=\{0\},
	\]
	что противоречит
	\[
	u\neq0.
	\]

	Следовательно,
	\[
	\boxed{\widetilde A=0.}
	\]

	И значит,
	\[
	\boxed{M^\perp\subset\ker A.}
	\]

	\medskip

	\textbf{6. Почему на самом деле \(M^\perp=\ker A\)?}

	Одно включение уже получили:
	\[
	M^\perp\subset\ker A.
	\]

	Теперь хотим получить обратное.

	Пусть
	\[
	x\in\ker A.
	\]
	Тогда
	\[
	Ax=0,
	\]
	то есть \(x\) --- собственный вектор для собственного значения \(0\).

	Любой вектор
	\[
	e\in H_{\lambda_j},
	\qquad
	\lambda_j\neq0,
	\]
	является собственным вектором для другого собственного значения.

	Так как \(A\) самосопряжен, собственные подпространства для различных
	собственных значений ортогональны:
	\[
	x\perp e.
	\]

	Это верно для всех векторов, порождающих \(M\). Поэтому
	\[
	x\perp M,
	\]
	то есть
	\[
	x\in M^\perp.
	\]

	Следовательно,
	\[
	\ker A\subset M^\perp.
	\]

	Итак,
	\[
	\boxed{
		M^\perp=\ker A.
	}
	\]

	\medskip

	\textbf{7. Теперь получаем базис всего пространства.}

	Поскольку \(M\) замкнуто,
	\[
	H=M\oplus_\perp M^\perp.
	\]

	Но мы доказали
	\[
	M^\perp=\ker A.
	\]

	Значит,
	\[
	\boxed{
		H=M\oplus_\perp\ker A.
	}
	\]

	В \(M\) уже имеется ОНБ
	\[
	\{e_k^{(j)}\},
	\]
	состоящий из собственных векторов с ненулевыми собственными значениями.

	В \(\ker A\) выбираем ОНБ
	\[
	\{f_s\}.
	\]

	Для каждого такого вектора
	\[
	Af_s=0,
	\]
	поэтому это собственные векторы с собственным значением \(0\).

	Так как
	\[
	M\perp\ker A,
	\]
	объединение
	\[
	\{e_k^{(j)}\}\cup\{f_s\}
	\]
	ортонормировано.

	А поскольку
	\[
	H=M\oplus_\perp\ker A,
	\]
	оно полно.

	Следовательно, это ОНБ всего \(H\), состоящий из собственных векторов \(A\).

	\medskip

	\textbf{8. Где здесь используется сепарабельность?}

	Сепарабельность нужна, чтобы ортонормированный базис можно было
	представлять как конечную или счетную систему.

	В частности, ОНБ пространства
	\[
	\ker A
	\]
	можно записать в виде
	\[
	f_1,f_2,\ldots
	\]
	с учетом возможной конечности системы.

	Сама главная идея доказательства от этого не меняется.

	\medskip

	\textbf{9. Что происходит, если \(A=0\)?}

	Никакой отдельной сложной ситуации нет.

	В этом случае ненулевых собственных значений вообще нет, поэтому
	\[
	M=\{0\}.
	\]

	Следовательно,
	\[
	M^\perp=H=\ker A.
	\]

	Любой ОНБ пространства \(H\) состоит из собственных векторов:
	\[
	Ae=0.
	\]

	То есть общий ход доказательства автоматически включает случай \(A=0\).

	\medskip

	\textbf{Вся идея теоремы в одной цепочке:}

	Сначала собираем все ненулевые собственные направления:
	\[
	M=
	\overline{\operatorname{span}}
	\{H_\lambda:\lambda\neq0\}.
	\]

	Потом смотрим, что осталось:
	\[
	M^\perp.
	\]

	Если бы на \(M^\perp\) оператор был ненулевым, то
	\[
	A|_{M^\perp}\neq0
	\]
	по лемме породил бы новый ненулевой собственный вектор.

	Но этот вектор должен был бы уже лежать в \(M\), поскольку
	\(M\) содержит все ненулевые собственные направления.

	Получили бы
	\[
	0\neq u\in M\cap M^\perp,
	\]
	что невозможно.

	Поэтому
	\[
	A|_{M^\perp}=0,
	\]
	то есть
	\[
	M^\perp=\ker A.
	\]

	Следовательно,
	\[
	\boxed{
		H=
		\underbrace{M}_{\lambda\neq0}
		\oplus_\perp
		\underbrace{\ker A}_{\lambda=0}.
	}
	\]

	То есть всё пространство разбилось на ортогональную сумму собственных
	подпространств.

	Именно это и дает ортонормированный базис из собственных векторов.
\end{idea}

\subsection{Производная Фреше, производная Гато. Формула конечных приращений.}

\begin{definition}[Производная Фреше]
	Пусть \(E_1,E_2\) --- вещественные банаховы пространства,
	\(D\subset E_1\) --- область,
	\[
	F:D\to E_2.
	\]
	Функция \(F\) называется \textbf{дифференцируемой по Фреше} в точке
	\(x_0\in D\), если существует оператор
	\[
	A\in\mathcal L(E_1,E_2),
	\]
	такой, что
	\[
	F(x_0+h)-F(x_0)=Ah+r(h),
	\qquad
	\frac{\|r(h)\|}{\|h\|}\xrightarrow[h\to0]{}0.
	\]
	Оператор \(A\) называется \textbf{производной Фреше} функции \(F\) в точке \(x_0\):
	\[
	A=F'(x_0).
	\]
\end{definition}

То есть
\[
F(x_0+h)-F(x_0)
=
F'(x_0)h+o(\|h\|).
\]

\begin{definition}[Дифференциал]
	Дифференциал функции \(F\) в точке \(x_0\) на приращении \(h\) есть
	\[
	dF(x_0,h)=F'(x_0)h.
	\]
\end{definition}

\begin{definition}[Производная Гато]
	Пусть \(h\in E_1\).
	Производной Гато функции \(F\) в точке \(x_0\) по направлению \(h\)
	называется предел
	\[
	D_GF(x_0;h)
	=
	\left.\frac{d}{dt}F(x_0+th)\right|_{t=0}
	=
	\lim_{t\to0}
	\frac{F(x_0+th)-F(x_0)}{t},
	\]
	если этот предел существует.

	Если существует оператор
	\[
	A\in\mathcal L(E_1,E_2)
	\]
	такой, что для любого \(h\in E_1\)
	\[
	D_GF(x_0;h)=Ah,
	\]
	то \(A\) называется производной Гато:
	\[
	A=F'_G(x_0).
	\]
\end{definition}

\begin{theorem}[Связь производных Фреше и Гато]
	Если \(F\) дифференцируема по Фреше в точке \(x_0\), то она
	дифференцируема по Гато в этой точке и
	\[
	F'_G(x_0)=F'(x_0).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать}, что производная по любому направлению \(h\) существует
	и равна \(F'(x_0)h\).

	Из дифференцируемости по Фреше
	\[
	F(x_0+u)-F(x_0)=F'(x_0)u+o(\|u\|).
	\]

	\textbf{Для движения в направлении \(h\) положим} \(u=th\). Тогда
	\[
	F(x_0+th)-F(x_0)
	=
	tF'(x_0)h+o(|t|).
	\]

	Делим на \(t\) и переходим к пределу:
	\[
	\lim_{t\to0}
	\frac{F(x_0+th)-F(x_0)}{t}
	=
	F'(x_0)h.
	\]

	Следовательно,
	\[
	F'_G(x_0)=F'(x_0).
	\]
\end{proof}

\begin{theorem}[Формула конечных приращений]
	Пусть \(E_1,E_2\) --- вещественные банаховы пространства,
	\(D\subset E_1\) --- выпуклая область,
	\[
	F:D\to E_2
	\]
	дифференцируема по Фреше.

	Тогда для любых \(x_0,x_1\in D\)
	\[
	\boxed{
		\|F(x_1)-F(x_0)\|
		\le
		\sup_{x\in[x_0,x_1]}\|F'(x)\|
		\,\|x_1-x_0\|
	}.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим свести задачу к обычной формуле Лагранжа.}
	Для этого возьмём \(f\in E_2^*\) и параметризуем отрезок
	\[
	x(t)=x_0+t(x_1-x_0),\qquad t\in[0,1].
	\]
	Рассмотрим вещественную функцию
	\[
	\varphi(t)=f(F(x(t))).
	\]

	По правилу дифференцирования сложной функции
	\[
	\varphi'(t)
	=
	f\bigl(F'(x(t))(x_1-x_0)\bigr).
	\]

	По формуле Лагранжа для некоторого \(\xi\in(0,1)\)
	\[
	f(F(x_1)-F(x_0))
	=
	f\bigl(F'(x(\xi))(x_1-x_0)\bigr).
	\]
	Поэтому
	\[
	|f(F(x_1)-F(x_0))|
	\le
	\|f\|
	\sup_{x\in[x_0,x_1]}\|F'(x)\|
	\|x_1-x_0\|.
	\]

	Берём супремум по всем \(\|f\|\le1\). По теореме Хана--Банаха
	\[
	\|y\|=\sup_{\|f\|\le1}|f(y)|.
	\]
	Следовательно,
	\[
	\boxed{
		\|F(x_1)-F(x_0)\|
		\le
		\sup_{x\in[x_0,x_1]}\|F'(x)\|
		\|x_1-x_0\|
	}.
	\]
\end{proof}

\begin{theorem}[Производная сложной функции]
	Пусть
	\[
	F:E_1\to E_2,\qquad G:E_2\to E_3,
	\]
	\(F\) дифференцируема по Фреше в точке \(x_0\), а \(G\) ---
	в точке
	\[
	y_0=F(x_0).
	\]
	Тогда \(G\circ F\) дифференцируема по Фреше в точке \(x_0\), причём
	\[
	\boxed{
		(G\circ F)'(x_0)=G'(F(x_0))\circ F'(x_0)
	}.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим} представить приращение \(G\circ F\) как линейную часть
	плюс \(o(\|\Delta x\|)\).

	Из дифференцируемости \(F\)
	\[
	\Delta y
	=
	F'(x_0)\Delta x+o(\|\Delta x\|).
	\]

	Из дифференцируемости \(G\)
	\[
	\Delta z
	=
	G'(y_0)\Delta y+o(\|\Delta y\|).
	\]

	Подставляем первое во второе:
	\[
	\Delta z
	=
	G'(y_0)F'(x_0)\Delta x
	+
	G'(y_0)o(\|\Delta x\|)
	+
	o(\|\Delta y\|).
	\]

	Первый остаток есть \(o(\|\Delta x\|)\), так как \(G'(y_0)\) ограничен.

	Кроме того,
	\[
	\|\Delta y\|
	\le
	\|F'(x_0)\|\|\Delta x\|+o(\|\Delta x\|)
	=
	O(\|\Delta x\|),
	\]
	поэтому
	\[
	o(\|\Delta y\|)=o(\|\Delta x\|).
	\]

	Значит,
	\[
	\Delta z
	=
	G'(y_0)F'(x_0)\Delta x+o(\|\Delta x\|),
	\]
	откуда
	\[
	(G\circ F)'(x_0)
	=
	G'(F(x_0))\circ F'(x_0).
	\]
\end{proof}

\subsection{Определения и основные свойства. Формула умножения. Преобразование Фурье свертки.}

\begin{definition}[Преобразование Фурье в \(L_1(\mathbb R)\)]
	Пусть
	\[
	f\in L_1(\mathbb R).
	\]
	Преобразованием Фурье функции \(f\) называется функция
	\[
	\widehat f(y)
	=
	\int_{\mathbb R} f(x)e^{-ixy}\,dx.
	\]
\end{definition}

\begin{theorem}[Основные свойства преобразования Фурье]
	Для любого \(f\in L_1(\mathbb R)\):
	\begin{enumerate}
		\item
		\[
		\|\widehat f\|_\infty\leq \|f\|_{L_1};
		\]

		\item \(\widehat f\) непрерывна;

		\item
		\[
		\widehat f(y)\to0,
		\qquad |y|\to\infty.
		\]
	\end{enumerate}

	Таким образом,
	\[
	\mathcal F:L_1(\mathbb R)\to C_0(\mathbb R)
	\]
	--- линейный ограниченный оператор.
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим оценить норму \(\widehat f\).}

		Для любого \(y\in\mathbb R\)
		\[
		|\widehat f(y)|
		=
		\left|
		\int_{\mathbb R}f(x)e^{-ixy}\,dx
		\right|
		\leq
		\int_{\mathbb R}|f(x)|\,dx
		=
		\|f\|_{L_1}.
		\]
		Следовательно,
		\[
		\|\widehat f\|_\infty\leq\|f\|_{L_1}.
		\]

		\item
		\textbf{Хотим доказать непрерывность \(\widehat f\).}

		Пусть \(y_n\to y\). Тогда
		\[
		f(x)e^{-ixy_n}\to f(x)e^{-ixy}
		\]
		для почти всех \(x\), причем
		\[
		|f(x)e^{-ixy_n}|=|f(x)|\in L_1.
		\]
		По теореме Лебега
		\[
		\widehat f(y_n)\to\widehat f(y).
		\]

		\item
		\textbf{Хотим показать убывание на бесконечности.}

		Для \(\varphi\in C_c^1(\mathbb R)\) интегрированием по частям получаем
		\[
		\widehat\varphi(y)
		=
		\frac{1}{iy}
		\int_{\mathbb R}\varphi'(x)e^{-ixy}\,dx,
		\]
		поэтому
		\[
		|\widehat\varphi(y)|
		\leq
		\frac{\|\varphi'\|_{L_1}}{|y|}
		\to0.
		\]

		Теперь для \(f\in L_1\) выбираем
		\[
		\varphi\in C_c^1(\mathbb R),
		\qquad
		\|f-\varphi\|_{L_1}<\varepsilon.
		\]
		Тогда
		\[
		|\widehat f(y)|
		\leq
		|\widehat{f-\varphi}(y)|+|\widehat\varphi(y)|
		\leq
		\varepsilon+|\widehat\varphi(y)|.
		\]
		При достаточно больших \(|y|\)
		\[
		|\widehat\varphi(y)|<\varepsilon,
		\]
		следовательно,
		\[
		|\widehat f(y)|<2\varepsilon.
		\]
	\end{enumerate}
\end{proof}

\begin{corollary}
	Если
	\[
	f_n\to f
	\quad\text{в }L_1,
	\]
	то
	\[
	\widehat f_n\to\widehat f
	\quad\text{равномерно на }\mathbb R.
	\]
\end{corollary}

\begin{proof}
	Действительно,
	\[
	\|\widehat f_n-\widehat f\|_\infty
	=
	\|\widehat{f_n-f}\|_\infty
	\leq
	\|f_n-f\|_{L_1}
	\to0.
	\]
\end{proof}

\begin{theorem}[Формула умножения]
	Если
	\[
	f,g\in L_1(\mathbb R),
	\]
	то
	\[
	\int_{\mathbb R}\widehat f(y)g(y)\,dy
	=
	\int_{\mathbb R}f(x)\widehat g(x)\,dx.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим поменять порядок интегрирования.}
	Для этого проверим абсолютную интегрируемость:
	\[
	\int_{\mathbb R}\int_{\mathbb R}
	|f(x)g(y)e^{-ixy}|\,dx\,dy
	=
	\|f\|_{L_1}\|g\|_{L_1}
	<\infty.
	\]
	По теореме Фубини
	\[
	\begin{aligned}
		\int_{\mathbb R}\widehat f(y)g(y)\,dy
		&=
		\int_{\mathbb R}
		\left(
		\int_{\mathbb R}f(x)e^{-ixy}\,dx
		\right)g(y)\,dy
		\\
		&=
		\int_{\mathbb R}f(x)
		\left(
		\int_{\mathbb R}g(y)e^{-ixy}\,dy
		\right)dx
		\\
		&=
		\int_{\mathbb R}f(x)\widehat g(x)\,dx.
	\end{aligned}
	\]
\end{proof}

\begin{definition}[Свертка]
	Пусть
	\[
	f,g\in L_1(\mathbb R).
	\]
	Их сверткой называется функция
	\[
	(f*g)(x)
	=
	\int_{\mathbb R}f(y)g(x-y)\,dy.
	\]
\end{definition}

\begin{theorem}
	Если
	\[
	f,g\in L_1(\mathbb R),
	\]
	то
	\[
	f*g\in L_1(\mathbb R)
	\]
	и
	\[
	\|f*g\|_{L_1}
	\leq
	\|f\|_{L_1}\|g\|_{L_1}.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать интегрируемость свертки.}

	По теореме Тонелли
	\[
	\begin{aligned}
		\|f*g\|_{L_1}
		&\leq
		\int_{\mathbb R}\int_{\mathbb R}
		|f(y)||g(x-y)|\,dy\,dx
		\\
		&=
		\int_{\mathbb R}|f(y)|
		\left(
		\int_{\mathbb R}|g(x-y)|\,dx
		\right)dy
		\\
		&=
		\|f\|_{L_1}\|g\|_{L_1}.
	\end{aligned}
	\]
\end{proof}

\begin{theorem}[Преобразование Фурье свертки]
	Если
	\[
	f,g\in L_1(\mathbb R),
	\]
	то
	\[
	\widehat{f*g}
	=
	\widehat f\,\widehat g.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим раскрыть свертку и поменять порядок интегрирования.}

	Так как
	\[
	\int_{\mathbb R}\int_{\mathbb R}
	|f(y)g(x-y)|\,dy\,dx
	=
	\|f\|_{L_1}\|g\|_{L_1}
	<\infty,
	\]
	по теореме Фубини
	\[
	\begin{aligned}
		\widehat{f*g}(z)
		&=
		\int_{\mathbb R}
		\left(
		\int_{\mathbb R}f(y)g(x-y)\,dy
		\right)e^{-ixz}\,dx
		\\
		&=
		\int_{\mathbb R}f(y)
		\left(
		\int_{\mathbb R}g(x-y)e^{-ixz}\,dx
		\right)dy.
	\end{aligned}
	\]

	\textbf{Хотим выделить преобразование Фурье функции \(g\).}
	Делаем замену
	\[
	t=x-y.
	\]
	Тогда
	\[
	\begin{aligned}
		\widehat{f*g}(z)
		&=
		\int_{\mathbb R}f(y)
		\left(
		\int_{\mathbb R}g(t)e^{-i(t+y)z}\,dt
		\right)dy
		\\
		&=
		\left(
		\int_{\mathbb R}f(y)e^{-iyz}\,dy
		\right)
		\left(
		\int_{\mathbb R}g(t)e^{-itz}\,dt
		\right)
		\\
		&=
		\widehat f(z)\widehat g(z).
	\end{aligned}
	\]
\end{proof}

\subsection{Преобразование Фурье в пространстве \texorpdfstring{\(L_2(\mathbb R)\)}{L2(R)}}

Рассмотрим пространство Шварца \(S(\mathbb R)\).
Используем без доказательства следующие факты:
\[
\overline{S(\mathbb R)}^{\,L_2}=L_2(\mathbb R),
\qquad
\mathcal F:S(\mathbb R)\to S(\mathbb R)
\]
--- биекция, причем
\[
\|\widehat f\|_{L_2}=\|f\|_{L_2},
\qquad f\in S(\mathbb R).
\]

Здесь
\[
\widehat f(y)
=
\frac{1}{\sqrt{2\pi}}
\int_{\mathbb R}f(x)e^{-ixy}\,dx.
\]

Так как \(S(\mathbb R)\) плотно в \(L_2(\mathbb R)\),
а \(\mathcal F\) является изометрией на \(S\),
она единственным образом продолжается до линейной изометрии
\[
\mathcal F:L_2(\mathbb R)\to L_2(\mathbb R).
\]

\begin{theorem}[Сохранение скалярного произведения]
	Для любых
	\[
	f,g\in L_2(\mathbb R)
	\]
	выполнено
	\[
	(f,g)=(\widehat f,\widehat g).
	\]
\end{theorem}

\begin{proof}
	\textbf{Сначала докажем равенство для \(f,g\in S\).}

	По формуле обращения
	\[
	\overline{g(x)}
	=
	\frac{1}{\sqrt{2\pi}}
	\int_{\mathbb R}
	\overline{\widehat g(y)}e^{-ixy}\,dy.
	\]

	По теореме Фубини
	\[
	\begin{aligned}
		(f,g)
		&=
		\int_{\mathbb R}f(x)\overline{g(x)}\,dx
		\\
		&=
		\frac{1}{\sqrt{2\pi}}
		\int_{\mathbb R}\int_{\mathbb R}
		f(x)\overline{\widehat g(y)}e^{-ixy}\,dy\,dx
		\\
		&=
		\int_{\mathbb R}
		\widehat f(y)\overline{\widehat g(y)}\,dy
		\\
		&=
		(\widehat f,\widehat g).
	\end{aligned}
	\]

	\textbf{Теперь переходим к \(f,g\in L_2\).}

	Берем
	\[
	f_n,g_n\in S,
	\qquad
	f_n\to f,
	\qquad
	g_n\to g
	\quad\text{в }L_2.
	\]

	Из непрерывности \(\mathcal F\)
	\[
	\widehat f_n\to\widehat f,
	\qquad
	\widehat g_n\to\widehat g.
	\]

	По непрерывности скалярного произведения
	\[
	(f_n,g_n)\to(f,g),
	\qquad
	(\widehat f_n,\widehat g_n)
	\to
	(\widehat f,\widehat g).
	\]

	Так как
	\[
	(f_n,g_n)
	=
	(\widehat f_n,\widehat g_n),
	\]
	переходом к пределу получаем
	\[
	(f,g)=(\widehat f,\widehat g).
	\]
\end{proof}

\begin{theorem}[Основные свойства преобразования Фурье в \(L_2\)]
	Для
	\[
	\mathcal F:L_2(\mathbb R)\to L_2(\mathbb R)
	\]
	выполнено:
	\begin{enumerate}
		\item
		\[
		\operatorname{Im}\mathcal F=L_2(\mathbb R);
		\]

		\item
		\[
		\ker\mathcal F=\{0\};
		\]

		\item \(\mathcal F\) не является компактным оператором;

		\item
		\[
		\mathcal F^4=I.
		\]
	\end{enumerate}
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим доказать сюръективность.}

		Пусть
		\[
		f\in L_2.
		\]
		Так как \(S\) плотно в \(L_2\), существуют
		\[
		f_n\in S,
		\qquad
		f_n\to f.
		\]

		Так как
		\[
		\mathcal F:S\to S
		\]
		--- биекция, для каждого \(n\) существует
		\[
		g_n\in S:
		\qquad
		\widehat g_n=f_n.
		\]

		По изометричности
		\[
		\|g_n-g_m\|_{L_2}
		=
		\|\widehat g_n-\widehat g_m\|_{L_2}
		=
		\|f_n-f_m\|_{L_2}.
		\]

		Поэтому \(\{g_n\}\) фундаментальна, а значит
		\[
		g_n\to g
		\]
		для некоторого \(g\in L_2\).

		По непрерывности \(\mathcal F\)
		\[
		\widehat g_n\to\widehat g.
		\]
		Но
		\[
		\widehat g_n=f_n\to f,
		\]
		следовательно,
		\[
		\widehat g=f.
		\]

		Значит,
		\[
		\operatorname{Im}\mathcal F=L_2.
		\]

		\item
		\textbf{Хотим доказать инъективность.}

		Если
		\[
		\widehat f=0,
		\]
		то по изометричности
		\[
		\|f\|_{L_2}
		=
		\|\widehat f\|_{L_2}
		=
		0.
		\]
		Следовательно,
		\[
		f=0,
		\]
		то есть
		\[
		\ker\mathcal F=\{0\}.
		\]

		\item
		\textbf{Хотим доказать, что \(\mathcal F\) не компактен.}

		Предположим противное: \(\mathcal F\) компактен.
		Тогда композиция
		\[
		\mathcal F^4
		\]
		также компактна.

		Но
		\[
		\mathcal F^4=I.
		\]

		Получаем, что тождественный оператор
		\[
		I:L_2\to L_2
		\]
		компактен, что невозможно, поскольку \(L_2(\mathbb R)\)
		бесконечномерно.

		Следовательно, \(\mathcal F\) не компактен.

		\item
		\textbf{Хотим вычислить четвертую степень \(\mathcal F\).}

		На пространстве \(S\) из формулы обращения следует
		\[
		\mathcal F^2f(x)=f(-x).
		\]

		Поэтому
		\[
		\mathcal F^4f=f,
		\qquad f\in S.
		\]

		Так как \(S\) плотно в \(L_2\), а \(\mathcal F\) непрерывен,
		равенство продолжается на всё \(L_2\):
		\[
		\mathcal F^4=I.
		\]
	\end{enumerate}
\end{proof}

\begin{idea}
	\textbf{Зачем вообще нужно пространство Шварца и что мы здесь делаем.}

	В пространстве \(L_1(\mathbb R)\) преобразование Фурье можно определить напрямую:
	\[
	\widehat f(y)
	=
	\int_{\mathbb R}f(x)e^{-ixy}\,dx,
	\]
	поскольку
	\[
	\int_{\mathbb R}|f(x)e^{-ixy}|\,dx
	=
	\int_{\mathbb R}|f(x)|\,dx
	<\infty.
	\]

	\textbf{В \(L_2\) так сделать нельзя.}

	Из
	\[
	f\in L_2(\mathbb R)
	\]
	вообще не следует, что
	\[
	f\in L_1(\mathbb R).
	\]
	Поэтому для произвольного \(f\in L_2\) интеграл
	\[
	\int_{\mathbb R}f(x)e^{-ixy}\,dx
	\]
	может не существовать как обычный интеграл.

	То есть наша цель:
	\[
	\boxed{\text{определить преобразование Фурье для всех }f\in L_2}
	\]
	но напрямую по интегральной формуле это сделать нельзя.

	\medskip

	\textbf{1. Поэтому сначала берем ``хорошие'' функции.}

	Рассматриваем пространство Шварца
	\[
	S(\mathbb R)
	=
	\left\{
	f\in C^\infty(\mathbb R):
	\sup_{x\in\mathbb R}|x^m f^{(n)}(x)|<\infty
	\quad
	\forall m,n\geq0
	\right\}.
	\]

	Смысл условия: функции из \(S\) гладкие и вместе со всеми производными
	убывают на бесконечности быстрее любой степени.

	В частности,
	\[
	S(\mathbb R)\subset L_1(\mathbb R)\cap L_2(\mathbb R).
	\]

	Поэтому для \(f\in S\) интегральная формула преобразования Фурье
	точно имеет смысл:
	\[
	\widehat f(y)
	=
	\frac{1}{\sqrt{2\pi}}
	\int_{\mathbb R}f(x)e^{-ixy}\,dx.
	\]

	То есть сначала мы умеем корректно работать с
	\[
	\mathcal F:S\to S.
	\]

	\medskip

	\textbf{2. Почему именно \(S\), а не какое-то случайное подпространство?}

	У пространства Шварца есть два нужных нам свойства,
	которые используем без доказательства:
	\[
	\boxed{\overline{S}^{\,L_2}=L_2}
	\]
	и
	\[
	\boxed{\mathcal F:S\to S\text{ --- биекция}.}
	\]

	Первое означает, что любую функцию
	\[
	f\in L_2
	\]
	можно приблизить функциями из \(S\):
	\[
	f_n\in S,
	\qquad
	f_n\to f
	\quad\text{в }L_2.
	\]

	Поэтому, если мы хорошо определим преобразование Фурье на \(S\)
	и докажем его непрерывность в норме \(L_2\),
	то сможем распространить его на всё \(L_2\).

	\medskip

	\textbf{3. Главное свойство на \(S\) --- Фурье сохраняет норму.}

	Для \(f\in S\) доказывается равенство Планшереля:
	\[
	\|\widehat f\|_{L_2}
	=
	\|f\|_{L_2}.
	\]

	То есть
	\[
	\mathcal F:S\to S
	\]
	является изометрией.

	Это именно то, что позволяет перейти от \(S\) ко всему \(L_2\).

	\medskip

	\textbf{4. Как по \(S\) определить Фурье произвольной функции \(f\in L_2\)?}

	Берем последовательность хороших функций
	\[
	f_n\in S,
	\qquad
	f_n\to f
	\quad\text{в }L_2.
	\]

	У каждой \(f_n\) преобразование Фурье уже определено обычным интегралом:
	\[
	\widehat f_n
	=
	\mathcal Ff_n.
	\]

	Хотим определить
	\[
	\widehat f
	\]
	как предел этих преобразований.

	Из изометричности
	\[
	\|\widehat f_n-\widehat f_m\|_{L_2}
	=
	\|f_n-f_m\|_{L_2}.
	\]

	Так как
	\[
	f_n\to f,
	\]
	последовательность \(\{f_n\}\) фундаментальна.
	Следовательно, фундаментальна и
	\[
	\{\widehat f_n\}.
	\]

	Пространство \(L_2\) полно, поэтому существует
	\[
	g\in L_2:
	\qquad
	\widehat f_n\to g.
	\]

	Именно этот предел и объявляем преобразованием Фурье функции \(f\):
	\[
	\boxed{
		\mathcal Ff
		:=
		\lim_{n\to\infty}\mathcal Ff_n
	}.
	\]

	То есть для общей функции \(f\in L_2\)
	преобразование Фурье определяется уже
	\textbf{не обязательно обычным интегралом},
	а пределом в норме \(L_2\).

	\medskip

	\textbf{5. Почему результат не зависит от того, как именно мы приблизили \(f\)?}

	Пусть есть две последовательности
	\[
	f_n\to f,
	\qquad
	g_n\to f,
	\qquad
	f_n,g_n\in S.
	\]

	Тогда
	\[
	\|f_n-g_n\|_{L_2}\to0.
	\]

	По изометричности
	\[
	\|\widehat f_n-\widehat g_n\|_{L_2}
	=
	\|f_n-g_n\|_{L_2}
	\to0.
	\]

	Значит, обе последовательности преобразований дают один и тот же предел.

	Следовательно, \(\mathcal Ff\) определено корректно.

	\medskip

	\textbf{Итого вся конструкция выглядит так:}
	\[
	\boxed{
		S
		\overset{\text{плотно}}{\subset}
		L_2
	}
	\]
	\[
	\boxed{
		\mathcal F:S\to S
		\text{ уже определено обычным интегралом}
	}
	\]
	\[
	\boxed{
		\|\mathcal Ff\|_2=\|f\|_2
	}
	\]
	\[
	\Downarrow
	\]
	\[
	\boxed{
		\mathcal F
		\text{ единственным образом продолжается }
		S\to S
		\quad\text{до}\quad
		L_2\to L_2.
	}
	\]

	После этого мы уже изучаем свойства полученного оператора на \(L_2\):
	\[
	(f,g)=(\widehat f,\widehat g),
	\qquad
	\ker\mathcal F=\{0\},
	\qquad
	\operatorname{Im}\mathcal F=L_2,
	\qquad
	\mathcal F^4=I,
	\]
	и отсюда, в частности, следует, что \(\mathcal F\) не компактен.

	Таким образом, пространство Шварца здесь нужно не само по себе:
	это просто удобное плотное множество ``хороших'' функций,
	на котором преобразование Фурье сначала можно определить
	и изучить обычными интегралами, а затем по непрерывности
	перенести на всё \(L_2\).
\end{idea}

\subsection{Операторы Гильберта--Шмидта. Компактность}

\begin{definition}[Оператор Гильберта--Шмидта]
	Оператор
	\[
	A:L_2[a,b]\to L_2[a,b]
	\]
	называется оператором Гильберта--Шмидта, если
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt,
	\]
	где
	\[
	K\in L_2([a,b]^2).
	\]
	Функция \(K(x,t)\) называется ядром оператора.
\end{definition}

\begin{remark}
	Для любого \(f\in L_2[a,b]\)
	\[
	\|Af\|_{L_2}
	\leq
	\|K\|_{L_2([a,b]^2)}\|f\|_{L_2}.
	\]
	В частности,
	\[
	\|A\|
	\leq
	\|K\|_{L_2([a,b]^2)}.
	\]
\end{remark}

\begin{proof}
	\textbf{Хотим оценить \(Af\).}

	По неравенству Коши--Буняковского для почти всех \(x\)
	\[
	\begin{aligned}
		|Af(x)|^2
		&=
		\left|
		\int_a^b K(x,t)f(t)\,dt
		\right|^2
		\\
		&\leq
		\left(
		\int_a^b|K(x,t)|^2\,dt
		\right)
		\|f\|_{L_2}^2.
	\end{aligned}
	\]

	Интегрируя по \(x\), получаем
	\[
	\begin{aligned}
		\|Af\|_{L_2}^2
		&\leq
		\int_a^b\int_a^b
		|K(x,t)|^2\,dt\,dx\,
		\|f\|_{L_2}^2
		\\
		&=
		\|K\|_{L_2([a,b]^2)}^2
		\|f\|_{L_2}^2.
	\end{aligned}
	\]
\end{proof}

\begin{theorem}
	Пусть \(\{\varphi_n\}\) --- ОНБ в \(L_2[a,b]\).
	Тогда
	\[
	\{\varphi_i(x)\varphi_j(t)\}_{i,j=1}^{\infty}
	\]
	--- ОНБ в \(L_2([a,b]^2)\).
\end{theorem}

\begin{remark}
	Этот факт используем без доказательства.
\end{remark}

\begin{theorem}[Компактность оператора Гильберта--Шмидта]
	Каждый оператор Гильберта--Шмидта
	\[
	A:L_2[a,b]\to L_2[a,b]
	\]
	компактен.
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим приблизить ядро \(K\) конечными суммами.}

		Так как
		\[
		K\in L_2([a,b]^2)
		\]
		и
		\[
		\{\varphi_i(x)\varphi_j(t)\}
		\]
		--- ОНБ, раскладываем \(K\) в ряд Фурье:
		\[
		K(x,t)
		=
		\sum_{i,j=1}^{\infty}
		a_{ij}\varphi_i(x)\varphi_j(t)
		\]
		со сходимостью в \(L_2([a,b]^2)\).

		Берем частичные суммы
		\[
		K_N(x,t)
		=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)\varphi_j(t).
		\]

		Тогда
		\[
		\|K_N-K\|_{L_2([a,b]^2)}
		\to0.
		\]

		\item
		\textbf{Хотим построить конечномерные операторы, приближающие \(A\).}

		Ядру \(K_N\) сопоставим оператор
		\[
		(A_Nf)(x)
		=
		\int_a^b K_N(x,t)f(t)\,dt.
		\]

		Раскрывая \(K_N\), получаем
		\[
		(A_Nf)(x)
		=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
		\]

		Следовательно,
		\[
		A_Nf
		\in
		\operatorname{span}
		\{\varphi_1,\ldots,\varphi_N\}.
		\]

		Значит,
		\[
		\dim\operatorname{Im}A_N\leq N,
		\]
		то есть \(A_N\) --- конечномерный, а значит компактный оператор.

		\item
		\textbf{Хотим показать, что \(A_N\to A\) по операторной норме.}

		Имеем
		\[
		((A_N-A)f)(x)
		=
		\int_a^b
		(K_N(x,t)-K(x,t))f(t)\,dt.
		\]

		По неравенству Коши--Буняковского
		\[
		|((A_N-A)f)(x)|^2
		\leq
		\left(
		\int_a^b|K_N(x,t)-K(x,t)|^2\,dt
		\right)
		\|f\|_{L_2}^2.
		\]

		Интегрируем по \(x\):
		\[
		\|(A_N-A)f\|_{L_2}^2
		\leq
		\|K_N-K\|_{L_2([a,b]^2)}^2
		\|f\|_{L_2}^2.
		\]

		Следовательно,
		\[
		\boxed{
			\|A_N-A\|
			\leq
			\|K_N-K\|_{L_2([a,b]^2)}
			\to0.
		}
		\]

		\item
		\textbf{Завершаем доказательство.}

		Каждый \(A_N\) компактен и
		\[
		A_N\to A
		\]
		по операторной норме.

		Предел компактных операторов по операторной норме компактен.

		Следовательно,
		\[
		\boxed{A\text{ компактен}.}
		\]
	\end{enumerate}
\end{proof}

\begin{idea}
	\textbf{Что вообще происходит в этом билете и почему оператор Гильберта--Шмидта компактен.}

	Пусть
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt,
	\qquad
	K\in L_2([a,b]^2).
	\]

	Здесь оператор \(A\) берет функцию \(f(t)\) и по ней строит новую функцию
	\(Af(x)\).

	При каждом фиксированном \(x\) мы берем функцию
	\[
	t\mapsto K(x,t)
	\]
	и вычисляем ее скалярное произведение с \(f(t)\):
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt.
	\]

	То есть ядро \(K(x,t)\) полностью определяет действие оператора.

	\medskip

	\textbf{1. Чего мы хотим добиться?}

	Хотим доказать:
	\[
	\boxed{A\text{ компактен}.}
	\]

	Удобный известный критерий:

	если существуют конечномерные операторы \(A_N\), для которых
	\[
	\|A_N-A\|\to0,
	\]
	то \(A\) компактен.

	Действительно, конечномерные операторы компактны, а предел компактных
	операторов по операторной норме компактен.

	Поэтому вся наша задача сводится к следующей:
	\[
	\boxed{
		\text{построить конечномерные }A_N
		\text{ так, чтобы }
		A_N\to A.
	}
	\]

	\medskip

	\textbf{2. Но как приблизить сам оператор \(A\)?}

	Оператор \(A\) задается ядром \(K\).

	Поэтому естественная идея:
	\[
	\boxed{
		\text{сначала приблизить ядро }K
		\text{ простыми ядрами }K_N.
	}
	\]

	Если
	\[
	K_N\approx K,
	\]
	то надеемся, что соответствующие операторы
	\[
	(A_Nf)(x)
	=
	\int_a^b K_N(x,t)f(t)\,dt
	\]
	тоже будут удовлетворять
	\[
	A_N\approx A.
	\]

	Останется подобрать \(K_N\) так, чтобы операторы \(A_N\)
	были конечномерными.

	\medskip

	\textbf{3. Почему здесь появляется ОНБ?}

	Возьмем ортонормированный базис
	\[
	\{\varphi_n\}
	\]
	в пространстве \(L_2[a,b]\).

	Есть известный факт:
	\[
	\boxed{
		\{\varphi_i(x)\varphi_j(t)\}_{i,j=1}^{\infty}
		\text{ --- ОНБ в }L_2([a,b]^2).
	}
	\]

	Почему это нам полезно?

	Потому что наше ядро
	\[
	K=K(x,t)
	\]
	--- функция двух переменных и
	\[
	K\in L_2([a,b]^2).
	\]

	Значит, как любой элемент гильбертова пространства с ОНБ,
	его можно разложить в ряд:
	\[
	K(x,t)
	=
	\sum_{i,j=1}^{\infty}
	a_{ij}\varphi_i(x)\varphi_j(t).
	\]

	Причем ряд сходится именно в норме \(L_2([a,b]^2)\).

	Берем конечную часть:
	\[
	K_N(x,t)
	=
	\sum_{i,j=1}^{N}
	a_{ij}\varphi_i(x)\varphi_j(t).
	\]

	Тогда
	\[
	\boxed{
		\|K_N-K\|_{L_2([a,b]^2)}
		\to0.
	}
	\]

	То есть сложное ядро \(K\) приблизили конечной суммой очень простых
	функций вида
	\[
	\varphi_i(x)\varphi_j(t).
	\]

	\medskip

	\textbf{4. Почему именно такие простые ядра дают конечномерный оператор?}

	Рассмотрим сначала вообще одно ядро
	\[
	K(x,t)=\varphi_i(x)\varphi_j(t).
	\]

	Тогда
	\[
	\begin{aligned}
		(Af)(x)
		&=
		\int_a^b
		\varphi_i(x)\varphi_j(t)f(t)\,dt
		\\
		&=
		\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
	\end{aligned}
	\]

	Интеграл
	\[
	\int_a^b\varphi_j(t)f(t)\,dt
	\]
	--- просто некоторое число, зависящее от \(f\).

	Значит,
	\[
	Af=c\varphi_i.
	\]

	То есть независимо от того, какую функцию \(f\) мы подали на вход,
	результат всегда лежит в одномерном пространстве
	\[
	\operatorname{span}\{\varphi_i\}.
	\]

	Именно поэтому ядра вида
	\[
	\varphi_i(x)\varphi_j(t)
	\]
	так удобны.

	\medskip

	Теперь для
	\[
	K_N(x,t)
	=
	\sum_{i,j=1}^{N}
	a_{ij}\varphi_i(x)\varphi_j(t)
	\]
	получаем
	\[
	\begin{aligned}
		(A_Nf)(x)
		&=
		\int_a^bK_N(x,t)f(t)\,dt
		\\
		&=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
	\end{aligned}
	\]

	Все интегралы здесь --- просто числа.

	Поэтому результат имеет вид
	\[
	A_Nf
	=
	c_1\varphi_1+\ldots+c_N\varphi_N.
	\]

	Следовательно,
	\[
	\boxed{
		\operatorname{Im}A_N
		\subset
		\operatorname{span}\{\varphi_1,\ldots,\varphi_N\}.
	}
	\]

	Значит,
	\[
	\dim\operatorname{Im}A_N\leq N.
	\]

	То есть \(A_N\) --- конечномерный оператор, а значит компактный.

	\medskip

	\textbf{5. Теперь главное: почему из \(K_N\to K\) следует \(A_N\to A\)?}

	Хотим оценить разность операторов.

	Для \(f\in L_2[a,b]\)
	\[
	((A_N-A)f)(x)
	=
	\int_a^b
	(K_N(x,t)-K(x,t))f(t)\,dt.
	\]

	При фиксированном \(x\) это скалярное произведение двух функций
	переменной \(t\):
	\[
	K_N(x,\cdot)-K(x,\cdot)
	\qquad\text{и}\qquad
	f.
	\]

	По неравенству Коши--Буняковского
	\[
	\left|
	((A_N-A)f)(x)
	\right|^2
	\leq
	\left(
	\int_a^b
	|K_N(x,t)-K(x,t)|^2\,dt
	\right)
	\|f\|_{L_2}^2.
	\]

	Теперь хотим получить норму всей функции \((A_N-A)f\),
	поэтому интегрируем по \(x\):
	\[
	\begin{aligned}
		\|(A_N-A)f\|_{L_2}^2
		&=
		\int_a^b
		|((A_N-A)f)(x)|^2\,dx
		\\
		&\leq
		\int_a^b
		\left(
		\int_a^b
		|K_N(x,t)-K(x,t)|^2\,dt
		\right)dx
		\,
		\|f\|_{L_2}^2.
	\end{aligned}
	\]

	Но двойной интеграл --- это ровно квадрат нормы ядра:
	\[
	\int_a^b\int_a^b
	|K_N(x,t)-K(x,t)|^2\,dt\,dx
	=
	\|K_N-K\|_{L_2([a,b]^2)}^2.
	\]

	Поэтому
	\[
	\|(A_N-A)f\|_{L_2}
	\leq
	\|K_N-K\|_{L_2([a,b]^2)}
	\|f\|_{L_2}.
	\]

	Делим на \(\|f\|_{L_2}\) и берем супремум по \(f\neq0\):
	\[
	\boxed{
		\|A_N-A\|
		\leq
		\|K_N-K\|_{L_2([a,b]^2)}.
	}
	\]

	Но мы уже знаем, что
	\[
	\|K_N-K\|_{L_2([a,b]^2)}\to0.
	\]

	Следовательно,
	\[
	\boxed{
		\|A_N-A\|\to0.
	}
	\]

	\medskip

	\textbf{6. Теперь всё доказательство складывается в одну цепочку.}

	Имеем
	\[
	K\in L_2([a,b]^2).
	\]

	Разлагаем ядро по ОНБ и берем частичные суммы:
	\[
	K_N\to K
	\quad\text{в }L_2([a,b]^2).
	\]

	Каждому \(K_N\) соответствует оператор
	\[
	A_N,
	\]
	причем
	\[
	\operatorname{Im}A_N
	\subset
	\operatorname{span}\{\varphi_1,\ldots,\varphi_N\}.
	\]

	Поэтому
	\[
	A_N
	\text{ конечномерный}
	\quad\Longrightarrow\quad
	A_N
	\text{ компактный}.
	\]

	Кроме того,
	\[
	\|A_N-A\|
	\leq
	\|K_N-K\|_2
	\to0.
	\]

	Следовательно, \(A\) --- предел компактных операторов
	по операторной норме, а значит
	\[
	\boxed{A\text{ компактен}.}
	\]

	\medskip

	\textbf{Главная идея билета в одной строке:}
	\[
	\boxed{
		\text{приближаем ядро конечными суммами}
		\Rightarrow
		\text{получаем конечномерные операторы}
		\Rightarrow
		\text{они сходятся к }A
		\Rightarrow
		A\text{ компактен}.
	}
	\]

	То есть пространство \(L_2([a,b]^2)\), ОНБ и ряд Фурье здесь нужны
	не ради самих себя.

	Они нужны только для одной цели:
	\[
	\boxed{
		\text{приблизить произвольное ядро }K(x,t)
		\text{ конечной суммой разделяющихся функций }
		\varphi_i(x)\varphi_j(t).
	}
	\]

	А такие конечные суммы автоматически порождают конечномерные операторы.
\end{idea}

\subsection{Изометричность вложения \(E\) в \(E^{**}\). Критерий слабой сходимости последовательности}

\begin{definition}[Слабая сходимость]
	Пусть \(E\) --- нормированное пространство. Последовательность
	\(\{x_n\}\subset E\) называется \textbf{слабо сходящейся} к \(x\in E\), если
	\[
	\forall f\in E^*
	\qquad
	f(x_n)\to f(x).
	\]
	Обозначение:
	\[
	x_n\weakconv x.
	\]
\end{definition}

\begin{definition}[Каноническое вложение \(E\) в \(E^{**}\)]
	Пусть \(E\) --- нормированное пространство. Для каждого \(x\in E\) определим функционал
	\[
	F_x:E^*\to\K,
	\qquad
	F_x(f)=f(x).
	\]
	Отображение
	\[
	J:E\to E^{**},
	\qquad
	Jx=F_x,
	\]
	называется \textbf{каноническим вложением} пространства \(E\) в \(E^{**}\).
\end{definition}

\begin{theorem}[Изометричность канонического вложения]
	Каноническое вложение
	\[
	J:E\to E^{**},
	\qquad
	(Jx)(f)=f(x),
	\]
	является линейной изометрией:
	\[
	\|Jx\|_{E^{**}}=\|x\|_E
	\qquad
	\forall x\in E.
	\]
\end{theorem}

\begin{proof}

	\textbf{1. Линейность отображения \(J\).}

	Для любых \(x,y\in E\), \(\alpha,\beta\in\K\) и \(f\in E^*\) имеем
	\[
	J(\alpha x+\beta y)(f)
	=
	f(\alpha x+\beta y)
	=
	\alpha f(x)+\beta f(y)
	=
	(\alpha Jx+\beta Jy)(f).
	\]
	Значит,
	\[
	J(\alpha x+\beta y)=\alpha Jx+\beta Jy.
	\]

	\textbf{2. Верхняя оценка нормы.}

	Для любого \(f\in E^*\) с \(\|f\|\leqslant 1\):
	\[
	|(Jx)(f)|
	=
	|f(x)|
	\leqslant
	\|f\|\,\|x\|
	\leqslant
	\|x\|.
	\]

	Следовательно,
	\[
	\|Jx\|
	=
	\sup_{\|f\|\leqslant 1}|(Jx)(f)|
	\leqslant
	\|x\|.
	\]о

	\textbf{3. Нижняя оценка нормы.}

	Если \(x=0\), то равенство очевидно. Пусть \(x\ne0\). По следствию из теоремы Хана--Банаха
	существует функционал \(f_0\in E^*\), такой что
	\[
	\|f_0\|=1,
	\qquad
	|f_0(x)|=\|x\|.
	\]
	Тогда
	\[
	\|Jx\|
	=
	\sup_{\|f\|\leqslant1}|(Jx)(f)|
	\geqslant
	|(Jx)(f_0)|
	=
	|f_0(x)|
	=
	\|x\|.
	\]

	Итак,
	\[
	\|Jx\|\leqslant \|x\|
	\qquad
	\text{и}
	\qquad
	\|Jx\|\geqslant \|x\|.
	\]
	Следовательно,
	\[
	\|Jx\|=\|x\|.
	\]
	Отображение \(J\) линейно и сохраняет норму, значит, \(J\) --- линейная изометрия.
\end{proof}

\begin{theorem}[Критерий слабой сходимости последовательности]
	Пусть \(E\) --- нормированное пространство.

	Последовательность \(x_n\) слабо сходится к \(x\) тогда и только
	тогда, когда выполнены два условия:

	\begin{enumerate}
		\item Последовательность норм ограничена:
		\(\exists M:\ \sup_n \|x_n\|\leqslant M.\)

		\item Сходимость есть на плотном множестве:
		\(\exists S\subset E^*,\ \overline{[S]}=E^*\) такое, что
		\(\forall s\in S:\ s(x_n)\to s(x).\)
	\end{enumerate}
\end{theorem}

\begin{proof}
	\textbf{1. Переход к операторам \(Jx_n\).}

	По определению слабой сходимости и канонического вложения \(J\):
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\forall f\in E^*:\
	(Jx_n)(f)\to(Jx)(f).
	\]
	То есть
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	Jx_n\to Jx
	\quad\text{поточечно на }E^*.
	\]

	\textbf{2. Применение критерия поточечной сходимости.}

	Имеем
	\[
	Jx_n,Jx\in E^{**}=\mathcal{L}(E^*,\K).
	\]
	Пространство \(E^*\) банахово, а по условию
	\[
	\overline{[S]}=E^*.
	\]
	По критерию поточечной сходимости
	\[
	Jx_n\to Jx
	\]
	поточечно тогда и только тогда, когда
	\[
	\{\|Jx_n\|\}\ \text{ограничена}
	\]
	и
	\[
	(Jx_n)(s)\to(Jx)(s)
	\qquad
	\forall s\in S.
	\]

	\textbf{3. Возврат к \(x_n\).}

	Так как \(J\) --- изометрия и
	\[
	(Jx_n)(s)=s(x_n),
	\]
	получаем
	\[
	\{\|x_n\|\}\ \text{ограничена},
	\qquad
	s(x_n)\to s(x)
	\quad
	\forall s\in S.
	\]

	Следовательно,
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\begin{cases}
		\{\|x_n\|\}\ \text{ограничена},\\
		s(x_n)\to s(x),\quad \forall s\in S.
	\end{cases}
	\]
\end{proof}

\subsection{Слабая сходимость и линейные ограниченные операторы. Слабо ограниченные множества. Теорема Хана}

\begin{theorem}[Слабая сходимость и ограниченный линейный оператор]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in \mathcal{L}(E_1,E_2).
	\]
	Если
	\[
	x_n\weakconv x
	\qquad \text{в } E_1,
	\]
	то
	\[
	Ax_n\weakconv Ax
	\qquad \text{в } E_2.
	\]
\end{theorem}

\begin{proof}
	\textbf{1. Построим функционал на \(E_1\).}

	Возьмём произвольный \(g\in E_2^*\) и рассмотрим
	\[
	f=g\circ A.
	\]
	Покажем, что \(f\in E_1^*\). Он линеен, и
	\[
	|f(y)|
	=
	|g(Ay)|
	\leqslant
	\|g\|\,\|A\|\,\|y\|.
	\]
	Значит,
	\[
	f=g\circ A\in E_1^*.
	\]

	\textbf{2. Применим слабую сходимость.}

	Из
	\[
	x_n\weakconv x
	\]
	для \(f=g\circ A\in E_1^*\) получаем
	\[
	(g\circ A)(x_n)\to(g\circ A)(x),
	\]
	то есть
	\[
	g(Ax_n)\to g(Ax).
	\]

	Так как \(g\in E_2^*\) произволен,
	\[
	Ax_n\weakconv Ax.
	\]
\end{proof}

\begin{definition}[Слабо ограниченное множество]
	Пусть \(E\) --- нормированное пространство. Множество
	\[
	S\subset E
	\]
	называется \textbf{слабо ограниченным}, если для любого функционала
	\(f\in E^*\) множество
	\[
	f(S)=\{f(x):x\in S\}
	\]
	ограничено в \(\K\).

	Иными словами,
	\[
	\forall f\in E^*
	\quad
	\exists C(f)>0
	\quad
	\forall x\in S
	\qquad
	|f(x)|\leqslant C(f).
	\]
	Важно: константа \(C(f)\) может зависеть от функционала \(f\), но не зависит от точки \(x\in S\).
\end{definition}

\begin{theorem}[Хана]
	Пусть \(E\) --- нормированное пространство и \(S\subset E\). Тогда
	\[
	S \text{ слабо ограничено}
	\quad\Longleftrightarrow\quad
	S \text{ ограничено}.
	\]
\end{theorem}

\begin{proof}
	Докажем обе импликации.

	\textbf{1. Ограниченность \(\Rightarrow\) слабая ограниченность.}

	Пусть \(S\) ограничено. Тогда существует \(R>0\), такое что
	\[
	\|x\|\leqslant R
	\qquad
	\forall x\in S.
	\]
	Возьмём произвольный функционал \(f\in E^*\). Тогда для любого \(x\in S\)
	\[
	|f(x)|
	\leqslant
	\|f\|\cdot\|x\|
	\leqslant
	\|f\|\cdot R.
	\]
	Значит, множество \(f(S)\) ограничено. Так как \(f\in E^*\) был произвольным,
	то \(S\) слабо ограничено.

	\textbf{2. Слабая ограниченность \(\Rightarrow\) ограниченность.}

	Докажем от противного. Пусть \(S\) слабо ограничено, но не ограничено.

	\begin{itemize}
		\item \textbf{Выберем быстро растущую последовательность.}

		Из неограниченности \(S\) следует, что для каждого \(n\in\mathbb{N}\)
		можно выбрать
		\[
		x_n\in S
		\]
		так, что
		\[
		\|x_n\|>n^2.
		\]
		Положим
		\[
		y_n=\frac{x_n}{n}.
		\]

		\item \textbf{Покажем, что \(y_n\weakconv 0\).}

		Возьмём произвольный функционал \(f\in E^*\). Так как \(S\) слабо ограничено,
		то существует \(C(f)>0\), такое что
		\[
		|f(x)|\leqslant C(f)
		\qquad
		\forall x\in S.
		\]
		В частности,
		\[
		|f(x_n)|\leqslant C(f).
		\]
		Тогда
		\[
		|f(y_n)|
		=
		\left|f\left(\frac{x_n}{n}\right)\right|
		=
		\frac{1}{n}|f(x_n)|
		\leqslant
		\frac{C(f)}{n}
		\to 0.
		\]
		Следовательно,
		\[
		f(y_n)\to 0=f(0)
		\qquad
		\forall f\in E^*.
		\]
		По определению слабой сходимости
		\[
		y_n\weakconv 0.
		\]

		\item \textbf{Получим противоречие с ограниченностью слабо сходящейся последовательности.}

		По лемме из предыдущего билета всякая слабо сходящаяся последовательность ограничена.
		Значит, последовательность \(\{y_n\}\) должна быть ограничена.

		Но по построению
		\[
		\|y_n\|
		=
		\left\|\frac{x_n}{n}\right\|
		=
		\frac{1}{n}\|x_n\|
		>
		\frac{1}{n}\cdot n^2
		=
		n.
		\]
		Следовательно,
		\[
		\|y_n\|>n
		\qquad
		\forall n\in\mathbb{N}.
		\]
		То есть \(\{y_n\}\) не ограничена. Противоречие.
	\end{itemize}

	Значит, предположение о неограниченности \(S\) неверно. Следовательно,
	\[
	S \text{ ограничено}.
	\]
\end{proof}

\subsection{Замкнутый шар в гильбертовом пространстве слабо секвенциально компактен (теорема Банаха)}

\begin{definition}[Слабая секвенциальная компактность]
	Пусть \(E\) --- нормированное пространство. Множество
	\[
	M\subset E
	\]
	называется \textbf{слабо секвенциально компактным}, если из любой последовательности
	\[
	\{x_n\}\subset M
	\]
	можно выделить подпоследовательность \(\{x_{n_k}\}\), слабо сходящуюся к некоторому
	элементу \(x\in M\):
	\[
	x_{n_k}\weakconv x.
	\]
\end{definition}

\begin{remark}[Слабая сходимость в гильбертовом пространстве]
	Пусть \(H\) --- гильбертово пространство. По теореме Рисса--Фреше каждый функционал
	\(f\in H^*\) имеет вид
	\[
	f(z)=(z,y)
	\]
	для некоторого \(y\in H\). Поэтому
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\forall y\in H
	\qquad
	(x_n,y)\to(x,y).
	\]
\end{remark}

\begin{theorem}[Банаха]
	Пусть \(H\) --- гильбертово пространство и
	\[
	\overline{B}_R(0)=\{x\in H:\|x\|\leqslant R\}.
	\]
	Тогда замкнутый шар \(\overline{B}_R(0)\) слабо секвенциально компактен.

\end{theorem}

\begin{proof}
	Пусть
	\[
	x_n\in\overline{B}_R(0),
	\qquad
	\|x_n\|\leqslant R.
	\]
	Нужно выделить подпоследовательность, слабо сходящуюся в \(H\).

	\textbf{1. Сведение к подпространству, порождённому последовательностью.}

	Для слабой сходимости нужно получить сходимость скалярных произведений
	\[
	(x_{n_k},h)
	\qquad
	\forall h\in H.
	\]
	Сразу выбирать подпоследовательность для всех \(h\in H\) невозможно.
	Поэтому рассмотрим
	\[
	L=\overline{\operatorname{span}}\{x_1,x_2,\ldots\}.
	\]
	Все \(x_n\) лежат в \(L\), а
	\[
	L_0=\operatorname{span}\{x_1,x_2,\ldots\}
	\]
	плотно в \(L\).

	Сначала построим слабый предел в \(L\), а затем перейдём к \(H\).

	\textbf{2. Получим сходимость на плотном множестве \(L_0\).}

	Для каждого фиксированного \(m\)
	\[
	|(x_m,x_n)|
	\leqslant
	\|x_m\|\,\|x_n\|
	\leqslant
	R^2.
	\]
	Значит, последовательность чисел \(\{(x_m,x_n)\}\) ограничена, поэтому из неё можно выделить сходящуюся подпоследовательность.

	Последовательно делая это для \(m=1,2,\ldots\) и применяя диагональный метод, получаем подпоследовательность
	\[
	y_k=x_{n_k},
	\]
	для которой при каждом фиксированном \(m\)
	\[
	(x_m,y_k)
	\]
	сходится.в

	\textbf{3. Построим кандидат на слабый предел.}

	Для каждого \(z\in L_0\) последовательность \((z,y_k)\) сходится, поэтому корректно определён функционал
	\[
	F(z)=\lim_{k\to\infty}(z,y_k).
	\]
	Он линеен как предел линейных функционалов.

	Покажем, что \(F\) ограничен. Так как \(\|y_k\|\leqslant R\), то
	\[
	|F(z)|
	=
	\lim_{k\to\infty}|(z,y_k)|
	\leqslant
	R\|z\|.
	\]
	Значит,
	\[
	\|F\|\leqslant R.
	\]

	Так как \(L_0\) плотно в \(L\), ограниченный функционал \(F\) единственным образом
	продолжается на всё \(L\), то есть
	\[
	F\in L^*.
	\]

	Теперь по теореме Рисса--Фреше существует \(x\in L\), такой что
	\[
	F(z)=(z,x)
	\qquad
	\forall z\in L,
	\]
	причём
	\[
	\|x\|=\|F\|\leqslant R.
	\]
	Следовательно,
	\[
	x\in\overline{B}_R(0).
	\]

	\textbf{4. Получим слабую сходимость.}

	По построению для любого \(z\in L_0\)
	\[
	(z,y_k)\to(z,x).
	\]
	Кроме того,
	\[
	\|y_k\|\leqslant R,
	\]
	а \(L_0\) плотно в \(L\). Поэтому по критерию слабой сходимости
	\[
	y_k\weakconv x
	\qquad
	\text{в }L.
	\]

	Осталось перейти к \(H\). Для любого \(h\in H\)
	\[
	h=l+l^\perp,
	\qquad
	l\in L,\quad l^\perp\in L^\perp.
	\]
	Так как \(y_k,x\in L\),
	\[
	(y_k,h)=(y_k,l)\to(x,l)=(x,h).
	\]
	Следовательно,
	\[
	y_k\weakconv x
	\qquad
	\text{в }H.
	\]

	Таким образом, \(\overline{B}_R(0)\) слабо секвенциально компактен.
\end{proof}пун

\subsection{Обратимость линейного, ограниченного снизу \texorpdfstring{\((\|Ax\|\geqslant k\|x\|)\)}{(||Ax|| >= k||x||)} оператора}

\begin{definition}[Обратимость оператора на образе]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Оператор \(A\) называется \textbf{обратимым на своём образе}, если для любого
	\[
	y\in \operatorname{Im}A
	\]
	уравнение
	\[
	Ax=y
	\]
	имеет единственное решение \(x\in E_1\).

	В этом случае определён обратный оператор
	\[
	A^{-1}:\operatorname{Im}A\to E_1,
	\]
	задаваемый правилом
	\[
	A^{-1}y=x
	\quad\Longleftrightarrow\quad
	Ax=y.
	\]
\end{definition}

\begin{definition}[Ограниченный снизу оператор]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Оператор \(A\) называется \textbf{ограниченным снизу}, если существует число
	\[
	k>0,
	\]
	такое что
	\[
	\|Ax\|_{E_2}\geqslant k\|x\|_{E_1}
	\qquad
	\forall x\in E_1.
	\]
\end{definition}

\begin{lemma}[Ограниченность снизу влечёт инъективность]
	Если оператор
	\[
	A\in\mathcal{L}(E_1,E_2)
	\]
	ограничен снизу, то
	\[
	\ker A=\{0\}.
	\]
	Следовательно, \(A\) инъективен.
\end{lemma}

\begin{theorem}[Критерий ограниченности обратного оператора]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Тогда обратный оператор
	\[
	A^{-1}:\operatorname{Im}A\to E_1
	\]
	существует и ограничен тогда и только тогда, когда оператор \(A\) ограничен снизу, то есть
	\[
	\exists k>0
	\quad
	\forall x\in E_1
	\qquad
	\|Ax\|\geqslant k\|x\|.
	\]
\end{theorem}


\begin{remark}[Полезное следствие: замкнутость образа]
	Если \(E_1\) --- банахово пространство и оператор
	\[
	A\in\mathcal{L}(E_1,E_2)
	\]
	ограничен снизу, то
	\[
	\operatorname{Im}A
	\]
	замкнут в \(E_2\).

\end{remark}

\subsection{Обратимость возмущенного оператора \(A+\Delta A\)}

\begin{theorem}[Обратимость возмущенного оператора]
	\textbf{Условия:}
	\begin{itemize}
		\item $E$ --- банахово пространство;
		\item $A \in \mathcal{L}(E)$ --- обратимый оператор, причем
		$A^{-1} \in \mathcal{L}(E)$;
		\item $\Delta A \in \mathcal{L}(E)$;
		\item выполнено условие
		\[
		\|A^{-1}\|\,\|\Delta A\| < 1.
		\]
	\end{itemize}

	\textbf{Следовательно:}
	\begin{itemize}
		\item оператор $A+\Delta A$ обратим;
		\item $(A+\Delta A)^{-1} \in \mathcal{L}(E)$.
	\end{itemize}
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item \textbf{Сведение к оператору вида $I+B$.}
		Положим
		\[
		B:=A^{-1}\Delta A.
		\]
		Тогда
		\[
		\|B\|
		\leqslant
		\|A^{-1}\|\,\|\Delta A\|
		<1.
		\]
		Кроме того,
		\[
		A+\Delta A
		=
		A(I+A^{-1}\Delta A)
		=
		A(I+B).
		\]
		Так как $A$ обратим, достаточно доказать обратимость $I+B$.

		\item \textbf{Ряд Неймана.}
		Сначала рассмотрим ряд
		\[
		S:=\sum_{n=0}^{\infty}B^n.
		\]
		Так как
		\[
		\|B^n\|\leqslant\|B\|^n
		\]
		и при $\|B\|<1$ геометрический ряд
		\[
		\sum_{n=0}^{\infty}\|B\|^n
		\]
		сходится, то ряд $S$ сходится в банаховом пространстве
		$\mathcal{L}(E)$.

		Для частичной суммы
		\[
		S_N=I+B+\ldots+B^N
		\]
		имеем
		\[
		(I-B)S_N=S_N(I-B)=I-B^{N+1}.
		\]
		Поскольку
		\[
		\|B^{N+1}\|
		\leqslant
		\|B\|^{N+1}
		\to0,
		\]
		при $N\to\infty$ получаем
		\[
		(I-B)S=S(I-B)=I.
		\]
		Следовательно,
		\[
		(I-B)^{-1}
		=
		\sum_{n=0}^{\infty}B^n.
		\]

		\item \textbf{Обратимость $I+B$.}
		Применим полученный результат к оператору $-B$.
		Так как
		\[
		\|-B\|=\|B\|<1,
		\]
		то $I-(-B)=I+B$ обратим и
		\[
		(I+B)^{-1}
		=
		\sum_{n=0}^{\infty}(-B)^n
		=
		I-B+B^2-B^3+\ldots.
		\]

		\item \textbf{Обратимость $A+\Delta A$.}
		Мы получили
		\[
		A+\Delta A=A(I+B),
		\]
		где оба множителя обратимы. Поэтому $A+\Delta A$ обратим, причем
		\[
		(A+\Delta A)^{-1}
		=
		(I+B)^{-1}A^{-1}.
		\]
		Так как
		\[
		(I+B)^{-1},\,A^{-1}\in\mathcal{L}(E),
		\]
		их композиция также принадлежит $\mathcal{L}(E)$, поэтому
		\[
		(A+\Delta A)^{-1}\in\mathcal{L}(E).
		\]
	\end{enumerate}
\end{proof}

\begin{corollary}
	В условиях теоремы
	\[
	(A+\Delta A)^{-1}
	=
	\sum_{n=0}^{\infty}
	(-A^{-1}\Delta A)^n A^{-1}.
	\]
	Кроме того,
	\[
	\|(A+\Delta A)^{-1}\|
	\leqslant
	\frac{\|A^{-1}\|}
	{1-\|A^{-1}\|\,\|\Delta A\|}.
	\]
\end{corollary}

\begin{proof}
	Из доказательства теоремы
	\[
	(I+B)^{-1}
	=
	\sum_{n=0}^{\infty}(-B)^n,
	\qquad
	B=A^{-1}\Delta A.
	\]
	Поэтому
	\[
	(A+\Delta A)^{-1}
	=
	(I+B)^{-1}A^{-1}
	=
	\sum_{n=0}^{\infty}
	(-A^{-1}\Delta A)^n A^{-1}.
	\]

	Также
	\[
	\|(I+B)^{-1}\|
	\leqslant
	\sum_{n=0}^{\infty}\|B\|^n
	=
	\frac{1}{1-\|B\|}.
	\]
	Следовательно,
	\[
	\|(A+\Delta A)^{-1}\|
	\leqslant
	\frac{\|A^{-1}\|}
	{1-\|A^{-1}\|\,\|\Delta A\|}.
	\]
\end{proof}

\begin{idea}
	\textbf{Суть:} Обратимость устойчива относительно достаточно малых
	возмущений. Если $A$ обратим и
	\[
	\|\Delta A\| < \frac{1}{\|A^{-1}\|},
	\]
	то $A+\Delta A$ также имеет ограниченный обратный оператор.
\end{idea}

\subsection{Формулировка теоремы Банаха об обратном операторе.
	Доказательство теоремы Банаха в случае гильбертова пространства}

\begin{theorem}[Банаха об обратном операторе]
	Пусть \(E_1,E_2\) --- банаховы пространства,
	\[
	A\in\mathcal{L}(E_1,E_2),
	\]
	и \(A:E_1\to E_2\) --- биекция. Тогда
	\[
	A^{-1}\in\mathcal{L}(E_2,E_1).
	\]
\end{theorem}

\begin{theorem}[Гильбертов случай]
	Пусть \(H\) --- гильбертово пространство,
	\[
	A\in\mathcal{L}(H),
	\]
	и \(A:H\to H\) --- биекция. Тогда
	\[
	A^{-1}\in\mathcal{L}(H).
	\]
\end{theorem}

\begin{proof}
	Сначала получим ограниченный обратный к \(A^*\), а затем применим то же рассуждение
	к оператору \(A^*\).

	\textbf{1. Докажем, что \(A^*\) ограничен снизу.}

	По билету 8.1 для этого достаточно получить оценку
	\[
	\exists m>0
	\quad
	\forall x\in H
	\qquad
	\|A^*x\|\geqslant m\|x\|.
	\]

	Оценка однородна по \(x\), поэтому достаточно рассмотреть векторы, для которых
	\[
	\|A^*x\|=1.
	\]
	Положим
	\[
	S=\{x\in H:\|A^*x\|=1\}.
	\]
	Тогда достаточно доказать, что \(S\) ограничено:
	\[
	\exists C>0
	\quad
	\forall x\in S
	\qquad
	\|x\|\leqslant C.
	\]

	По теореме Хана достаточно показать, что \(S\) слабо ограничено.

	Пусть \(y\in H\). Так как \(A\) сюръективен, существует \(z\in H\), такое что
	\[
	Az=y.
	\]
	Тогда для любого \(x\in S\)
	\[
	|(x,y)|
	=
	|(x,Az)|
	=
	|(A^*x,z)|
	\leqslant
	\|A^*x\|\|z\|
	=
	\|z\|.
	\]
	Значит, \(S\) слабо ограничено, а по теореме Хана --- ограничено:
	\[
	\exists C>0
	\quad
	\forall x\in S
	\qquad
	\|x\|\leqslant C.
	\]

	Теперь перейдём от ограниченности \(S\) к оценке снизу.
	Для \(u\neq0\) сначала проверим, что \(A^*u\neq0\).

	Так как \(A\) сюръективен,
	\[
	\operatorname{Im}A=H,
	\]
	поэтому
	\[
	\ker A^*
	=
	(\operatorname{Im}A)^\perp
	=
	\{0\}.
	\]
	Следовательно,
	\[
	\frac{u}{\|A^*u\|}\in S.
	\]
	Отсюда
	\[
	\frac{\|u\|}{\|A^*u\|}
	\leqslant C,
	\]
	то есть
	\[
	\|A^*u\|
	\geqslant
	\frac1C\|u\|.
	\]
	Значит, \(A^*\) ограничен снизу.

	\textbf{2. Покажем, что \(A^*\) --- биекция.}

	Инъективность уже получена:
	\[
	\ker A^*=\{0\}.
	\]

	Осталось доказать сюръективность. Так как \(A\) инъективен,
	\[
	\ker A=\{0\}.
	\]
	По разложению
	\[
	H=\overline{\operatorname{Im}A^*}\oplus\ker A
	\]
	получаем
	\[
	\overline{\operatorname{Im}A^*}=H.
	\]

	С другой стороны, \(A^*\) ограничен снизу, поэтому по билету 8.1
	\[
	\operatorname{Im}A^*
	\]
	замкнут. Следовательно,
	\[
	\operatorname{Im}A^*
	=
	\overline{\operatorname{Im}A^*}
	=
	H.
	\]

	Таким образом, \(A^*\) --- биекция. Так как \(A^*\) ограничен снизу,
	по билету 8.1
	\[
	(A^*)^{-1}\in\mathcal L(H).
	\]

	\textbf{3. Применим доказанное к \(A^*\).}

	Оператор
	\[
	A^*\in\mathcal L(H)
	\]
	является биекцией. Поэтому, повторяя доказанное выше для \(A^*\), получаем
	\[
	((A^*)^*)^{-1}\in\mathcal L(H).
	\]
	Так как
	\[
	(A^*)^*=A,
	\]
	имеем
	\[
	\boxed{A^{-1}\in\mathcal L(H)}.
	\]
\end{proof}

\subsection{Операторнозначные функции комплексного переменного. Аналитичность резольвенты. Спектральный радиус. Основная теорема о спектре}

\begin{definition}[Операторнозначная функция комплексного переменного]
	Пусть \(E\) --- банахово пространство над \(\mathbb{C}\), а \(D\subset\mathbb{C}\).
	Отображение
	\[
	F:D\to\mathcal{L}(E)
	\]
	называется \textbf{операторнозначной функцией комплексного переменного}.
\end{definition}

\begin{definition}[Аналитичность операторнозначной функции]
	Функция
	\[
	F:D\to\mathcal{L}(E)
	\]
	называется \textbf{аналитической} в точке \(\lambda_0\in D\), если существует предел
	\[
	F'(\lambda_0)
	=
	\lim_{\lambda\to\lambda_0}
	\frac{F(\lambda)-F(\lambda_0)}{\lambda-\lambda_0}
	\]
	в норме пространства \(\mathcal{L}(E)\).

	Если \(F\) аналитична в каждой точке области \(D\), то \(F\) называется аналитической в \(D\).
\end{definition}


\begin{definition}[Резольвентное множество, резольвента и спектр]
	Пусть \(E\) --- комплексное банахово пространство,
	\[
	A\in\mathcal{L}(E).
	\]
	Для \(\lambda\in\mathbb{C}\) обозначим
	\[
	A_\lambda=A-\lambda I.
	\]

	Число \(\lambda\in\mathbb{C}\) называется \textbf{регулярным значением} оператора \(A\), если оператор
	\[
	A_\lambda=A-\lambda I
	\]
	непрерывно обратим, то есть
	\[
	A_\lambda^{-1}\in\mathcal{L}(E).
	\]

	\textbf{Резольвентным множеством} называется множество
	\[
	\rho(A)=\{\lambda\in\mathbb{C}: A_\lambda^{-1}\in\mathcal{L}(E)\}.
	\]

	Для \(\lambda\in\rho(A)\) оператор
	\[
	R_\lambda=R_\lambda(A)=(A-\lambda I)^{-1}
	\]
	называется \textbf{резольвентой} оператора \(A\).

	\textbf{Спектром} оператора \(A\) называется множество
	\[
	\sigma(A)=\mathbb{C}\setminus\rho(A).
	\]
\end{definition}

\begin{definition}[Спектральный радиус]
	Спектральным радиусом оператора \(A\) называется число
	\[
	r(A)=\sup_{\lambda\in\sigma(A)}|\lambda|.
	\]
\end{definition}

\begin{theorem}[Основные свойства резольвенты и спектра]
	Пусть \(E\) --- комплексное банахово пространство,
	\[
	A\in\mathcal{L}(E).
	\]
	Тогда выполнены следующие свойства.

	\begin{enumerate}
		\item \textbf{Резольвента вне круга радиуса \(\|A\|\).}

		Если
		\[
		|\lambda|>\|A\|,
		\]
		то
		\[
		\lambda\in\rho(A),
		\]
		причём
		\[
		R_\lambda
		=
		(A-\lambda I)^{-1}
		=
		-\frac{1}{\lambda}
		\sum_{n=0}^{\infty}\frac{A^n}{\lambda^n}.
		\]

		Следовательно,
		\[
		\sigma(A)
		\subset
		\{\lambda\in\mathbb{C}:|\lambda|\leqslant\|A\|\}.
		\]

		\item \textbf{Открытость резольвентного множества.}

		Множество
		\[
		\rho(A)
		\]
		открыто в \(\mathbb{C}\).

		\item \textbf{Аналитичность резольвенты.}

		Функция
		\[
		R_\lambda=(A-\lambda I)^{-1}
		\]
		аналитична на \(\rho(A)\), причём
		\[
		R'_\lambda=R_\lambda^2,
		\qquad
		\lambda\in\rho(A).
		\]
	\end{enumerate}
\end{theorem}

\begin{definition}[Спектральный радиус]
	Спектральным радиусом оператора \(A\) называется число
	\[
	r(A)
	=
	\sup_{\lambda\in\sigma(A)}|\lambda|.
	\]
\end{definition}

\begin{theorem}[Основная теорема о спектре]
	Пусть \(E\) --- комплексное банахово пространство,
	\[
	A\in\mathcal{L}(E).
	\]
	Тогда:

	\begin{enumerate}
		\item спектр
		\[
		\sigma(A)
		\]
		является непустым компактным подмножеством \(\mathbb{C}\);

		\item спектральный радиус задаётся формулой
		\[
		r(A)
		=
		\lim_{n\to\infty}
		\sqrt[n]{\|A^n\|}.
		\]
	\end{enumerate}
\end{theorem}
