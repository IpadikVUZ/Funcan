\subsection{Изометричность вложения \(E\) в \(E^{**}\). Критерий слабой сходимости последовательности}

\begin{definition}[Слабая сходимость]
	Пусть \(E\) --- нормированное пространство над полем \(\K\).
	Говорят, что последовательность \(\{x_n\}_{n=1}^{\infty}\subset E\)
	\textbf{слабо сходится} к элементу \(x\in E\), если
	\[
	\forall f\in E^*
	\qquad
	f(x_n)\to f(x).
	\]
	В этом случае пишут
	\[
	x_n \weakconv x.
	\]
\end{definition}

\begin{remark}[Полезные свойства слабой сходимости]
	Для слабой сходимости полезно помнить следующие факты.

	\begin{enumerate}
		\item \textbf{Сходимость по норме влечёт слабую сходимость.}

		Если
		\[
		\|x_n-x\|\to 0,
		\]
		то
		\[
		x_n\weakconv x.
		\]
		Действительно, для любого \(f\in E^*\)
		\[
		|f(x_n)-f(x)|
		=
		|f(x_n-x)|
		\leqslant
		\|f\|\cdot\|x_n-x\|
		\to 0.
		\]

		\item \textbf{Слабый предел единственен.}

		Если
		\[
		x_n\weakconv x'
		\qquad\text{и}\qquad
		x_n\weakconv x'',
		\]
		то
		\[
		x'=x''.
		\]
		Это следует из того, что
		\[
		f(x')=f(x'')
		\qquad
		\forall f\in E^*,
		\]
		а функционалы из \(E^*\) разделяют точки по следствию из теоремы
		Хана--Банаха.

		\item \textbf{В конечномерном пространстве слабая и сильная сходимости совпадают.}

		Если
		\[
		\dim E<\infty,
		\]
		то
		\[
		x_n\weakconv x
		\quad\Longleftrightarrow\quad
		\|x_n-x\|\to 0.
		\]
		Идея: слабая сходимость даёт сходимость всех координат, а в
		конечномерном пространстве покоординатная сходимость эквивалентна
		сходимости по норме.

		\item \textbf{В бесконечномерном пространстве слабая сходимость слабее сильной.}

		Например, в \(\ell_2\), если
		\[
		e_n=(0,\ldots,0,1,0,\ldots),
		\]
		где единица стоит на \(n\)-м месте, то
		\[
		e_n\weakconv 0,
		\]
		но
		\[
		\|e_n\|=1,
		\]
		поэтому сходимости по норме к нулю нет.

		\item \textbf{В гильбертовом пространстве слабая сходимость проверяется через скалярное произведение.}

		Если \(H\) --- гильбертово пространство, то по теореме Рисса
		\[
		x_n\weakconv x
		\quad\Longleftrightarrow\quad
		\forall y\in H
		\qquad
		\langle x_n,y\rangle\to \langle x,y\rangle.
		\]
	\end{enumerate}
\end{remark}

\begin{definition}[Каноническое вложение \(E\) в \(E^{**}\)]
	Пусть \(E\) --- нормированное пространство. Для каждого \(x\in E\)
	определим функционал
	\[
	F_x:E^*\to \K
	\]
	по формуле
	\[
	F_x(f)=f(x),
	\qquad
	f\in E^*.
	\]
	Тогда отображение
	\[
	J:E\to E^{**},
	\qquad
	Jx=F_x,
	\]
	называется \textbf{каноническим вложением} пространства \(E\) в \(E^{**}\).
\end{definition}

\begin{theorem}[Изометричность канонического вложения]
	Каноническое вложение
	\[
	J:E\to E^{**},
	\qquad
	(Jx)(f)=f(x),
	\]
	является линейной изометрией, то есть
	\[
	\|Jx\|_{E^{**}}=\|x\|_E
	\qquad
	\forall x\in E.
	\]
\end{theorem}

\begin{proof}
	Докажем по шагам.

	\begin{enumerate}
		\item \textbf{Проверим, что \(Jx\in E^{**}\).}

		Зафиксируем \(x\in E\). Нужно показать, что
		\[
		F_x:E^*\to \K,
		\qquad
		F_x(f)=f(x),
		\]
		является линейным ограниченным функционалом на \(E^*\).

		Линейность следует из линейности функционалов \(f\in E^*\):
		\[
		F_x(\alpha f+\beta g)
		=
		(\alpha f+\beta g)(x)
		=
		\alpha f(x)+\beta g(x)
		=
		\alpha F_x(f)+\beta F_x(g).
		\]

		\item \textbf{Проверим ограниченность \(F_x\).}

		Для любого \(f\in E^*\) имеем
		\[
		|F_x(f)|
		=
		|f(x)|
		\leqslant
		\|f\|\cdot \|x\|.
		\]
		Значит,
		\[
		\|F_x\|
		=
		\sup_{\|f\|\leqslant 1}|F_x(f)|
		\leqslant
		\|x\|.
		\]
		Следовательно,
		\[
		F_x\in E^{**}.
		\]

		\item \textbf{Проверим линейность отображения \(J\).}

		Для любых \(x,y\in E\), \(\alpha,\beta\in \K\) и \(f\in E^*\)
		имеем
		\[
		J(\alpha x+\beta y)(f)
		=
		f(\alpha x+\beta y)
		=
		\alpha f(x)+\beta f(y)
		=
		\alpha (Jx)(f)+\beta (Jy)(f).
		\]
		Значит,
		\[
		J(\alpha x+\beta y)=\alpha Jx+\beta Jy.
		\]

		\item \textbf{Докажем верхнюю оценку нормы.}

		Из ограниченности \(F_x\) уже получили
		\[
		\|Jx\|
		=
		\|F_x\|
		\leqslant
		\|x\|.
		\]

		\item \textbf{Докажем нижнюю оценку нормы.}

		Если \(x=0\), то
		\[
		\|Jx\|=\|x\|=0.
		\]

		Пусть теперь \(x\ne 0\). По следствию из теоремы Хана--Банаха
		существует функционал \(f_0\in E^*\), такой что
		\[
		\|f_0\|=1
		\qquad\text{и}\qquad
		|f_0(x)|=\|x\|.
		\]
		Тогда
		\[
		\|Jx\|
		=
		\sup_{\|f\|\leqslant 1}|(Jx)(f)|
		\geqslant
		|(Jx)(f_0)|
		=
		|f_0(x)|
		=
		\|x\|.
		\]

		\item \textbf{Объединяем оценки.}

		Мы доказали
		\[
		\|Jx\|\leqslant \|x\|
		\]
		и
		\[
		\|Jx\|\geqslant \|x\|.
		\]
		Следовательно,
		\[
		\|Jx\|=\|x\|.
		\]

		\item \textbf{Получаем изометрию.}

		Отображение \(J\) линейно и сохраняет норму, поэтому \(J\) является
		линейной изометрией.
	\end{enumerate}
\end{proof}

\begin{remark}[Связь слабой сходимости и поточечной сходимости в \(E^{**}\)]
	По определению канонического вложения
	\[
	(Jx_n)(f)=f(x_n),
	\qquad
	(Jx)(f)=f(x).
	\]
	Поэтому
	\[
	x_n\weakconv x
	\]
	тогда и только тогда, когда
	\[
	\forall f\in E^*
	\qquad
	(Jx_n)(f)\to (Jx)(f).
	\]
	То есть слабая сходимость \(x_n\weakconv x\) означает поточечную сходимость
	функционалов \(Jx_n\) к \(Jx\) на пространстве \(E^*\).
\end{remark}

\begin{lemma}[Слабо сходящаяся последовательность ограничена]
	Пусть \(E\) --- нормированное пространство. Если
	\[
	x_n\weakconv x,
	\]
	то последовательность норм
	\[
	\{\|x_n\|\}_{n=1}^{\infty}
	\]
	ограничена.
\end{lemma}

\begin{proof}
	Докажем через каноническое вложение и теорему Банаха--Штейнгауза.

	\begin{enumerate}
		\item \textbf{Перейдём к функционалам на \(E^*\).}

		Рассмотрим последовательность функционалов
		\[
		Jx_n\in E^{**}.
		\]
		Каждый \(Jx_n\) действует на \(E^*\) по формуле
		\[
		(Jx_n)(f)=f(x_n).
		\]

		\item \textbf{Проверим поточечную ограниченность.}

		Зафиксируем произвольный \(f\in E^*\). Так как
		\[
		x_n\weakconv x,
		\]
		то по определению слабой сходимости
		\[
		f(x_n)\to f(x).
		\]
		Следовательно, числовая последовательность \(\{f(x_n)\}\) ограничена.

		Но
		\[
		(Jx_n)(f)=f(x_n).
		\]
		Поэтому
		\[
		\sup_n |(Jx_n)(f)|<\infty
		\qquad
		\forall f\in E^*.
		\]

		\item \textbf{Применим теорему Банаха--Штейнгауза.}

		Пространство \(E^*\) является банаховым. Семейство операторов
		\[
		\{Jx_n\}\subset \mathcal{L}(E^*,\K)
		\]
		поточечно ограничено. Поэтому по теореме Банаха--Штейнгауза
		\[
		\sup_n\|Jx_n\|<\infty.
		\]

		\item \textbf{Используем изометричность вложения.}

		По теореме об изометричности канонического вложения
		\[
		\|Jx_n\|=\|x_n\|.
		\]
		Значит,
		\[
		\sup_n\|x_n\|<\infty.
		\]
	\end{enumerate}
\end{proof}

\begin{theorem}[Критерий слабой сходимости последовательности]
	Пусть \(E\) --- нормированное пространство,
	\[
	S\subset E^*,
	\qquad
	\overline{\operatorname{span}S}=E^*.
	\]
	Тогда
	\[
	x_n\weakconv x
	\]
	тогда и только тогда, когда выполнены два условия:
	\[
	\begin{cases}
		\text{\(1)\) последовательность \(\{\|x_n\|\}\) ограничена;}\\
		\text{\(2)\) для любого \(s\in S\) выполнено \(s(x_n)\to s(x)\).}
	\end{cases}
	\]
\end{theorem}

\begin{proof}
	Докажем обе стороны.

	\begin{enumerate}
		\item \textbf{Необходимость.}

		Пусть
		\[
		x_n\weakconv x.
		\]
		Тогда по лемме о слабой сходимости
		\[
		\sup_n\|x_n\|<\infty.
		\]
		Это даёт первое условие.

		Так как \(S\subset E^*\), то из определения слабой сходимости сразу
		следует, что
		\[
		s(x_n)\to s(x)
		\qquad
		\forall s\in S.
		\]
		Это даёт второе условие.

		\item \textbf{Достаточность.}

		Пусть выполнены условия:
		\[
		\sup_n\|x_n\|<\infty
		\]
		и
		\[
		s(x_n)\to s(x)
		\qquad
		\forall s\in S.
		\]
		Нужно доказать, что
		\[
		f(x_n)\to f(x)
		\qquad
		\forall f\in E^*.
		\]

		\item \textbf{Сначала докажем сходимость для \(\operatorname{span}S\).}

		Пусть
		\[
		g\in \operatorname{span}S.
		\]
		Тогда
		\[
		g=\sum_{k=1}^{m}\alpha_k s_k,
		\qquad
		s_k\in S.
		\]
		По условию
		\[
		s_k(x_n)\to s_k(x)
		\qquad
		k=1,\ldots,m.
		\]
		Следовательно,
		\[
		g(x_n)
		=
		\sum_{k=1}^{m}\alpha_k s_k(x_n)
		\to
		\sum_{k=1}^{m}\alpha_k s_k(x)
		=
		g(x).
		\]

		\item \textbf{Переходим от \(\operatorname{span}S\) ко всему \(E^*\).}

		Возьмём произвольный функционал \(f\in E^*\).
		Так как
		\[
		\overline{\operatorname{span}S}=E^*,
		\]
		то для любого \(\delta>0\) существует
		\[
		g\in \operatorname{span}S
		\]
		такой, что
		\[
		\|f-g\|<\delta.
		\]

		Обозначим
		\[
		M=\sup_n\|x_n\|<\infty.
		\]

		\item \textbf{Оценим разность \( |f(x_n)-f(x)| \).}

		Имеем
		\[
		|f(x_n)-f(x)|
		\leqslant
		|(f-g)(x_n)|
		+
		|g(x_n)-g(x)|
		+
		|(g-f)(x)|.
		\]
		Крайние слагаемые оцениваются так:
		\[
		|(f-g)(x_n)|
		\leqslant
		\|f-g\|\cdot \|x_n\|
		\leqslant
		\delta M,
		\]
		и
		\[
		|(g-f)(x)|
		\leqslant
		\|g-f\|\cdot \|x\|
		\leqslant
		\delta \|x\|.
		\]
		Значит,
		\[
		|f(x_n)-f(x)|
		\leqslant
		\delta(M+\|x\|)
		+
		|g(x_n)-g(x)|.
		\]

		\item \textbf{Выберем \(\delta\) и используем сходимость для \(g\).}

		Пусть \(\varepsilon>0\). Возьмём
		\[
		\delta=\frac{\varepsilon}{2(M+\|x\|+1)}.
		\]
		Тогда
		\[
		\delta(M+\|x\|)<\frac{\varepsilon}{2}.
		\]

		Так как \(g\in \operatorname{span}S\), то уже доказано, что
		\[
		g(x_n)\to g(x).
		\]
		Поэтому существует \(N\), такое что при \(n\geqslant N\)
		\[
		|g(x_n)-g(x)|<\frac{\varepsilon}{2}.
		\]

		\item \textbf{Получаем сходимость для произвольного \(f\in E^*\).}

		При \(n\geqslant N\)
		\[
		|f(x_n)-f(x)|
		<
		\frac{\varepsilon}{2}
		+
		\frac{\varepsilon}{2}
		=
		\varepsilon.
		\]
		Значит,
		\[
		f(x_n)\to f(x).
		\]

		\item \textbf{Завершаем доказательство.}

		Функционал \(f\in E^*\) был произвольным. Следовательно,
		\[
		\forall f\in E^*
		\qquad
		f(x_n)\to f(x).
		\]
		По определению слабой сходимости это означает
		\[
		x_n\weakconv x.
		\]
	\end{enumerate}
\end{proof}

\subsection{Слабая сходимость и линейные ограниченные операторы. Слабо ограниченные множества. Теорема Хана}

\begin{theorem}[Слабая сходимость и ограниченный линейный оператор]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in \mathcal{L}(E_1,E_2).
	\]
	Если
	\[
	x_n\weakconv x
	\qquad \text{в } E_1,
	\]
	то
	\[
	Ax_n\weakconv Ax
	\qquad \text{в } E_2.
	\]
\end{theorem}

\begin{proof}
	Докажем по определению слабой сходимости.

	\begin{enumerate}
		\item \textbf{Возьмём произвольный функционал на \(E_2\).}

		Пусть
		\[
		g\in E_2^*.
		\]
		Чтобы доказать
		\[
		Ax_n\weakconv Ax,
		\]
		нужно проверить, что
		\[
		g(Ax_n)\to g(Ax).
		\]

		\item \textbf{Составим композицию с оператором \(A\).}

		Рассмотрим отображение
		\[
		f=g\circ A.
		\]
		То есть
		\[
		f(y)=g(Ay),
		\qquad
		y\in E_1.
		\]

		\item \textbf{Проверим, что \(f\in E_1^*\).}

		Линейность \(f\) следует из линейности \(A\) и \(g\).

		Докажем ограниченность. Для любого \(y\in E_1\)
		\[
		|f(y)|
		=
		|g(Ay)|
		\leqslant
		\|g\|\cdot \|Ay\|
		\leqslant
		\|g\|\cdot \|A\|\cdot \|y\|.
		\]
		Следовательно,
		\[
		f\in E_1^*.
		\]

		\item \textbf{Используем слабую сходимость \(x_n\weakconv x\).}

		Так как
		\[
		x_n\weakconv x
		\qquad \text{в } E_1,
		\]
		то для любого функционала \(f\in E_1^*\)
		\[
		f(x_n)\to f(x).
		\]
		Применяя это к функционалу \(f=g\circ A\), получаем
		\[
		(g\circ A)(x_n)\to (g\circ A)(x).
		\]

		\item \textbf{Возвращаемся к \(g\).}

		Имеем
		\[
		(g\circ A)(x_n)=g(Ax_n),
		\qquad
		(g\circ A)(x)=g(Ax).
		\]
		Значит,
		\[
		g(Ax_n)\to g(Ax).
		\]

		\item \textbf{Завершаем доказательство.}

		Функционал \(g\in E_2^*\) был произвольным. Поэтому
		\[
		\forall g\in E_2^*
		\qquad
		g(Ax_n)\to g(Ax).
		\]
		По определению слабой сходимости это означает
		\[
		Ax_n\weakconv Ax.
		\]
	\end{enumerate}
\end{proof}

\begin{definition}[Слабо ограниченное множество]
	Пусть \(E\) --- нормированное пространство. Множество
	\[
	S\subset E
	\]
	называется \textbf{слабо ограниченным}, если для любого функционала
	\(f\in E^*\) множество чисел
	\[
	f(S)=\{f(x):x\in S\}
	\]
	ограничено.

	Иными словами,
	\[
	\forall f\in E^*
	\quad
	\exists C(f)>0
	\quad
	\forall x\in S
	\qquad
	|f(x)|\leqslant C(f).
	\]
\end{definition}

\begin{remark}[Сравнение с обычной ограниченностью]
	Обычная ограниченность множества \(S\subset E\) означает, что
	\[
	\exists C>0
	\quad
	\forall x\in S
	\qquad
	\|x\|\leqslant C.
	\]

	Слабая ограниченность означает, что после применения любого функционала
	\(f\in E^*\) получаем ограниченное числовое множество:
	\[
	\forall f\in E^*
	\quad
	\exists C(f)>0
	\quad
	\forall x\in S
	\qquad
	|f(x)|\leqslant C(f).
	\]

	Важно, что константа \(C(f)\) может зависеть от функционала \(f\),
	но не зависит от точки \(x\in S\).
\end{remark}

\begin{theorem}[Хана]
	Пусть \(E\) --- нормированное пространство и \(S\subset E\). Тогда
	\[
	S \text{ слабо ограничено}
	\quad\Longleftrightarrow\quad
	S \text{ ограничено}.
	\]
\end{theorem}

\begin{proof}
	Докажем обе импликации.

	\begin{enumerate}
		\item \textbf{Обычная ограниченность влечёт слабую ограниченность.}

		Пусть \(S\) ограничено. Тогда
		\[
		\exists R>0
		\quad
		\forall x\in S
		\qquad
		\|x\|\leqslant R.
		\]

		Возьмём произвольный функционал \(f\in E^*\). Тогда для любого
		\(x\in S\)
		\[
		|f(x)|
		\leqslant
		\|f\|\cdot \|x\|
		\leqslant
		\|f\|\cdot R.
		\]
		Значит, множество \(f(S)\) ограничено.

		Так как \(f\in E^*\) был произвольным, получаем, что \(S\) слабо
		ограничено.

		\item \textbf{Докажем обратную импликацию от противного.}

		Пусть \(S\) слабо ограничено, но не ограничено.

		Неограниченность \(S\) означает, что для каждого \(n\in\mathbb{N}\)
		можно выбрать элемент
		\[
		x_n\in S
		\]
		такой, что
		\[
		\|x_n\|>n^2.
		\]

		\item \textbf{Построим последовательность, которая слабо сходится к нулю.}

		Положим
		\[
		y_n=\frac{x_n}{n}.
		\]
		Покажем, что
		\[
		y_n\weakconv 0.
		\]

		Возьмём произвольный функционал \(f\in E^*\). Так как \(S\) слабо
		ограничено, то множество \(f(S)\) ограничено. Значит,
		\[
		\exists C(f)>0
		\quad
		\forall x\in S
		\qquad
		|f(x)|\leqslant C(f).
		\]
		В частности,
		\[
		|f(x_n)|\leqslant C(f).
		\]
		Тогда
		\[
		|f(y_n)|
		=
		\left|f\left(\frac{x_n}{n}\right)\right|
		=
		\frac{1}{n}|f(x_n)|
		\leqslant
		\frac{C(f)}{n}
		\to 0.
		\]
		Следовательно,
		\[
		f(y_n)\to 0=f(0).
		\]
		Так как \(f\in E^*\) был произвольным, то
		\[
		y_n\weakconv 0.
		\]

		\item \textbf{Используем ограниченность слабо сходящейся последовательности.}

		По лемме о слабо сходящейся последовательности из предыдущего билета,
		если
		\[
		y_n\weakconv 0,
		\]
		то последовательность \(\{y_n\}\) ограничена.

		То есть должно существовать число \(M>0\), такое что
		\[
		\|y_n\|\leqslant M
		\qquad
		\forall n.
		\]

		\item \textbf{Получаем противоречие.}

		Но по построению
		\[
		\|y_n\|
		=
		\left\|\frac{x_n}{n}\right\|
		=
		\frac{1}{n}\|x_n\|
		>
		\frac{1}{n}\cdot n^2
		=
		n.
		\]
		Следовательно,
		\[
		\|y_n\|>n
		\qquad
		\forall n.
		\]
		То есть последовательность \(\{y_n\}\) не ограничена.

		Мы получили противоречие с тем, что всякая слабо сходящаяся
		последовательность ограничена.

		\item \textbf{Завершаем доказательство.}

		Значит, предположение о неограниченности \(S\) неверно.
		Следовательно,
		\[
		S \text{ ограничено}.
		\]
	\end{enumerate}
\end{proof}

\subsection{Замкнутый шар в гильбертовом пространстве слабо секвенциально компактен (теорема Банаха)}

\begin{definition}[Слабая секвенциальная компактность]
	Пусть \(E\) --- нормированное пространство. Множество
	\[
	M\subset E
	\]
	называется \textbf{слабо секвенциально компактным}, если из любой
	последовательности \(\{x_n\}\subset M\) можно выделить подпоследовательность
	\(\{x_{n_k}\}\), слабо сходящуюся к некоторому элементу \(x\in M\):
	\[
	x_{n_k}\weakconv x.
	\]
\end{definition}

\begin{remark}[Слабая сходимость в гильбертовом пространстве]
	Пусть \(H\) --- гильбертово пространство. По теореме Рисса--Фреше каждый
	функционал \(f\in H^*\) имеет вид
	\[
	f(z)=(z,y)
	\]
	для некоторого \(y\in H\). Поэтому
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\forall y\in H
	\qquad
	(x_n,y)\to (x,y).
	\]
\end{remark}

\begin{theorem}[Банаха]
	Пусть \(H\) --- гильбертово пространство и
	\[
	\overline{B}_R(0)=\{x\in H:\|x\|\leqslant R\}.
	\]
	Тогда замкнутый шар \(\overline{B}_R(0)\) слабо секвенциально компактен.

	Иными словами, из любой последовательности
	\[
	x_n\in \overline{B}_R(0)
	\]
	можно выделить подпоследовательность \(\{x_{n_k}\}\), такую что
	\[
	x_{n_k}\weakconv x
	\]
	для некоторого
	\[
	x\in \overline{B}_R(0).
	\]
\end{theorem}

\begin{idea}
	Берём последовательность в шаре. Она может жить в несепарабельном пространстве,
	но сама последовательность порождает сепарабельное подпространство
	\[
	L=\overline{\operatorname{span}}\{x_1,x_2,\ldots\}.
	\]
	Дальше диагональным методом добиваемся сходимости всех чисел
	\[
	(x_m,x_{n_k})
	\]
	при фиксированном \(m\). Это даст сходимость на плотном множестве в \(L\).

	Затем по теореме Рисса--Фреше предъявляем элемент \(x\in L\), к которому
	подпоследовательность слабо сходится.
\end{idea}

\begin{proof}
	Докажем по шагам.

	\begin{enumerate}
		\item \textbf{Возьмём произвольную последовательность из шара.}

		Пусть
		\[
		x_n\in \overline{B}_R(0),
		\qquad n\in\mathbb{N}.
		\]
		То есть
		\[
		\|x_n\|\leqslant R
		\qquad
		\forall n\in\mathbb{N}.
		\]
		Нужно выделить подпоследовательность, слабо сходящуюся к элементу
		этого же шара.

		\item \textbf{Перейдём к подпространству, порождённому этой последовательностью.}

		Рассмотрим
		\[
		L=\overline{\operatorname{span}}\{x_1,x_2,\ldots\}\subset H.
		\]
		Тогда
		\[
		x_n\in L
		\qquad
		\forall n.
		\]
		Подпространство \(L\) замкнуто и порождено счётным множеством, поэтому
		оно сепарабельно.

		Достаточно сначала доказать слабую сходимость в \(L\), а затем перейти
		от \(L\) ко всему \(H\) через ортогональное разложение
		\[
		H=L\oplus L^\perp.
		\]

		\item \textbf{Подготовим диагональный процесс.}

		Для фиксированного \(m\in\mathbb{N}\) рассмотрим числовую последовательность
		\[
		(x_m,x_n),
		\qquad n=1,2,\ldots.
		\]
		Она ограничена, потому что по неравенству Коши--Буняковского
		\[
		|(x_m,x_n)|
		\leqslant
		\|x_m\|\cdot\|x_n\|
		\leqslant
		R^2.
		\]
		Значит, из неё можно выделять сходящиеся подпоследовательности.

		\item \textbf{Применим диагональный метод Кантора.}

		Сначала из \(\{x_n\}\) выделим подпоследовательность, на которой сходится
		последовательность чисел
		\[
		(x_1,x_n).
		\]
		Из неё выделим подпоследовательность, на которой сходится
		\[
		(x_2,x_n).
		\]
		Далее продолжаем этот процесс.

		В результате получаем систему вложенных подпоследовательностей. Берём
		диагональную подпоследовательность и обозначаем её снова через
		\[
		y_k=x_{n_k}.
		\]
		Тогда для каждого фиксированного \(m\in\mathbb{N}\) существует предел
		\[
		\lim_{k\to\infty}(x_m,y_k).
		\]

		\item \textbf{Докажем сходимость скалярных произведений на плотном множестве.}

		Пусть
		\[
		z\in \operatorname{span}\{x_1,x_2,\ldots\}.
		\]
		Тогда
		\[
		z=\sum_{m=1}^{N}\alpha_m x_m.
		\]
		Следовательно,
		\[
		(z,y_k)
		=
		\sum_{m=1}^{N}\overline{\alpha_m}(x_m,y_k)
		\]
		с точностью до соглашения о линейности скалярного произведения.

		Так как для каждого \(m\) последовательность \((x_m,y_k)\) сходится,
		то сходится и конечная линейная комбинация:
		\[
		(z,y_k)
		\]
		]
		имеет предел при \(k\to\infty\).

		\item \textbf{Определим функционал на плотном подпространстве.}

		На линейной оболочке
		\[
		L_0=\operatorname{span}\{x_1,x_2,\ldots\}
		\]
		зададим функционал
		\[
		F(z)=\lim_{k\to\infty}(z,y_k).
		\]
		По предыдущему шагу этот предел существует для любого \(z\in L_0\).

		Функционал \(F\) линеен, поскольку предел сохраняет линейные операции.

		\item \textbf{Проверим ограниченность \(F\).}

		Для любого \(z\in L_0\) имеем
		\[
		|F(z)|
		=
		\left|\lim_{k\to\infty}(z,y_k)\right|
		=
		\lim_{k\to\infty}|(z,y_k)|.
		\]
		По неравенству Коши--Буняковского
		\[
		|(z,y_k)|
		\leqslant
		\|z\|\cdot \|y_k\|
		\leqslant
		R\|z\|.
		\]
		Значит,
		\[
		|F(z)|\leqslant R\|z\|.
		\]
		Следовательно, \(F\) ограничен на \(L_0\).

		\item \textbf{Продолжим функционал на всё \(L\).}

		Так как
		\[
		L=\overline{L_0},
		\]
		а \(F\) ограничен на \(L_0\), то \(F\) единственным образом продолжается
		до ограниченного линейного функционала на \(L\). Обозначим продолжение
		тем же символом:
		\[
		F\in L^*.
		\]
		Причём
		\[
		\|F\|\leqslant R.
		\]

		\item \textbf{Применим теорему Рисса--Фреше в пространстве \(L\).}

		Так как \(L\) --- замкнутое подпространство гильбертова пространства,
		то \(L\) само является гильбертовым пространством.

		По теореме Рисса--Фреше существует элемент \(x\in L\), такой что
		\[
		F(z)=(z,x)
		\qquad
		\forall z\in L.
		\]
		Кроме того,
		\[
		\|x\|=\|F\|\leqslant R.
		\]
		Значит,
		\[
		x\in \overline{B}_R(0).
		\]

		\item \textbf{Получим слабую сходимость в \(L\).}

		Покажем, что
		\[
		y_k\weakconv x
		\qquad
		\text{в } L.
		\]
		Для этого нужно доказать, что для любого \(z\in L\)
		\[
		(z,y_k)\to (z,x).
		\]

		Но по построению функционала \(F\)
		\[
		F(z)=\lim_{k\to\infty}(z,y_k).
		\]
		С другой стороны, по выбору \(x\)
		\[
		F(z)=(z,x).
		\]
		Следовательно,
		\[
		(z,y_k)\to (z,x)
		\qquad
		\forall z\in L.
		\]
		То есть
		\[
		y_k\weakconv x
		\qquad
		\text{в } L.
		\]

		\item \textbf{Перейдём от \(L\) ко всему \(H\).}

		Так как \(L\) --- замкнутое подпространство гильбертова пространства,
		по теореме о проекции
		\[
		H=L\oplus L^\perp.
		\]
		Значит, любой элемент \(h\in H\) единственным образом представляется в виде
		\[
		h=l+l^\perp,
		\qquad
		l\in L,\quad l^\perp\in L^\perp.
		\]

		Так как
		\[
		y_k\in L
		\qquad\text{и}\qquad
		x\in L,
		\]
		то
		\[
		(y_k,l^\perp)=0,
		\qquad
		(x,l^\perp)=0.
		\]
		Поэтому
		\[
		(y_k,h)
		=
		(y_k,l+l^\perp)
		=
		(y_k,l)
		\to
		(x,l)
		=
		(x,l+l^\perp)
		=
		(x,h).
		\]

		\item \textbf{Завершаем доказательство.}

		Мы доказали, что для любого \(h\in H\)
		\[
		(y_k,h)\to (x,h).
		\]
		По критерию слабой сходимости в гильбертовом пространстве это означает
		\[
		y_k\weakconv x
		\qquad
		\text{в } H.
		\]
		При этом
		\[
		x\in \overline{B}_R(0).
		\]
		Следовательно, из любой последовательности в шаре
		\(\overline{B}_R(0)\) можно выделить подпоследовательность, слабо
		сходящуюся к элементу того же шара.

		Значит, замкнутый шар в гильбертовом пространстве слабо секвенциально
		компактен.
	\end{enumerate}
\end{proof}

\subsection{Обратимость линейного, ограниченного снизу \texorpdfstring{\((\|Ax\|\geqslant k\|x\|)\)}{(||Ax|| >= k||x||)} оператора}

\begin{definition}[Обратимость оператора на образе]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Оператор \(A\) называется \textbf{обратимым на своём образе}, если для любого
	\[
	y\in \operatorname{Im}A
	\]
	уравнение
	\[
	Ax=y
	\]
	имеет единственное решение \(x\in E_1\).

	В этом случае определён обратный оператор
	\[
	A^{-1}:\operatorname{Im}A\to E_1,
	\]
	задаваемый правилом
	\[
	A^{-1}y=x
	\quad\Longleftrightarrow\quad
	Ax=y.
	\]
\end{definition}

\begin{remark}[Левый и правый обратные]
	Для оператора
	\[
	A:E_1\to E_2
	\]
	можно различать два типа обратных операторов.

	\begin{enumerate}
		\item \textbf{Левый обратный} отвечает за единственность решения:
		\[
		LA=I_{E_1}.
		\]

		\item \textbf{Правый обратный} отвечает за существование решения на образе:
		\[
		AR=I_{\operatorname{Im}A}.
		\]
	\end{enumerate}

	Если оба оператора существуют и совпадают, то получаем обычный обратный оператор
	\[
	A^{-1}.
	\]
	В нашем билете обратный оператор рассматривается именно как оператор
	\[
	A^{-1}:\operatorname{Im}A\to E_1.
	\]
\end{remark}

\begin{definition}[Ограниченный снизу оператор]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Оператор \(A\) называется \textbf{ограниченным снизу}, если существует число
	\[
	k>0,
	\]
	такое что
	\[
	\|Ax\|_{E_2}\geqslant k\|x\|_{E_1}
	\qquad
	\forall x\in E_1.
	\]
\end{definition}

\begin{lemma}[Ограниченность снизу влечёт инъективность]
	Если оператор
	\[
	A\in\mathcal{L}(E_1,E_2)
	\]
	ограничен снизу, то
	\[
	\ker A=\{0\}.
	\]
	Следовательно, \(A\) инъективен.
\end{lemma}

\begin{proof}
	Докажем напрямую.

	\begin{enumerate}
		\item \textbf{Возьмём элемент ядра.}

		Пусть
		\[
		x\in\ker A.
		\]
		Тогда
		\[
		Ax=0.
		\]

		\item \textbf{Используем ограниченность снизу.}

		Так как \(A\) ограничен снизу, существует \(k>0\), такое что
		\[
		\|Ax\|\geqslant k\|x\|.
		\]
		Но \(Ax=0\), поэтому
		\[
		0=\|Ax\|\geqslant k\|x\|.
		\]

		\item \textbf{Получаем, что \(x=0\).}

		Так как \(k>0\), из
		\[
		k\|x\|\leqslant 0
		\]
		следует
		\[
		\|x\|=0.
		\]
		Значит,
		\[
		x=0.
		\]

		\item \textbf{Итог.}

		Мы доказали, что любой элемент ядра равен нулю. Поэтому
		\[
		\ker A=\{0\}.
		\]
		Следовательно, \(A\) инъективен.
	\end{enumerate}
\end{proof}

\begin{theorem}[Критерий ограниченности обратного оператора]
	Пусть \(E_1,E_2\) --- нормированные пространства,
	\[
	A\in\mathcal{L}(E_1,E_2).
	\]
	Тогда обратный оператор
	\[
	A^{-1}:\operatorname{Im}A\to E_1
	\]
	существует и ограничен тогда и только тогда, когда оператор \(A\) ограничен снизу, то есть
	\[
	\exists k>0
	\quad
	\forall x\in E_1
	\qquad
	\|Ax\|\geqslant k\|x\|.
	\]
\end{theorem}

\begin{proof}
	Докажем обе импликации.

	\begin{enumerate}
		\item \textbf{Пусть обратный оператор существует и ограничен.}

		Предположим, что
		\[
		A^{-1}\in\mathcal{L}(\operatorname{Im}A,E_1).
		\]
		Нужно доказать, что \(A\) ограничен снизу.

		Возьмём произвольный элемент
		\[
		x\in E_1.
		\]
		Тогда
		\[
		Ax\in \operatorname{Im}A.
		\]
		По определению обратного оператора
		\[
		A^{-1}Ax=x.
		\]
		Следовательно,
		\[
		\|x\|
		=
		\|A^{-1}Ax\|.
		\]

		\item \textbf{Используем ограниченность \(A^{-1}\).}

		Так как \(A^{-1}\) ограничен, то
		\[
		\|A^{-1}Ax\|
		\leqslant
		\|A^{-1}\|\cdot\|Ax\|.
		\]
		Значит,
		\[
		\|x\|
		\leqslant
		\|A^{-1}\|\cdot\|Ax\|.
		\]
		Если \(x\ne 0\), то \(\|A^{-1}\|\ne 0\), и поэтому
		\[
		\|Ax\|
		\geqslant
		\frac{1}{\|A^{-1}\|}\|x\|.
		\]
		При \(x=0\) это неравенство также верно.

		Таким образом, \(A\) ограничен снизу с константой
		\[
		k=\frac{1}{\|A^{-1}\|}.
		\]

		\item \textbf{Пусть теперь \(A\) ограничен снизу.}

		Предположим, что
		\[
		\exists k>0
		\quad
		\forall x\in E_1
		\qquad
		\|Ax\|\geqslant k\|x\|.
		\]
		Нужно доказать, что обратный оператор
		\[
		A^{-1}:\operatorname{Im}A\to E_1
		\]
		существует и ограничен.

		\item \textbf{Докажем существование обратного оператора.}

		По лемме из ограниченности снизу следует, что
		\[
		\ker A=\{0\}.
		\]
		Значит, \(A\) инъективен.

		Поэтому для каждого
		\[
		y\in \operatorname{Im}A
		\]
		существует единственный \(x\in E_1\), такой что
		\[
		Ax=y.
		\]
		Следовательно, корректно определён обратный оператор
		\[
		A^{-1}:\operatorname{Im}A\to E_1.
		\]

		\item \textbf{Проверим линейность обратного оператора.}

		Пусть
		\[
		y_1,y_2\in \operatorname{Im}A,
		\qquad
		\alpha,\beta\in\mathbb{K}.
		\]
		Тогда существуют единственные \(x_1,x_2\in E_1\), такие что
		\[
		Ax_1=y_1,
		\qquad
		Ax_2=y_2.
		\]
		Имеем
		\[
		A(\alpha x_1+\beta x_2)
		=
		\alpha Ax_1+\beta Ax_2
		=
		\alpha y_1+\beta y_2.
		\]
		По единственности прообраза
		\[
		A^{-1}(\alpha y_1+\beta y_2)
		=
		\alpha x_1+\beta x_2.
		\]
		То есть
		\[
		A^{-1}(\alpha y_1+\beta y_2)
		=
		\alpha A^{-1}y_1+\beta A^{-1}y_2.
		\]
		Следовательно, \(A^{-1}\) линеен.

		\item \textbf{Докажем ограниченность обратного оператора.}

		Возьмём произвольный элемент
		\[
		y\in \operatorname{Im}A.
		\]
		Тогда существует \(x\in E_1\), такой что
		\[
		y=Ax.
		\]
		По определению обратного оператора
		\[
		A^{-1}y=x.
		\]
		Используем ограниченность \(A\) снизу:
		\[
		\|y\|
		=
		\|Ax\|
		\geqslant
		k\|x\|
		=
		k\|A^{-1}y\|.
		\]
		Отсюда
		\[
		\|A^{-1}y\|
		\leqslant
		\frac{1}{k}\|y\|.
		\]

		\item \textbf{Получаем ограниченность \(A^{-1}\).}

		Так как для любого \(y\in\operatorname{Im}A\) выполнено
		\[
		\|A^{-1}y\|
		\leqslant
		\frac{1}{k}\|y\|,
		\]
		то
		\[
		A^{-1}\in\mathcal{L}(\operatorname{Im}A,E_1)
		\]
		и
		\[
		\|A^{-1}\|\leqslant \frac{1}{k}.
		\]

		\item \textbf{Итог.}

		Мы доказали, что
		\[
		A^{-1}\in\mathcal{L}(\operatorname{Im}A,E_1)
		\quad\Longleftrightarrow\quad
		\exists k>0:\ \|Ax\|\geqslant k\|x\|
		\quad \forall x\in E_1.
		\]
	\end{enumerate}
\end{proof}


\begin{remark}[Полезное следствие: замкнутость образа]
	Если \(E_1\) --- банахово пространство и оператор
	\[
	A\in\mathcal{L}(E_1,E_2)
	\]
	ограничен снизу, то
	\[
	\operatorname{Im}A
	\]
	замкнут в \(E_2\).

\end{remark}

\subsection{Обратимость возмущенного оператора \(A+\Delta A\)}

\begin{definition}[Ряд Неймана]
	Пусть \(E\) --- банахово пространство и
	\[
	B\in\mathcal{L}(E).
	\]
	Ряд
	\[
	\sum_{n=0}^{\infty} B^n
	\]
	называется \textbf{рядом Неймана}.
\end{definition}

\begin{theorem}[Обратимость оператора \(I+B\)]
	Пусть \(E\) --- банахово пространство,
	\[
	B\in\mathcal{L}(E),
	\qquad
	\|B\|<1.
	\]
	Тогда оператор
	\[
	I+B
	\]
	обратим, причём
	\[
	(I+B)^{-1}
	=
	\sum_{n=0}^{\infty}(-1)^n B^n.
	\]
	В частности,
	\[
	(I+B)^{-1}\in\mathcal{L}(E).
	\]
\end{theorem}

\begin{proof}
	Докажем по шагам.

	\begin{enumerate}
		\item \textbf{Запишем предполагаемый обратный оператор.}

		Если бы \(B\) был числом и \(|B|<1\), то
		\[
		(1+B)^{-1}
		=
		1-B+B^2-B^3+\ldots
		\]
		Поэтому для оператора \(B\) рассмотрим частичные суммы
		\[
		S_n=I-B+B^2-\ldots+(-1)^nB^n
		=
		\sum_{k=0}^{n}(-1)^kB^k.
		\]

		\item \textbf{Покажем, что \(\{S_n\}\) --- фундаментальная последовательность.}

		Пусть \(p\in\mathbb{N}\). Тогда
		\[
		S_{n+p}-S_n
		=
		\sum_{k=n+1}^{n+p}(-1)^kB^k.
		\]
		Следовательно,
		\[
		\|S_{n+p}-S_n\|
		\leqslant
		\sum_{k=n+1}^{n+p}\|B^k\|.
		\]
		Так как
		\[
		\|B^k\|\leqslant \|B\|^k,
		\]
		то
		\[
		\|S_{n+p}-S_n\|
		\leqslant
		\sum_{k=n+1}^{n+p}\|B\|^k.
		\]
		Поскольку
		\[
		\|B\|<1,
		\]
		числовой геометрический ряд
		\[
		\sum_{k=0}^{\infty}\|B\|^k
		\]
		сходится. Поэтому его хвосты стремятся к нулю, и значит
		\[
		\{S_n\}
		\]
		--- фундаментальная последовательность в \(\mathcal{L}(E)\).

		\item \textbf{Используем полноту пространства операторов.}

		Так как \(E\) --- банахово пространство, то
		\[
		\mathcal{L}(E)
		\]
		тоже является банаховым пространством.

		Следовательно, существует предел
		\[
		S=\lim_{n\to\infty}S_n
		\qquad
		\text{в норме операторов}.
		\]
		То есть
		\[
		S=\sum_{n=0}^{\infty}(-1)^nB^n
		\in \mathcal{L}(E).
		\]

		\item \textbf{Покажем, что \(S\) --- правый обратный к \(I+B\).}

		Для частичных сумм имеем телескопическое равенство:
		\[
		(I+B)S_n
		=
		(I+B)(I-B+B^2-\ldots+(-1)^nB^n)
		=
		I+(-1)^nB^{n+1}.
		\]
		Так как
		\[
		\|B^{n+1}\|\leqslant \|B\|^{n+1}\to 0,
		\]
		то
		\[
		I+(-1)^nB^{n+1}\to I.
		\]
		Переходя к пределу, получаем
		\[
		(I+B)S=I.
		\]

		\item \textbf{Покажем, что \(S\) --- левый обратный к \(I+B\).}

		Аналогично,
		\[
		S_n(I+B)
		=
		I+(-1)^nB^{n+1}.
		\]
		Здесь можно переставлять \(B\) со степенями \(B^k\), потому что это степени одного и того же
		оператора.

		Переходя к пределу, получаем
		\[
		S(I+B)=I.
		\]

		\item \textbf{Завершаем доказательство.}

		Мы доказали
		\[
		(I+B)S=I
		\qquad
		\text{и}
		\qquad
		S(I+B)=I.
		\]
		Значит,
		\[
		S=(I+B)^{-1}.
		\]
		Следовательно,
		\[
		(I+B)^{-1}
		=
		\sum_{n=0}^{\infty}(-1)^nB^n
		\in\mathcal{L}(E).
		\]
	\end{enumerate}
\end{proof}


\begin{theorem}[Обратимость возмущенного оператора]
	Пусть \(E\) --- банахово пространство,
	\[
	A,\Delta A\in\mathcal{L}(E).
	\]
	Пусть оператор \(A\) обратим и
	\[
	A^{-1}\in\mathcal{L}(E).
	\]
	Если
	\[
	\|\Delta A\|<\frac{1}{\|A^{-1}\|},
	\]
	то оператор
	\[
	A+\Delta A
	\]
	обратим, причём
	\[
	(A+\Delta A)^{-1}\in\mathcal{L}(E).
	\]
\end{theorem}

\begin{proof}
	Докажем по шагам.

	\begin{enumerate}
		\item \textbf{Вынесем \(A\) за скобку.}

		Имеем
		\[
		A+\Delta A
		=
		A(I+A^{-1}\Delta A).
		\]
		Действительно,
		\[
		A(I+A^{-1}\Delta A)
		=
		A+AA^{-1}\Delta A
		=
		A+\Delta A.
		\]

		\item \textbf{Проверим малость оператора \(A^{-1}\Delta A\).}

		По условию
		\[
		\|\Delta A\|<\frac{1}{\|A^{-1}\|}.
		\]
		Тогда
		\[
		\|A^{-1}\Delta A\|
		\leqslant
		\|A^{-1}\|\cdot\|\Delta A\|
		<1.
		\]

		\item \textbf{Применим теорему о ряде Неймана.}

		К оператору
		\[
		B=A^{-1}\Delta A
		\]
		можно применить предыдущую теорему, так как
		\[
		\|B\|<1.
		\]
		Значит,
		\[
		I+A^{-1}\Delta A
		\]
		обратим, и
		\[
		(I+A^{-1}\Delta A)^{-1}\in\mathcal{L}(E).
		\]

		\item \textbf{Получаем обратимость \(A+\Delta A\).}

		Оператор \(A\) обратим по условию, а оператор
		\[
		I+A^{-1}\Delta A
		\]
		обратим по предыдущему пункту.

		Следовательно, их произведение
		\[
		A(I+A^{-1}\Delta A)=A+\Delta A
		\]
		тоже обратимо.

		\item \textbf{Запишем формулу для обратного оператора.}

		Из равенства
		\[
		A+\Delta A=A(I+A^{-1}\Delta A)
		\]
		получаем
		\[
		(A+\Delta A)^{-1}
		=
		(I+A^{-1}\Delta A)^{-1}A^{-1}.
		\]

		Поэтому
		\[
		(A+\Delta A)^{-1}\in\mathcal{L}(E).
		\]
	\end{enumerate}
\end{proof}

\begin{remark}[Оценки для обратного возмущенного оператора]
	В условиях предыдущей теоремы выполнены оценки:
	\[
	\|(A+\Delta A)^{-1}\|
	\leqslant
	\frac{\|A^{-1}\|}{1-\|A^{-1}\|\cdot\|\Delta A\|}
	\]
	и
	\[
	\|A^{-1}-(A+\Delta A)^{-1}\|
	\leqslant
	\frac{\|A^{-1}\|^2\cdot\|\Delta A\|}
	{1-\|A^{-1}\|\cdot\|\Delta A\|}.
	\]
\end{remark}
