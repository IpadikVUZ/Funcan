\subsection{Теорема о спектре самосопряженного оператора:
	\texorpdfstring{\(\sigma(A)\subset[m_-,m_+]\), \(r(A)=\|A\|\)}
	{sigma(A) subset [m-,m+], r(A)=||A||}}

\begin{definition}
	Пусть \(A\in\mathcal L(H)\) --- самосопряженный оператор. Положим
	\[
	m_-=\inf_{\|x\|=1}(Ax,x),
	\qquad
	m_+=\sup_{\|x\|=1}(Ax,x).
	\]
	Так как
	\[
	(Ax,x)\in\mathbb R,
	\qquad
	|(Ax,x)|\leqslant\|A\|
	\quad (\|x\|=1),
	\]
	числа \(m_-,m_+\) корректно определены.
\end{definition}

\begin{lemma}[Норма квадрата самосопряженного оператора]
	Если \(A=A^*\), то
	\[
	\|A^2\|=\|A\|^2.
	\]
\end{lemma}

\begin{proof}
	Оценка
	\[
	\|A^2\|\leqslant\|A\|^2
	\]
	очевидна.
	
	Для \(\|x\|=1\)
	\[
	\|Ax\|^2
	=
	(A^2x,x)
	\leqslant
	\|A^2\|.
	\]
	Беря супремум по \(\|x\|=1\), получаем
	\[
	\|A\|^2\leqslant\|A^2\|.
	\]
	Следовательно,
	\[
	\|A^2\|=\|A\|^2.
	\]
\end{proof}

\begin{remark}[Полускалярное произведение]
	Если \(B=B^*\) и
	\[
	(Bx,x)\geqslant0
	\qquad
	\forall x\in H,
	\]
	то
	\[
	\langle x,y\rangle_B=(Bx,y)
	\]
	задаёт полускалярное произведение, для которого
	\[
	|\langle x,y\rangle_B|^2
	\leqslant
	\langle x,x\rangle_B
	\langle y,y\rangle_B.
	\]
\end{remark}

\begin{theorem}[Спектр самосопряженного оператора]
	Пусть \(A\in\mathcal L(H)\) --- самосопряженный оператор. Тогда
	\[
	\sigma(A)\subset[m_-,m_+],
	\qquad
	m_-,m_+\in\sigma(A),
	\]
	и
	\[
	r(A)=\max\{|m_-|,|m_+|\}=\|A\|.
	\]
\end{theorem}

\begin{proof}
	\textbf{1. Докажем, что \(\sigma(A)\subset[m_-,m_+]\).}
	
	По предыдущему билету
	\[
	\sigma(A)\subset\mathbb R.
	\]
	Поэтому достаточно показать, что при
	\[
	\lambda>m_+
	\quad\text{или}\quad
	\lambda<m_-
	\]
	оператор \(A_\lambda=A-\lambda I\) ограничен снизу.
	
	Пусть \(\lambda>m_+\). Для \(x\neq0\)
	\[
	\begin{aligned}
		\|A_\lambda x\|\|x\|
		&\geqslant
		|(A_\lambda x,x)|
		\\
		&=
		\lambda\|x\|^2-(Ax,x)
		\\
		&\geqslant
		(\lambda-m_+)\|x\|^2.
	\end{aligned}
	\]
	Следовательно,
	\[
	\|A_\lambda x\|
	\geqslant
	(\lambda-m_+)\|x\|.
	\]
	По критерию предыдущего билета
	\[
	\lambda\in\rho(A).
	\]
	
	При \(\lambda<m_-\) аналогично
	\[
	\|A_\lambda x\|
	\geqslant
	(m_--\lambda)\|x\|,
	\]
	поэтому также
	\[
	\lambda\in\rho(A).
	\]
	
	Значит,
	\[
	\boxed{\sigma(A)\subset[m_-,m_+]}.
	\]
	
	\textbf{2. Докажем, что \(m_+,m_-\in\sigma(A)\).}
	
	Сначала покажем
	\[
	m_+\in\sigma(A).
	\]
	По определению супремума существует последовательность
	\[
	\|x_n\|=1,
	\qquad
	(Ax_n,x_n)\to m_+.
	\]
	
	По критерию принадлежности спектру хотим получить
	\[
	(A-m_+I)x_n\to0.
	\]
	Для перехода от сходимости скалярных произведений к сходимости по норме положим
	\[
	B=m_+I-A.
	\]
	Тогда \(B=B^*\) и
	\[
	(Bx,x)\geqslant0
	\qquad
	\forall x\in H.
	\]
	
	По неравенству Коши--Буняковского для полускалярного произведения
	\[
	\begin{aligned}
		\|Bx_n\|^4
		&=
		|\langle x_n,Bx_n\rangle_B|^2
		\\
		&\leqslant
		\langle x_n,x_n\rangle_B
		\langle Bx_n,Bx_n\rangle_B
		\\
		&\leqslant
		\bigl(m_+-(Ax_n,x_n)\bigr)\|B\|^3
		\to0.
	\end{aligned}
	\]
	Следовательно,
	\[
	Bx_n\to0,
	\]
	то есть
	\[
	(A-m_+I)x_n\to0.
	\]
	Так как \(\|x_n\|=1\), по критерию принадлежности спектру
	\[
	m_+\in\sigma(A).
	\]
	
	Аналогично, взяв
	\[
	(Ay_n,y_n)\to m_-,
	\qquad
	B=A-m_-I,
	\]
	получаем
	\[
	m_-\in\sigma(A).
	\]
	
	Таким образом,
	\[
	\sigma(A)\subset[m_-,m_+],
	\qquad
	m_-,m_+\in\sigma(A),
	\]
	поэтому
	\[
	r(A)=\max\{|m_-|,|m_+|\}.
	\]
	
	\textbf{3. Докажем, что \(r(A)=\|A\|\).}
	
	По лемме
	\[
	\|A^2\|=\|A\|^2.
	\]
	Так как \(A^{2^k}\) самосопряжен, последовательным применением леммы
	\[
	\|A^{2^k}\|=\|A\|^{2^k}.
	\]
	
	По формуле спектрального радиуса
	\[
	r(A)
	=
	\lim_{n\to\infty}\sqrt[n]{\|A^n\|}.
	\]
	По подпоследовательности \(n=2^k\)
	\[
	r(A)
	=
	\lim_{k\to\infty}
	\sqrt[2^k]{\|A^{2^k}\|}
	=
	\|A\|.
	\]
	
	Итак,
	\[
	\boxed{
		r(A)=\max\{|m_-|,|m_+|\}=\|A\|.
	}
	\]
	
\end{proof}
