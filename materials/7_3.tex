\subsection{Замкнутый шар в гильбертовом пространстве слабо секвенциально компактен (теорема Банаха)}

\begin{definition}[Слабая секвенциальная компактность]
	Пусть \(E\) --- нормированное пространство. Множество
	\[
	M\subset E
	\]
	называется \textbf{слабо секвенциально компактным}, если из любой последовательности
	\[
	\{x_n\}\subset M
	\]
	можно выделить подпоследовательность \(\{x_{n_k}\}\), слабо сходящуюся к некоторому
	элементу \(x\in M\):
	\[
	x_{n_k}\weakconv x.
	\]
\end{definition}

\begin{remark}[Слабая сходимость в гильбертовом пространстве]
	Пусть \(H\) --- гильбертово пространство. По теореме Рисса--Фреше каждый функционал
	\(f\in H^*\) имеет вид
	\[
	f(z)=(z,y)
	\]
	для некоторого \(y\in H\). Поэтому
	\[
	x_n\weakconv x
	\quad\Longleftrightarrow\quad
	\forall y\in H
	\qquad
	(x_n,y)\to(x,y).
	\]
\end{remark}

\begin{theorem}[Банаха]
	Пусть \(H\) --- гильбертово пространство и
	\[
	\overline{B}_R(0)=\{x\in H:\|x\|\leqslant R\}.
	\]
	Тогда замкнутый шар \(\overline{B}_R(0)\) слабо секвенциально компактен.
	
\end{theorem}

\begin{proof}
	Пусть
	\[
	x_n\in\overline{B}_R(0),
	\qquad
	\|x_n\|\leqslant R.
	\]
	Нужно выделить подпоследовательность, слабо сходящуюся в \(H\).
	
	\textbf{1. Сведение к подпространству, порождённому последовательностью.}
	
	Для слабой сходимости нужно получить сходимость скалярных произведений
	\[
	(x_{n_k},h)
	\qquad
	\forall h\in H.
	\]
	Сразу выбирать подпоследовательность для всех \(h\in H\) невозможно.
	Поэтому рассмотрим
	\[
	L=\overline{\operatorname{span}}\{x_1,x_2,\ldots\}.
	\]
	Все \(x_n\) лежат в \(L\), а
	\[
	L_0=\operatorname{span}\{x_1,x_2,\ldots\}
	\]
	плотно в \(L\).
	
	Сначала построим слабый предел в \(L\), а затем перейдём к \(H\).
	
	\textbf{2. Получим сходимость на плотном множестве \(L_0\).}
	
	Для каждого фиксированного \(m\)
	\[
	|(x_m,x_n)|
	\leqslant
	\|x_m\|\,\|x_n\|
	\leqslant
	R^2.
	\]
	Значит, последовательность чисел \(\{(x_m,x_n)\}\) ограничена, поэтому из неё можно выделить сходящуюся подпоследовательность.
	
	Последовательно делая это для \(m=1,2,\ldots\) и применяя диагональный метод, получаем подпоследовательность
	\[
	y_k=x_{n_k},
	\]
	для которой при каждом фиксированном \(m\)
	\[
	(x_m,y_k)
	\]
	сходится.в
	
	\textbf{3. Построим кандидат на слабый предел.}
	
	Для каждого \(z\in L_0\) последовательность \((z,y_k)\) сходится, поэтому корректно определён функционал
	\[
	F(z)=\lim_{k\to\infty}(z,y_k).
	\]
	Он линеен как предел линейных функционалов.
	
	Покажем, что \(F\) ограничен. Так как \(\|y_k\|\leqslant R\), то
	\[
	|F(z)|
	=
	\lim_{k\to\infty}|(z,y_k)|
	\leqslant
	R\|z\|.
	\]
	Значит,
	\[
	\|F\|\leqslant R.
	\]
	
	Так как \(L_0\) плотно в \(L\), ограниченный функционал \(F\) единственным образом
	продолжается на всё \(L\), то есть
	\[
	F\in L^*.
	\]
	
	Теперь по теореме Рисса--Фреше существует \(x\in L\), такой что
	\[
	F(z)=(z,x)
	\qquad
	\forall z\in L,
	\]
	причём
	\[
	\|x\|=\|F\|\leqslant R.
	\]
	Следовательно,
	\[
	x\in\overline{B}_R(0).
	\]
	
	\textbf{4. Получим слабую сходимость.}
	
	По построению для любого \(z\in L_0\)
	\[
	(z,y_k)\to(z,x).
	\]
	Кроме того,
	\[
	\|y_k\|\leqslant R,
	\]
	а \(L_0\) плотно в \(L\). Поэтому по критерию слабой сходимости
	\[
	y_k\weakconv x
	\qquad
	\text{в }L.
	\]
	
	Осталось перейти к \(H\). Для любого \(h\in H\)
	\[
	h=l+l^\perp,
	\qquad
	l\in L,\quad l^\perp\in L^\perp.
	\]
	Так как \(y_k,x\in L\),
	\[
	(y_k,h)=(y_k,l)\to(x,l)=(x,h).
	\]
	Следовательно,
	\[
	y_k\weakconv x
	\qquad
	\text{в }H.
	\]
	
	Таким образом, \(\overline{B}_R(0)\) слабо секвенциально компактен.
\end{proof}пун
