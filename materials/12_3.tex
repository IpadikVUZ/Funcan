\subsection{Теорема Фредгольма для компактных самосопряженных операторов}

\begin{definition}[Оператор \(A_\lambda\)]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{L}(H),
	\qquad
	\lambda\in\mathbb{C}.
	\]
	Обозначим
	\[
	A_\lambda=A-\lambda I.
	\]
\end{definition}

\begin{lemma}
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{K}(H)
	\]
	--- компактный самосопряженный оператор, и
	\[
	\lambda\neq0.
	\]
	Если
	\[
	\lambda\in\sigma(A),
	\]
	то \(\lambda\) является собственным значением оператора \(A\).
\end{lemma}

\begin{proof}
	По критерию принадлежности числа спектру существует последовательность \(\{x_n\}\subset H\),
	такая что
	\[
	\|x_n\|=1
	\qquad
	\forall n
	\]
	и
	\[
	A_\lambda x_n\to0.
	\]
	То есть
	\[
	Ax_n-\lambda x_n\to0.
	\]
	
	Так как \(A\) компактен, последовательность \(\{Ax_n\}\) имеет сходящуюся подпоследовательность:
	\[
	Ax_{n_k}\to y.
	\]
	Из условия
	\[
	Ax_{n_k}-\lambda x_{n_k}\to0
	\]
	получаем
	\[
	\lambda x_{n_k}\to y.
	\]
	Так как \(\lambda\neq0\), то
	\[
	x_{n_k}\to \frac{1}{\lambda}y.
	\]
	Обозначим
	\[
	x=\frac{1}{\lambda}y.
	\]
	Тогда \(\|x\|=1\), поэтому \(x\neq0\).
	
	Переходя к пределу в равенстве
	\[
	Ax_{n_k}-\lambda x_{n_k}\to0,
	\]
	получаем
	\[
	Ax-\lambda x=0.
	\]
	Значит,
	\[
	Ax=\lambda x,
	\]
	где \(x\neq0\). Следовательно, \(\lambda\) --- собственное значение оператора \(A\).
\end{proof}

\begin{lemma}
	Пусть \(A\) --- самосопряженный оператор в \(H\), а \(M\subset H\) --- подпространство,
	инвариантное относительно \(A\):
	\[
	A(M)\subset M.
	\]
	Тогда \(M^\perp\) тоже инвариантно относительно \(A\):
	\[
	A(M^\perp)\subset M^\perp.
	\]
\end{lemma}

\begin{proof}
	Пусть
	\[
	y\in M^\perp.
	\]
	Возьмём произвольный
	\[
	x\in M.
	\]
	Так как \(M\) инвариантно относительно \(A\), имеем
	\[
	Ax\in M.
	\]
	Поэтому
	\[
	(Ax,y)=0.
	\]
	Так как \(A\) самосопряжен,
	\[
	(x,Ay)=(Ax,y)=0.
	\]
	Это верно для любого \(x\in M\). Значит,
	\[
	Ay\in M^\perp.
	\]
	Следовательно,
	\[
	M^\perp
	\]
	инвариантно относительно \(A\).
\end{proof}

\begin{lemma}
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{K}(H)
	\]
	--- компактный самосопряженный оператор, и
	\[
	\lambda\neq0.
	\]
	Тогда
	\[
	\operatorname{Im}A_\lambda
	\]
	замкнут, то есть
	\[
	\overline{\operatorname{Im}A_\lambda}
	=
	\operatorname{Im}A_\lambda.
	\]
\end{lemma}

\begin{proof}
	Докажем, что замыкание образа на самом деле совпадает с образом.
	
	\textbf{1. Выделим нужное инвариантное подпространство.}
	
	По разложению из билета 11.2 имеем
	\[
	H=
	\overline{\operatorname{Im}A_\lambda}
	\oplus_{\perp}
	\ker A_\lambda.
	\]
	Значит,
	\[
	\overline{\operatorname{Im}A_\lambda}
	=
	(\ker A_\lambda)^\perp.
	\]
	
	Обозначим
	\[
	M=\ker A_\lambda.
	\]
	Тогда
	\[
	\overline{\operatorname{Im}A_\lambda}=M^\perp.
	\]
	
	Подпространство \(M\) инвариантно относительно \(A\). Действительно, если
	\[
	x\in M,
	\]
	то
	\[
	Ax=\lambda x,
	\]
	поэтому
	\[
	Ax\in M.
	\]
	По предыдущей лемме \(M^\perp\) тоже инвариантно относительно \(A\).
	
	\textbf{2. Ограничим оператор на \(M^\perp\).}
	
	Рассмотрим сужение
	\[
	\widehat A=A|_{M^\perp}:M^\perp\to M^\perp.
	\]
	Оператор \(\widehat A\) компактен и самосопряжен.
	
	Покажем, что \(\lambda\) не является собственным значением \(\widehat A\). Если бы существовал
	ненулевой \(x\in M^\perp\), такой что
	\[
	\widehat A x=\lambda x,
	\]
	то
	\[
	Ax=\lambda x.
	\]
	Значит,
	\[
	x\in\ker A_\lambda=M.
	\]
	Но одновременно
	\[
	x\in M^\perp.
	\]
	Следовательно,
	\[
	x\in M\cap M^\perp=\{0\},
	\]
	противоречие.
	
	Значит, \(\lambda\) не является собственным значением \(\widehat A\). По первой лемме, так как
	\[
	\lambda\neq0,
	\]
	получаем
	\[
	\lambda\notin\sigma(\widehat A).
	\]
	Следовательно,
	\[
	\lambda\in\rho(\widehat A).
	\]
	
	\textbf{3. Получим равенство образа и его замыкания.}
	
	Из
	\[
	\lambda\in\rho(\widehat A)
	\]
	следует, что оператор
	\[
	\widehat A-\lambda I
	\]
	отображает \(M^\perp\) на всё \(M^\perp\). То есть
	\[
	\operatorname{Im}(\widehat A-\lambda I)=M^\perp.
	\]
	Но \(\widehat A-\lambda I\) --- это сужение \(A_\lambda\) на \(M^\perp\). Поэтому
	\[
	M^\perp\subset\operatorname{Im}A_\lambda.
	\]
	С другой стороны,
	\[
	\operatorname{Im}A_\lambda
	\subset
	\overline{\operatorname{Im}A_\lambda}
	=
	M^\perp.
	\]
	Значит,
	\[
	\operatorname{Im}A_\lambda=M^\perp.
	\]
	Так как \(M^\perp\) замкнуто, получаем
	\[
	\operatorname{Im}A_\lambda
	\text{ замкнут}.
	\]
\end{proof}

\begin{theorem}[Фредгольма]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{K}(H)
	\]
	--- компактный самосопряженный оператор, и
	\[
	\lambda\neq0.
	\]
	Тогда
	\[
	H=
	\operatorname{Im}A_\lambda
	\oplus_{\perp}
	\ker A_\lambda.
	\]
\end{theorem}

\begin{proof}
	По разложению из билета 11.2 имеем
	\[
	H=
	\overline{\operatorname{Im}A_\lambda}
	\oplus_{\perp}
	\ker A_\lambda.
	\]
	По предыдущей лемме
	\[
	\overline{\operatorname{Im}A_\lambda}
	=
	\operatorname{Im}A_\lambda.
	\]
	Следовательно,
	\[
	H=
	\operatorname{Im}A_\lambda
	\oplus_{\perp}
	\ker A_\lambda.
	\]
\end{proof}
