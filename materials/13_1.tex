\subsection{Производная Фреше, производная Гато. Формула конечных приращений.}

\begin{definition}[Производная Фреше]
	Пусть \(E_1,E_2\) --- вещественные банаховы пространства,
	\(D\subset E_1\) --- область,
	\[
	F:D\to E_2.
	\]
	Функция \(F\) называется \textbf{дифференцируемой по Фреше} в точке
	\(x_0\in D\), если существует оператор
	\[
	A\in\mathcal L(E_1,E_2),
	\]
	такой, что
	\[
	F(x_0+h)-F(x_0)=Ah+r(h),
	\qquad
	\frac{\|r(h)\|}{\|h\|}\xrightarrow[h\to0]{}0.
	\]
	Оператор \(A\) называется \textbf{производной Фреше} функции \(F\) в точке \(x_0\):
	\[
	A=F'(x_0).
	\]
\end{definition}

То есть
\[
F(x_0+h)-F(x_0)
=
F'(x_0)h+o(\|h\|).
\]

\begin{definition}[Дифференциал]
	Дифференциал функции \(F\) в точке \(x_0\) на приращении \(h\) есть
	\[
	dF(x_0,h)=F'(x_0)h.
	\]
\end{definition}

\begin{definition}[Производная Гато]
	Пусть \(h\in E_1\).
	Производной Гато функции \(F\) в точке \(x_0\) по направлению \(h\)
	называется предел
	\[
	D_GF(x_0;h)
	=
	\left.\frac{d}{dt}F(x_0+th)\right|_{t=0}
	=
	\lim_{t\to0}
	\frac{F(x_0+th)-F(x_0)}{t},
	\]
	если этот предел существует.
	
	Если существует оператор
	\[
	A\in\mathcal L(E_1,E_2)
	\]
	такой, что для любого \(h\in E_1\)
	\[
	D_GF(x_0;h)=Ah,
	\]
	то \(A\) называется производной Гато:
	\[
	A=F'_G(x_0).
	\]
\end{definition}

\begin{theorem}[Связь производных Фреше и Гато]
	Если \(F\) дифференцируема по Фреше в точке \(x_0\), то она
	дифференцируема по Гато в этой точке и
	\[
	F'_G(x_0)=F'(x_0).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать}, что производная по любому направлению \(h\) существует
	и равна \(F'(x_0)h\).
	
	Из дифференцируемости по Фреше
	\[
	F(x_0+u)-F(x_0)=F'(x_0)u+o(\|u\|).
	\]
	
	\textbf{Для движения в направлении \(h\) положим} \(u=th\). Тогда
	\[
	F(x_0+th)-F(x_0)
	=
	tF'(x_0)h+o(|t|).
	\]
	
	Делим на \(t\) и переходим к пределу:
	\[
	\lim_{t\to0}
	\frac{F(x_0+th)-F(x_0)}{t}
	=
	F'(x_0)h.
	\]
	
	Следовательно,
	\[
	F'_G(x_0)=F'(x_0).
	\]
\end{proof}

\begin{theorem}[Формула конечных приращений]
	Пусть \(E_1,E_2\) --- вещественные банаховы пространства,
	\(D\subset E_1\) --- выпуклая область,
	\[
	F:D\to E_2
	\]
	дифференцируема по Фреше.
	
	Тогда для любых \(x_0,x_1\in D\)
	\[
	\boxed{
		\|F(x_1)-F(x_0)\|
		\le
		\sup_{x\in[x_0,x_1]}\|F'(x)\|
		\,\|x_1-x_0\|
	}.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим свести задачу к обычной формуле Лагранжа.}
	Для этого возьмём \(f\in E_2^*\) и параметризуем отрезок
	\[
	x(t)=x_0+t(x_1-x_0),\qquad t\in[0,1].
	\]
	Рассмотрим вещественную функцию
	\[
	\varphi(t)=f(F(x(t))).
	\]
	
	По правилу дифференцирования сложной функции
	\[
	\varphi'(t)
	=
	f\bigl(F'(x(t))(x_1-x_0)\bigr).
	\]
	
	По формуле Лагранжа для некоторого \(\xi\in(0,1)\)
	\[
	f(F(x_1)-F(x_0))
	=
	f\bigl(F'(x(\xi))(x_1-x_0)\bigr).
	\]
	Поэтому
	\[
	|f(F(x_1)-F(x_0))|
	\le
	\|f\|
	\sup_{x\in[x_0,x_1]}\|F'(x)\|
	\|x_1-x_0\|.
	\]
	
	Берём супремум по всем \(\|f\|\le1\). По теореме Хана--Банаха
	\[
	\|y\|=\sup_{\|f\|\le1}|f(y)|.
	\]
	Следовательно,
	\[
	\boxed{
		\|F(x_1)-F(x_0)\|
		\le
		\sup_{x\in[x_0,x_1]}\|F'(x)\|
		\|x_1-x_0\|
	}.
	\]
\end{proof}

\begin{theorem}[Производная сложной функции]
	Пусть
	\[
	F:E_1\to E_2,\qquad G:E_2\to E_3,
	\]
	\(F\) дифференцируема по Фреше в точке \(x_0\), а \(G\) ---
	в точке
	\[
	y_0=F(x_0).
	\]
	Тогда \(G\circ F\) дифференцируема по Фреше в точке \(x_0\), причём
	\[
	\boxed{
		(G\circ F)'(x_0)=G'(F(x_0))\circ F'(x_0)
	}.
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим} представить приращение \(G\circ F\) как линейную часть
	плюс \(o(\|\Delta x\|)\).
	
	Из дифференцируемости \(F\)
	\[
	\Delta y
	=
	F'(x_0)\Delta x+o(\|\Delta x\|).
	\]
	
	Из дифференцируемости \(G\)
	\[
	\Delta z
	=
	G'(y_0)\Delta y+o(\|\Delta y\|).
	\]
	
	Подставляем первое во второе:
	\[
	\Delta z
	=
	G'(y_0)F'(x_0)\Delta x
	+
	G'(y_0)o(\|\Delta x\|)
	+
	o(\|\Delta y\|).
	\]
	
	Первый остаток есть \(o(\|\Delta x\|)\), так как \(G'(y_0)\) ограничен.
	
	Кроме того,
	\[
	\|\Delta y\|
	\le
	\|F'(x_0)\|\|\Delta x\|+o(\|\Delta x\|)
	=
	O(\|\Delta x\|),
	\]
	поэтому
	\[
	o(\|\Delta y\|)=o(\|\Delta x\|).
	\]
	
	Значит,
	\[
	\Delta z
	=
	G'(y_0)F'(x_0)\Delta x+o(\|\Delta x\|),
	\]
	откуда
	\[
	(G\circ F)'(x_0)
	=
	G'(F(x_0))\circ F'(x_0).
	\]
\end{proof}
