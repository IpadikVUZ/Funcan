\subsection{Свойства компактных операторов}

\begin{definition}[Предкомпактное множество]
	Множество
	\[
	M\subset E
	\]
	называется предкомпактным, если его замыкание
	\[
	\overline M
	\]
	компактно.
\end{definition}

\begin{definition}[Компактный оператор]
	Пусть
	\[
	A\in\mathcal L(E_1,E_2).
	\]
	Оператор \(A\) называется компактным, если образ любого ограниченного
	множества является предкомпактным:
	\[
	M\subset E_1 \text{ ограничено}
	\quad\Longrightarrow\quad
	A(M)\subset E_2 \text{ предкомпактно}.
	\]
	
	Множество компактных операторов обозначается
	\[
	\mathcal K(E_1,E_2).
	\]
\end{definition}

\begin{remark}[Критерии компактности]
	Для линейного ограниченного оператора достаточно проверять единичный шар:
	\[
	A\in\mathcal K(E_1,E_2)
	\iff
	A(\overline B_1(0))
	\text{ предкомпактно}.
	\]
	
	Эквивалентно:
	для любой ограниченной последовательности
	\[
	\{x_n\}\subset E_1
	\]
	последовательность
	\[
	\{Ax_n\}
	\]
	имеет сходящуюся подпоследовательность.
\end{remark}

\begin{theorem}[Конечномерный случай]
	Пусть
	\[
	A\in\mathcal L(E_1,E_2).
	\]
	Если
	\[
	\dim E_1<\infty
	\qquad\text{или}\qquad
	\dim E_2<\infty,
	\]
	то
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать, что образ ограниченного множества предкомпактен.}
	
	Если
	\[
	\dim E_1<\infty,
	\]
	то
	\[
	\dim\operatorname{Im}A\leq\dim E_1<\infty.
	\]
	Образ ограниченного множества ограничен в конечномерном пространстве
	\(\operatorname{Im}A\), следовательно, предкомпактен.
	
	Если
	\[
	\dim E_2<\infty,
	\]
	то \(A(M)\) ограничено в конечномерном \(E_2\), следовательно,
	также предкомпактно.
\end{proof}

\begin{theorem}
	Если \(E\) бесконечномерно, то тождественный оператор не компактен:
	\[
	I\notin\mathcal K(E).
	\]
\end{theorem}

\begin{proof}
	\textbf{Предположим противное.}
	
	Если \(I\) компактен, то
	\[
	I(\overline B_1(0))
	=
	\overline B_1(0)
	\]
	предкомпактен.
	
	Но в бесконечномерном нормированном пространстве единичный шар
	не предкомпактен.
	
	Противоречие.
\end{proof}

\begin{theorem}[Идеальное свойство]
	Пусть
	\[
	A\in\mathcal K(E),
	\qquad
	B\in\mathcal L(E).
	\]
	Тогда
	\[
	AB,BA\in\mathcal K(E).
	\]
\end{theorem}

\begin{proof}
	Пусть \(M\subset E\) ограничено.
	
	\begin{enumerate}
		\item
		\textbf{Хотим доказать компактность \(AB\).}
		
		Так как \(B\) ограничен,
		\[
		B(M)
		\]
		ограничено.
		
		Так как \(A\) компактен,
		\[
		A(B(M))
		\]
		предкомпактно.
		
		Следовательно,
		\[
		AB\in\mathcal K(E).
		\]
		
		\item
		\textbf{Хотим доказать компактность \(BA\).}
		
		Так как \(A\) компактен,
		\[
		\overline{A(M)}
		\]
		компактно.
		
		Оператор \(B\) непрерывен, поэтому
		\[
		B(\overline{A(M)})
		\]
		компактно.
		
		Так как
		\[
		B(A(M))
		\subset
		B(\overline{A(M)}),
		\]
		множество \(B(A(M))\) предкомпактно.
		
		Следовательно,
		\[
		BA\in\mathcal K(E).
		\]
	\end{enumerate}
\end{proof}

\begin{corollary}
	Если \(E\) бесконечномерно и
	\[
	A\in\mathcal K(E),
	\]
	то \(A^{-1}\), если существует, не может быть ограниченным.
\end{corollary}

\begin{proof}
	Если бы
	\[
	A^{-1}\in\mathcal L(E),
	\]
	то по идеальному свойству
	\[
	I=A^{-1}A
	\]
	был бы компактным.
	
	Но в бесконечномерном пространстве
	\[
	I\notin\mathcal K(E).
	\]
\end{proof}

\begin{theorem}[Предел компактных операторов]
	Пусть \(E_2\) --- банахово пространство,
	\[
	A_n\in\mathcal K(E_1,E_2)
	\]
	и
	\[
	\|A_n-A\|\to0.
	\]
	Тогда
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{theorem}

\begin{proof}
	\textbf{Хотим показать, что \(A(\overline B_1(0))\) предкомпактно.}
	
	Так как \(E_2\) банахово, достаточно построить для него конечную
	\(\varepsilon\)-сеть.
	
	Пусть
	\[
	\varepsilon>0.
	\]
	
	\textbf{1. Хотим заменить \(A\) близким компактным оператором.}
	
	Выбираем \(N\), такое что
	\[
	\|A_N-A\|<\frac{\varepsilon}{2}.
	\]
	
	Тогда для
	\[
	\|x\|\leq1
	\]
	имеем
	\[
	\|Ax-A_Nx\|
	\leq
	\|A-A_N\|\|x\|
	<
	\frac{\varepsilon}{2}.
	\]
	
	\textbf{2. Используем компактность \(A_N\).}
	
	Так как \(A_N\) компактен, для множества
	\[
	A_N(\overline B_1(0))
	\]
	существует конечная \(\varepsilon/2\)-сеть
	\[
	y_1,\ldots,y_m.
	\]
	
	Поэтому для любого \(\|x\|\leq1\) найдется \(y_j\), такое что
	\[
	\|A_Nx-y_j\|<\frac{\varepsilon}{2}.
	\]
	
	Тогда
	\[
	\|Ax-y_j\|
	\leq
	\|Ax-A_Nx\|+\|A_Nx-y_j\|
	<
	\varepsilon.
	\]
	
	Значит,
	\[
	y_1,\ldots,y_m
	\]
	--- конечная \(\varepsilon\)-сеть для \(A(\overline B_1(0))\).
	
	Следовательно,
	\[
	A\in\mathcal K(E_1,E_2).
	\]
\end{proof}

\begin{theorem}[Компактный оператор улучшает слабую сходимость]
	Пусть
	\[
	A\in\mathcal K(E_1,E_2)
	\]
	и
	\[
	x_n\rightharpoonup x.
	\]
	Тогда
	\[
	\|Ax_n-Ax\|\to0.
	\]
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим использовать компактность \(A\).}
		
		Слабо сходящаяся последовательность ограничена, поэтому
		\[
		\{x_n\}\cup\{x\}
		\]
		ограничена.
		
		Так как \(A\) компактен,
		\[
		\{Ax_n\}\cup\{Ax\}
		\]
		предкомпактна.
		
		\item
		\textbf{Хотим определить возможный предел образов.}
		
		Линейный ограниченный оператор сохраняет слабую сходимость:
		\[
		x_n\rightharpoonup x
		\quad\Longrightarrow\quad
		Ax_n\rightharpoonup Ax.
		\]
		
		\item
		\textbf{Хотим получить сходимость по норме.}
		
		Предположим, что
		\[
		Ax_n\not\to Ax
		\]
		по норме.
		
		Тогда существуют \(\varepsilon>0\) и подпоследовательность
		\[
		Ax_{n_k},
		\]
		для которой
		\[
		\|Ax_{n_k}-Ax\|\geq\varepsilon.
		\]
		
		Из предкомпактности выделяем еще одну подпоследовательность
		\[
		Ax_{n_{k_j}}\to y
		\]
		по норме.
		
		Тогда
		\[
		Ax_{n_{k_j}}\rightharpoonup y.
		\]
		
		Но одновременно
		\[
		Ax_{n_{k_j}}\rightharpoonup Ax.
		\]
		
		Слабый предел единственен, поэтому
		\[
		y=Ax.
		\]
		
		Получили
		\[
		Ax_{n_{k_j}}\to Ax
		\]
		по норме, что противоречит
		\[
		\|Ax_{n_k}-Ax\|\geq\varepsilon.
		\]
		
		Следовательно,
		\[
		\boxed{\|Ax_n-Ax\|\to0.}
		\]
	\end{enumerate}
\end{proof}
