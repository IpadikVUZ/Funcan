\subsection{Операторы Гильберта--Шмидта. Компактность}

\begin{definition}[Оператор Гильберта--Шмидта]
	Оператор
	\[
	A:L_2[a,b]\to L_2[a,b]
	\]
	называется оператором Гильберта--Шмидта, если
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt,
	\]
	где
	\[
	K\in L_2([a,b]^2).
	\]
	Функция \(K(x,t)\) называется ядром оператора.
\end{definition}

\begin{remark}
	Для любого \(f\in L_2[a,b]\)
	\[
	\|Af\|_{L_2}
	\leq
	\|K\|_{L_2([a,b]^2)}\|f\|_{L_2}.
	\]
	В частности,
	\[
	\|A\|
	\leq
	\|K\|_{L_2([a,b]^2)}.
	\]
\end{remark}

\begin{proof}
	\textbf{Хотим оценить \(Af\).}
	
	По неравенству Коши--Буняковского для почти всех \(x\)
	\[
	\begin{aligned}
		|Af(x)|^2
		&=
		\left|
		\int_a^b K(x,t)f(t)\,dt
		\right|^2
		\\
		&\leq
		\left(
		\int_a^b|K(x,t)|^2\,dt
		\right)
		\|f\|_{L_2}^2.
	\end{aligned}
	\]
	
	Интегрируя по \(x\), получаем
	\[
	\begin{aligned}
		\|Af\|_{L_2}^2
		&\leq
		\int_a^b\int_a^b
		|K(x,t)|^2\,dt\,dx\,
		\|f\|_{L_2}^2
		\\
		&=
		\|K\|_{L_2([a,b]^2)}^2
		\|f\|_{L_2}^2.
	\end{aligned}
	\]
\end{proof}

\begin{theorem}
	Пусть \(\{\varphi_n\}\) --- ОНБ в \(L_2[a,b]\).
	Тогда
	\[
	\{\varphi_i(x)\varphi_j(t)\}_{i,j=1}^{\infty}
	\]
	--- ОНБ в \(L_2([a,b]^2)\).
\end{theorem}

\begin{remark}
	Этот факт используем без доказательства.
\end{remark}

\begin{theorem}[Компактность оператора Гильберта--Шмидта]
	Каждый оператор Гильберта--Шмидта
	\[
	A:L_2[a,b]\to L_2[a,b]
	\]
	компактен.
\end{theorem}

\begin{proof}
	\begin{enumerate}
		\item
		\textbf{Хотим приблизить ядро \(K\) конечными суммами.}
		
		Так как
		\[
		K\in L_2([a,b]^2)
		\]
		и
		\[
		\{\varphi_i(x)\varphi_j(t)\}
		\]
		--- ОНБ, раскладываем \(K\) в ряд Фурье:
		\[
		K(x,t)
		=
		\sum_{i,j=1}^{\infty}
		a_{ij}\varphi_i(x)\varphi_j(t)
		\]
		со сходимостью в \(L_2([a,b]^2)\).
		
		Берем частичные суммы
		\[
		K_N(x,t)
		=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)\varphi_j(t).
		\]
		
		Тогда
		\[
		\|K_N-K\|_{L_2([a,b]^2)}
		\to0.
		\]
		
		\item
		\textbf{Хотим построить конечномерные операторы, приближающие \(A\).}
		
		Ядру \(K_N\) сопоставим оператор
		\[
		(A_Nf)(x)
		=
		\int_a^b K_N(x,t)f(t)\,dt.
		\]
		
		Раскрывая \(K_N\), получаем
		\[
		(A_Nf)(x)
		=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
		\]
		
		Следовательно,
		\[
		A_Nf
		\in
		\operatorname{span}
		\{\varphi_1,\ldots,\varphi_N\}.
		\]
		
		Значит,
		\[
		\dim\operatorname{Im}A_N\leq N,
		\]
		то есть \(A_N\) --- конечномерный, а значит компактный оператор.
		
		\item
		\textbf{Хотим показать, что \(A_N\to A\) по операторной норме.}
		
		Имеем
		\[
		((A_N-A)f)(x)
		=
		\int_a^b
		(K_N(x,t)-K(x,t))f(t)\,dt.
		\]
		
		По неравенству Коши--Буняковского
		\[
		|((A_N-A)f)(x)|^2
		\leq
		\left(
		\int_a^b|K_N(x,t)-K(x,t)|^2\,dt
		\right)
		\|f\|_{L_2}^2.
		\]
		
		Интегрируем по \(x\):
		\[
		\|(A_N-A)f\|_{L_2}^2
		\leq
		\|K_N-K\|_{L_2([a,b]^2)}^2
		\|f\|_{L_2}^2.
		\]
		
		Следовательно,
		\[
		\boxed{
			\|A_N-A\|
			\leq
			\|K_N-K\|_{L_2([a,b]^2)}
			\to0.
		}
		\]
		
		\item
		\textbf{Завершаем доказательство.}
		
		Каждый \(A_N\) компактен и
		\[
		A_N\to A
		\]
		по операторной норме.
		
		Предел компактных операторов по операторной норме компактен.
		
		Следовательно,
		\[
		\boxed{A\text{ компактен}.}
		\]
	\end{enumerate}
\end{proof}

\begin{idea}
	\textbf{Что вообще происходит в этом билете и почему оператор Гильберта--Шмидта компактен.}
	
	Пусть
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt,
	\qquad
	K\in L_2([a,b]^2).
	\]
	
	Здесь оператор \(A\) берет функцию \(f(t)\) и по ней строит новую функцию
	\(Af(x)\).
	
	При каждом фиксированном \(x\) мы берем функцию
	\[
	t\mapsto K(x,t)
	\]
	и вычисляем ее скалярное произведение с \(f(t)\):
	\[
	(Af)(x)
	=
	\int_a^b K(x,t)f(t)\,dt.
	\]
	
	То есть ядро \(K(x,t)\) полностью определяет действие оператора.
	
	\medskip
	
	\textbf{1. Чего мы хотим добиться?}
	
	Хотим доказать:
	\[
	\boxed{A\text{ компактен}.}
	\]
	
	Удобный известный критерий:
	
	если существуют конечномерные операторы \(A_N\), для которых
	\[
	\|A_N-A\|\to0,
	\]
	то \(A\) компактен.
	
	Действительно, конечномерные операторы компактны, а предел компактных
	операторов по операторной норме компактен.
	
	Поэтому вся наша задача сводится к следующей:
	\[
	\boxed{
		\text{построить конечномерные }A_N
		\text{ так, чтобы }
		A_N\to A.
	}
	\]
	
	\medskip
	
	\textbf{2. Но как приблизить сам оператор \(A\)?}
	
	Оператор \(A\) задается ядром \(K\).
	
	Поэтому естественная идея:
	\[
	\boxed{
		\text{сначала приблизить ядро }K
		\text{ простыми ядрами }K_N.
	}
	\]
	
	Если
	\[
	K_N\approx K,
	\]
	то надеемся, что соответствующие операторы
	\[
	(A_Nf)(x)
	=
	\int_a^b K_N(x,t)f(t)\,dt
	\]
	тоже будут удовлетворять
	\[
	A_N\approx A.
	\]
	
	Останется подобрать \(K_N\) так, чтобы операторы \(A_N\)
	были конечномерными.
	
	\medskip
	
	\textbf{3. Почему здесь появляется ОНБ?}
	
	Возьмем ортонормированный базис
	\[
	\{\varphi_n\}
	\]
	в пространстве \(L_2[a,b]\).
	
	Есть известный факт:
	\[
	\boxed{
		\{\varphi_i(x)\varphi_j(t)\}_{i,j=1}^{\infty}
		\text{ --- ОНБ в }L_2([a,b]^2).
	}
	\]
	
	Почему это нам полезно?
	
	Потому что наше ядро
	\[
	K=K(x,t)
	\]
	--- функция двух переменных и
	\[
	K\in L_2([a,b]^2).
	\]
	
	Значит, как любой элемент гильбертова пространства с ОНБ,
	его можно разложить в ряд:
	\[
	K(x,t)
	=
	\sum_{i,j=1}^{\infty}
	a_{ij}\varphi_i(x)\varphi_j(t).
	\]
	
	Причем ряд сходится именно в норме \(L_2([a,b]^2)\).
	
	Берем конечную часть:
	\[
	K_N(x,t)
	=
	\sum_{i,j=1}^{N}
	a_{ij}\varphi_i(x)\varphi_j(t).
	\]
	
	Тогда
	\[
	\boxed{
		\|K_N-K\|_{L_2([a,b]^2)}
		\to0.
	}
	\]
	
	То есть сложное ядро \(K\) приблизили конечной суммой очень простых
	функций вида
	\[
	\varphi_i(x)\varphi_j(t).
	\]
	
	\medskip
	
	\textbf{4. Почему именно такие простые ядра дают конечномерный оператор?}
	
	Рассмотрим сначала вообще одно ядро
	\[
	K(x,t)=\varphi_i(x)\varphi_j(t).
	\]
	
	Тогда
	\[
	\begin{aligned}
		(Af)(x)
		&=
		\int_a^b
		\varphi_i(x)\varphi_j(t)f(t)\,dt
		\\
		&=
		\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
	\end{aligned}
	\]
	
	Интеграл
	\[
	\int_a^b\varphi_j(t)f(t)\,dt
	\]
	--- просто некоторое число, зависящее от \(f\).
	
	Значит,
	\[
	Af=c\varphi_i.
	\]
	
	То есть независимо от того, какую функцию \(f\) мы подали на вход,
	результат всегда лежит в одномерном пространстве
	\[
	\operatorname{span}\{\varphi_i\}.
	\]
	
	Именно поэтому ядра вида
	\[
	\varphi_i(x)\varphi_j(t)
	\]
	так удобны.
	
	\medskip
	
	Теперь для
	\[
	K_N(x,t)
	=
	\sum_{i,j=1}^{N}
	a_{ij}\varphi_i(x)\varphi_j(t)
	\]
	получаем
	\[
	\begin{aligned}
		(A_Nf)(x)
		&=
		\int_a^bK_N(x,t)f(t)\,dt
		\\
		&=
		\sum_{i,j=1}^{N}
		a_{ij}\varphi_i(x)
		\int_a^b\varphi_j(t)f(t)\,dt.
	\end{aligned}
	\]
	
	Все интегралы здесь --- просто числа.
	
	Поэтому результат имеет вид
	\[
	A_Nf
	=
	c_1\varphi_1+\ldots+c_N\varphi_N.
	\]
	
	Следовательно,
	\[
	\boxed{
		\operatorname{Im}A_N
		\subset
		\operatorname{span}\{\varphi_1,\ldots,\varphi_N\}.
	}
	\]
	
	Значит,
	\[
	\dim\operatorname{Im}A_N\leq N.
	\]
	
	То есть \(A_N\) --- конечномерный оператор, а значит компактный.
	
	\medskip
	
	\textbf{5. Теперь главное: почему из \(K_N\to K\) следует \(A_N\to A\)?}
	
	Хотим оценить разность операторов.
	
	Для \(f\in L_2[a,b]\)
	\[
	((A_N-A)f)(x)
	=
	\int_a^b
	(K_N(x,t)-K(x,t))f(t)\,dt.
	\]
	
	При фиксированном \(x\) это скалярное произведение двух функций
	переменной \(t\):
	\[
	K_N(x,\cdot)-K(x,\cdot)
	\qquad\text{и}\qquad
	f.
	\]
	
	По неравенству Коши--Буняковского
	\[
	\left|
	((A_N-A)f)(x)
	\right|^2
	\leq
	\left(
	\int_a^b
	|K_N(x,t)-K(x,t)|^2\,dt
	\right)
	\|f\|_{L_2}^2.
	\]
	
	Теперь хотим получить норму всей функции \((A_N-A)f\),
	поэтому интегрируем по \(x\):
	\[
	\begin{aligned}
		\|(A_N-A)f\|_{L_2}^2
		&=
		\int_a^b
		|((A_N-A)f)(x)|^2\,dx
		\\
		&\leq
		\int_a^b
		\left(
		\int_a^b
		|K_N(x,t)-K(x,t)|^2\,dt
		\right)dx
		\,
		\|f\|_{L_2}^2.
	\end{aligned}
	\]
	
	Но двойной интеграл --- это ровно квадрат нормы ядра:
	\[
	\int_a^b\int_a^b
	|K_N(x,t)-K(x,t)|^2\,dt\,dx
	=
	\|K_N-K\|_{L_2([a,b]^2)}^2.
	\]
	
	Поэтому
	\[
	\|(A_N-A)f\|_{L_2}
	\leq
	\|K_N-K\|_{L_2([a,b]^2)}
	\|f\|_{L_2}.
	\]
	
	Делим на \(\|f\|_{L_2}\) и берем супремум по \(f\neq0\):
	\[
	\boxed{
		\|A_N-A\|
		\leq
		\|K_N-K\|_{L_2([a,b]^2)}.
	}
	\]
	
	Но мы уже знаем, что
	\[
	\|K_N-K\|_{L_2([a,b]^2)}\to0.
	\]
	
	Следовательно,
	\[
	\boxed{
		\|A_N-A\|\to0.
	}
	\]
	
	\medskip
	
	\textbf{6. Теперь всё доказательство складывается в одну цепочку.}
	
	Имеем
	\[
	K\in L_2([a,b]^2).
	\]
	
	Разлагаем ядро по ОНБ и берем частичные суммы:
	\[
	K_N\to K
	\quad\text{в }L_2([a,b]^2).
	\]
	
	Каждому \(K_N\) соответствует оператор
	\[
	A_N,
	\]
	причем
	\[
	\operatorname{Im}A_N
	\subset
	\operatorname{span}\{\varphi_1,\ldots,\varphi_N\}.
	\]
	
	Поэтому
	\[
	A_N
	\text{ конечномерный}
	\quad\Longrightarrow\quad
	A_N
	\text{ компактный}.
	\]
	
	Кроме того,
	\[
	\|A_N-A\|
	\leq
	\|K_N-K\|_2
	\to0.
	\]
	
	Следовательно, \(A\) --- предел компактных операторов
	по операторной норме, а значит
	\[
	\boxed{A\text{ компактен}.}
	\]
	
	\medskip
	
	\textbf{Главная идея билета в одной строке:}
	\[
	\boxed{
		\text{приближаем ядро конечными суммами}
		\Rightarrow
		\text{получаем конечномерные операторы}
		\Rightarrow
		\text{они сходятся к }A
		\Rightarrow
		A\text{ компактен}.
	}
	\]
	
	То есть пространство \(L_2([a,b]^2)\), ОНБ и ряд Фурье здесь нужны
	не ради самих себя.
	
	Они нужны только для одной цели:
	\[
	\boxed{
		\text{приблизить произвольное ядро }K(x,t)
		\text{ конечной суммой разделяющихся функций }
		\varphi_i(x)\varphi_j(t).
	}
	\]
	
	А такие конечные суммы автоматически порождают конечномерные операторы.
\end{idea}
