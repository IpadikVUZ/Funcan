\subsection{Формулировка теоремы Банаха об обратном операторе. Доказательство теоремы Банаха в случае гильбертова пространства}

\begin{theorem}[Банаха об обратном операторе]
	Пусть \(E_1,E_2\) --- банаховы пространства,
	\[
	A\in\mathcal{L}(E_1,E_2),
	\]
	и оператор \(A\) является биекцией
	\[
	A:E_1\to E_2.
	\]
	Тогда обратный оператор
	\[
	A^{-1}:E_2\to E_1
	\]
	является линейным ограниченным оператором, то есть
	\[
	A^{-1}\in\mathcal{L}(E_2,E_1).
	\]
\end{theorem}

\begin{definition}[Сопряжённый оператор в гильбертовом пространстве]
	Пусть \(H\) --- гильбертово пространство и
	\[
	A\in\mathcal{L}(H).
	\]
	Сопряжённым к \(A\) называется оператор \(A^*\in\mathcal{L}(H)\), такой что
	\[
	(Ax,y)=(x,A^*y)
	\qquad
	\forall x,y\in H.
	\]
\end{definition}

\begin{remark}[Используемые факты о сопряжённом операторе]
	В доказательстве гильбертова случая используем стандартные факты:
	\[
	(\operatorname{Im}A)^\perp=\ker A^*,
	\]
	и
	\[
	H=\overline{\operatorname{Im}A}\oplus\ker A^*.
	\]
	Применяя это к \(A^*\), получаем также
	\[
	H=\overline{\operatorname{Im}A^*}\oplus\ker A.
	\]
\end{remark}

\begin{remark}[Используемый факт из следствия билета 8.1]
	Если \(E\) --- банахово пространство и
	\[
	A\in\mathcal{L}(E)
	\]
	ограничен снизу, то
	\[
	\operatorname{Im}A
	\]
	замкнут.
\end{remark}

\begin{theorem}[Банаха об обратном операторе в гильбертовом пространстве]
	Пусть \(H\) --- гильбертово пространство,
	\[
	A\in\mathcal{L}(H),
	\]
	и \(A\) --- биекция
	\[
	A:H\to H.
	\]
	Тогда
	\[
	A^{-1}\in\mathcal{L}(H).
	\]
\end{theorem}

\begin{proof}
	Главная идея: доказать, что \(A\) ограничен снизу. Тогда по теореме из билета 8.1 обратный
	оператор будет ограничен.
	
	Напрямую ограниченность снизу для \(A\) получить неудобно. Поэтому сначала докажем её для
	сопряжённого оператора \(A^*\), а затем вернёмся к \(A\).
	
	\textbf{1. Докажем, что \(A^*\) ограничен снизу.}
	
	Так как \(A\) --- биекция, то
	\[
	\operatorname{Im}A=H.
	\]
	Поэтому
	\[
	\ker A^*=(\operatorname{Im}A)^\perp=H^\perp=\{0\}.
	\]
	Значит, если \(x\ne0\), то
	\[
	A^*x\ne0.
	\]
	
	Рассмотрим множество
	\[
	S=\{x\in H:\|A^*x\|=1\}.
	\]
	Покажем, что \(S\) ограничено. Для этого достаточно по теореме Хана доказать, что \(S\)
	слабо ограничено.
	
	\begin{itemize}
		\item \textbf{Проверим слабую ограниченность \(S\).}
		
		Возьмём произвольный \(y\in H\). Так как \(A\) --- сюръекция, существует \(z\in H\), такое что
		\[
		Az=y.
		\]
		Тогда для любого \(x\in S\)
		\[
		|(x,y)|
		=
		|(x,Az)|
		=
		|(A^*x,z)|
		\leqslant
		\|A^*x\|\cdot\|z\|.
		\]
		Но \(x\in S\), поэтому
		\[
		\|A^*x\|=1.
		\]
		Следовательно,
		\[
		|(x,y)|\leqslant \|z\|.
		\]
		Для фиксированного \(y\) число \(\|z\|\) не зависит от \(x\in S\). Значит, \(S\) слабо
		ограничено.
		
		\item \textbf{Из слабой ограниченности получим обычную ограниченность.}
		
		По теореме Хана слабая ограниченность множества эквивалентна обычной ограниченности.
		Следовательно, существует \(C>0\), такое что
		\[
		\|x\|\leqslant C
		\qquad
		\forall x\in S.
		\]
		
		\item \textbf{Переведём ограниченность \(S\) в оценку снизу для \(A^*\).}
		
		Пусть \(u\in H\), \(u\ne0\). Так как \(\ker A^*=\{0\}\), имеем
		\[
		A^*u\ne0.
		\]
		Положим
		\[
		x=\frac{u}{\|A^*u\|}.
		\]
		Тогда
		\[
		\|A^*x\|
		=
		\left\|A^*\frac{u}{\|A^*u\|}\right\|
		=
		1,
		\]
		то есть \(x\in S\). Поэтому
		\[
		\|x\|\leqslant C.
		\]
		Подставляя \(x=u/\|A^*u\|\), получаем
		\[
		\frac{\|u\|}{\|A^*u\|}\leqslant C.
		\]
		Значит,
		\[
		\|A^*u\|\geqslant \frac{1}{C}\|u\|.
		\]
		При \(u=0\) это неравенство тоже верно.
	\end{itemize}
	
	Итак, \(A^*\) ограничен снизу:
	\[
	\exists m>0
	\quad
	\forall u\in H
	\qquad
	\|A^*u\|\geqslant m\|u\|.
	\]
	
	\textbf{2. Получим, что \(A^*\) является биекцией и имеет ограниченный обратный.}
	
	Из ограниченности снизу следует, что
	\[
	\operatorname{Im}A^*
	\]
	замкнут по предыдущей лемме.
	
	С другой стороны, так как \(A\) --- биекция, то
	\[
	\ker A=\{0\}.
	\]
	По ортогональному разложению для сопряжённого оператора
	\[
	H=\overline{\operatorname{Im}A^*}\oplus\ker A.
	\]
	Так как \(\ker A=\{0\}\), получаем
	\[
	\overline{\operatorname{Im}A^*}=H.
	\]
	Но \(\operatorname{Im}A^*\) уже замкнут, следовательно
	\[
	\operatorname{Im}A^*=H.
	\]
	
	Мы знаем, что:
	\[
	\ker A^*=\{0\},
	\qquad
	\operatorname{Im}A^*=H.
	\]
	Значит, \(A^*\) --- биекция.
	
	Кроме того, \(A^*\) ограничен снизу. Поэтому по критерию из билета 8.1
	\[
	(A^*)^{-1}\in\mathcal{L}(H).
	\]
	
	\textbf{3. Получим ограниченность снизу для \(A\) через \((A^*)^{-1}\).}
	
	Мы уже доказали, что \(A^*\) --- биекция и
	\[
	(A^*)^{-1}\in\mathcal{L}(H).
	\]
	Докажем, что \(A\) ограничен снизу.
	
	Возьмём произвольный \(u\in H\). В гильбертовом пространстве
	\[
	\|u\|
	=
	\sup_{\|y\|\leqslant 1}|(u,y)|.
	\]
	Пусть \(\|y\|\leqslant 1\). Так как \(A^*\) сюръективен, существует
	\[
	z=(A^*)^{-1}y,
	\]
	то есть
	\[
	A^*z=y.
	\]
	Тогда
	\[
	|(u,y)|
	=
	|(u,A^*z)|
	=
	|(Au,z)|
	\leqslant
	\|Au\|\cdot\|z\|.
	\]
	Но
	\[
	\|z\|
	=
	\|(A^*)^{-1}y\|
	\leqslant
	\|(A^*)^{-1}\|\cdot\|y\|
	\leqslant
	\|(A^*)^{-1}\|.
	\]
	Следовательно,
	\[
	|(u,y)|
	\leqslant
	\|(A^*)^{-1}\|\cdot\|Au\|.
	\]
	Берём супремум по всем \(y\), таким что \(\|y\|\leqslant1\):
	\[
	\|u\|
	\leqslant
	\|(A^*)^{-1}\|\cdot\|Au\|.
	\]
	Значит,
	\[
	\|Au\|
	\geqslant
	\frac{1}{\|(A^*)^{-1}\|}\|u\|.
	\]
	То есть \(A\) ограничен снизу.
	
	\textbf{4. Применим критерий из билета 8.1.}
	
	Оператор \(A\) является биекцией по условию. Значит,
	\[
	A^{-1}:H\to H
	\]
	существует как линейный оператор.
	
	Так как \(A\) ограничен снизу, по критерию из билета 8.1 получаем
	\[
	A^{-1}\in\mathcal{L}(H).
	\]
\end{proof}

\begin{remark}[Формула нормы в гильбертовом пространстве]
	В гильбертовом пространстве для любого \(u\in H\)
	\[
	\|u\|
	=
	\sup_{\|y\|\leqslant 1}|(u,y)|.
	\]
	Это гильбертов аналог общего факта
	\[
	\|u\|
	=
	\sup_{\|f\|\leqslant 1}|f(u)|
	\]
	из следствия теоремы Хана--Банаха. По теореме Рисса--Фреше каждый функционал
	в гильбертовом пространстве имеет вид
	\[
	f(v)=(v,y).
	\]
	Но в гильбертовом случае эту формулу можно доказать напрямую через неравенство
	Коши--Буняковского, поэтому для этого шага не нужно отдельно использовать
	теорему Хана--Банаха.
\end{remark}
