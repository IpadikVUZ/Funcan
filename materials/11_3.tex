\subsection{Критерий принадлежности числа спектру самосопряженного оператора.
	Вещественность спектра самосопряженного оператора}

\begin{definition}[Оператор \(A_\lambda\)]
	Пусть
	\[
	A\in\mathcal{L}(H),
	\qquad
	\lambda\in\mathbb{C}.
	\]
	Обозначим
	\[
	A_\lambda=A-\lambda I.
	\]
\end{definition}

\begin{theorem}[Критерий принадлежности числа спектру]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда
	\[
	\lambda\in\rho(A)
	\Longleftrightarrow
	A_\lambda \text{ ограничен снизу},
	\]
	то есть
	\[
	\exists m>0
	\quad
	\forall x\in H
	\qquad
	\|A_\lambda x\|\geqslant m\|x\|.
	\]
	
	Кроме того,
	\[
	\lambda\in\sigma(A)
	\Longleftrightarrow
	\exists \{x_n\}\subset H:
	\quad
	\|x_n\|=1,
	\qquad
	A_\lambda x_n\to0.
	\]
\end{theorem}

\begin{idea} 
	Иными словами, $x_n$ становятся всё более похожи на собственные векторы с собственным значением $\lambda$, хотя настоящего собственного вектора может не существовать.
\end{idea}

\begin{proof}
	\textbf{1. Докажем критерий для резольвенты.}
	
	\textbf{1.1. Пусть \(\lambda\in\rho(A)\). Покажем, что \(A_\lambda\) ограничен снизу.}
	
	По определению резольвентного множества
	\[
	A_\lambda^{-1}\in\mathcal{L}(H).
	\]
	По билету 8.1 отсюда следует, что \(A_\lambda\) ограничен снизу:
	\[
	\exists m>0
	\quad
	\forall x\in H
	\qquad
	\|A_\lambda x\|\geqslant m\|x\|.
	\]
	
	\textbf{1.2. Пусть \(A_\lambda\) ограничен снизу. Покажем, что
		\(\lambda\in\rho(A)\).}
	
	Для этого нужно показать, что \(A_\lambda\) непрерывно обратим.
	
	Из ограниченности снизу по билету 8.1 получаем
	\[
	\ker A_\lambda=\{0\}
	\]
	и
	\[
	\operatorname{Im}A_\lambda
	\quad\text{замкнут}.
	\]
	
	Теперь осталось показать сюръективность. По билету 11.2
\[
H=
\overline{\operatorname{Im}A_\lambda}
\oplus_{\perp}
\ker A_\lambda.
\]
Так как
\[
\ker A_\lambda=\{0\},
\qquad
\operatorname{Im}A_\lambda
\text{ замкнут},
\]
то
\[
\operatorname{Im}A_\lambda=H.
\]
	
	Значит, \(A_\lambda\) --- биекция. Так как он ограничен снизу,
	по билету 8.1
	\[
	A_\lambda^{-1}\in\mathcal{L}(H).
	\]
	Следовательно,
	\[
	\lambda\in\rho(A).
	\]
	
	Итак,
	\[
	\boxed{
		\lambda\in\rho(A)
		\Longleftrightarrow
		A_\lambda\text{ ограничен снизу}.
	}
	\]
	
	\textbf{2. Докажем критерий для спектра.}
	
	Из первого пункта
	\[
	\lambda\in\sigma(A)
	\Longleftrightarrow
	A_\lambda\text{ не ограничен снизу}.
	\]
	
	\textbf{2.1. Пусть \(\lambda\in\sigma(A)\). Построим последовательность \(x_n\).}
	
	Так как \(A_\lambda\) не ограничен снизу,
	\[
	\forall m>0
	\quad
	\exists u\neq0:
	\qquad
	\|A_\lambda u\|<m\|u\|.
	\]
	Для
	\[
	m=\frac1n
	\]
	выберем \(u_n\neq0\), такое что
	\[
	\|A_\lambda u_n\|
	<
	\frac1n\|u_n\|.
	\]
	Положим
	\[
	x_n=\frac{u_n}{\|u_n\|}.
	\]
	Тогда
	\[
	\|x_n\|=1
	\]
	и
	\[
	\|A_\lambda x_n\|
	<
	\frac1n.
	\]
	Следовательно,
	\[
	A_\lambda x_n\to0.
	\]
	
	\textbf{2.2. Пусть существует \(x_n\), такое что
		\(\|x_n\|=1\) и \(A_\lambda x_n\to0\).
		Покажем, что \(\lambda\in\sigma(A)\).}
	
	Если бы \(A_\lambda\) был ограничен снизу, то для некоторого \(m>0\)
	\[
	\|A_\lambda x_n\|
	\geqslant
	m\|x_n\|
	=
	m
	\]
	для всех \(n\), что противоречит
	\[
	A_\lambda x_n\to0.
	\]
	Значит, \(A_\lambda\) не ограничен снизу, поэтому
	\[
	\lambda\in\sigma(A).
	\]
	
	Итак,
	\[
	\boxed{
		\lambda\in\sigma(A)
		\Longleftrightarrow
		\exists x_n:
		\quad
		\|x_n\|=1,
		\qquad
		A_\lambda x_n\to0.
	}
	\]
\end{proof}

\begin{theorem}[Вещественность спектра]
	Пусть \(H\) --- комплексное гильбертово пространство,
	\[
	A\in\mathcal{L}(H)
	\]
	--- самосопряженный оператор. Тогда
	\[
	\sigma(A)\subset\mathbb{R}.
	\]
\end{theorem}

\begin{proof}
	Докажем равносильное утверждение:
	\[
	\lambda\notin\mathbb{R}
	\Longrightarrow
	\lambda\in\rho(A).
	\]
	
	Пусть
	\[
	\lambda=\mu+i\nu,
	\qquad
	\mu,\nu\in\mathbb{R},
	\qquad
	\nu\neq0.
	\]
	По критерию принадлежности числа резольвенте достаточно показать,
	что \(A_\lambda\) ограничен снизу.
	
	Положим
	\[
	A_\mu=A-\mu I.
	\]
	Так как \(A\) самосопряжен и \(\mu\in\mathbb{R}\),
	оператор \(A_\mu\) самосопряжен.
	
	Для любого \(x\in H\)
	\[
	A_\lambda x=A_\mu x-i\nu x,
	\]
	поэтому
	\[
	\begin{aligned}
		\|A_\lambda x\|^2
		&=
		\|A_\mu x-i\nu x\|^2
		\\
		&=
		\|A_\mu x\|^2
		-i\nu(x,A_\mu x)
		+i\nu(A_\mu x,x)
		+\nu^2\|x\|^2.
	\end{aligned}
	\]
	
	Так как \(A_\mu\) самосопряжен,
	\[
	(x,A_\mu x)=(A_\mu x,x).
	\]
	Поэтому смешанные слагаемые сокращаются:
	\[
	\|A_\lambda x\|^2
	=
	\|A_\mu x\|^2+\nu^2\|x\|^2
	\geqslant
	\nu^2\|x\|^2.
	\]
	Значит,
	\[
	\|A_\lambda x\|
	\geqslant
	|\nu|\|x\|.
	\]
	
	То есть \(A_\lambda\) ограничен снизу, поэтому
	\[
	\lambda\in\rho(A).
	\]
	Следовательно,
	\[
	\boxed{\sigma(A)\subset\mathbb{R}}.
	\]
\end{proof}
